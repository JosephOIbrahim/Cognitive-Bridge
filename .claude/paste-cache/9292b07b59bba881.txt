## ARCHITECTURE DECISION (NON-NEGOTIABLE)

The HDA creation flow is a **MODE within the existing chat panel**, NOT a separate panel.

```
SYNAPSE Panel (chat_panel.py)
└─ QStackedWidget [mode_stack]
   ├─ index 0: Chat Mode (existing — DO NOT MODIFY its internals)
   └─ index 1: HDA Mode
      └─ QStackedWidget [hda_stack]
         ├─ index 0: DescribeView  (prompt input)
         ├─ index 1: BuildingView  (progress)
         └─ index 2: ResultView    (outcome)
```

Toolbar gets a mode toggle: `[Chat] [Create HDA]`
Three-state flow: **Describe → Building → Result**
Monochromatic Pentagram styling. SIGNAL cyan (#00FFD1) as sole accent.

**Task T4: Define QStackedWidget transition helpers**

```python
# Add to styles.py

def animate_stack_transition(stacked_widget, new_index, duration_ms=200):
    """Smooth opacity fade between QStackedWidget pages.
    
    Note: QStackedWidget doesn't support slide animations natively.
    We use opacity fade via QGraphicsOpacityEffect.
    """
    from PySide2 import QtWidgets, QtCore
    
    current = stacked_widget.currentWidget()
    target = stacked_widget.widget(new_index)
    
    if current == target:
        return
    
    # Apply opacity effect to target
    effect = QtWidgets.QGraphicsOpacityEffect(target)
    target.setGraphicsEffect(effect)
    effect.setOpacity(0.0)
    
    # Switch immediately (target starts invisible)
    stacked_widget.setCurrentIndex(new_index)
    
    # Animate opacity 0 → 1
    anim = QtCore.QPropertyAnimation(effect, b"opacity")
    anim.setDuration(duration_ms)
    anim.setStartValue(0.0)
    anim.setEndValue(1.0)
    anim.setEasingCurve(QtCore.QEasingCurve.OutCubic)
    
    # Clean up effect after animation
    def cleanup():
        target.setGraphicsEffect(None)
    anim.finished.connect(cleanup)
    
    # Store reference to prevent GC
    stacked_widget._current_anim = anim
    anim.start()
```

```python
# Add new Qt signals (alongside existing ones):
# In the bridge class:
from PySide2.QtCore import Signal

class SynapseBridge(QtCore.QObject):
    # Existing signals
    message_received = Signal(dict)     # chat messages
    
    # NEW: HDA-specific signals
    hda_progress = Signal(dict)         # HDA build progress updates
    hda_result = Signal(dict)           # HDA build completion
    connection_changed = Signal(bool)   # connection state (may already exist)
    
    def _dispatch_message(self, data: dict):
        """Route incoming WebSocket message to appropriate signal."""
        msg_type = data.get("msg_type", "chat")  # backward compat
        
        if msg_type == "hda_progress":
            self.hda_progress.emit(data)
        elif msg_type == "hda_result":
            self.hda_result.emit(data)
        else:
            # Default: treat as chat message (backward compatible)
            self.message_received.emit(data)
```

```python
        # Checkboxes

        if not prompt:

        context = self.context_combo.currentText()
        options = {
            "include_help": self.chk_help.isChecked(),
            "add_to_toolbar": self.chk_toolbar.isChecked(),
        }
        self.generate_requested.emit(prompt, context, options)
    
    def reset(self):
        """Clear for next HDA creation."""
        self.prompt_input.clear()
        self.context_combo.setCurrentText("SOP")
        self.chk_help.setChecked(True)
        self.chk_toolbar.setChecked(False)
```

```python
# Create synapse/panel/hda_controller.py

from PySide2 import QtCore
from ..routing.hda_recipes import HDA_RECIPES

class HdaController(QtCore.QObject):
    """Orchestrates prompt → recipe → build → validate → report."""
    
    progress = QtCore.Signal(str, int, str)  # stage_name, pct, detail
    result = QtCore.Signal(dict)              # success/failure result
    error = QtCore.Signal(str)                # error message
    
    def __init__(self, bridge, parent=None):
        super().__init__(parent)
        self._bridge = bridge
        self._active = False
        
        # Connect bridge signals
        bridge.hda_progress.connect(self._on_bridge_progress)
        bridge.hda_result.connect(self._on_bridge_result)
    
    def execute(self, prompt: str, context: str, options: dict):
        """Start HDA creation from a natural language prompt."""
        if self._active:
            self.error.emit("HDA creation already in progress")
            return
        
        self._active = True
        self.progress.emit("parsing_prompt", 5, "Understanding your description...")
        
        # Select recipe based on prompt + context
        recipe = self._select_recipe(prompt, context)
        
        if recipe is None:
            self.progress.emit("selecting_recipe", 10, "No matching recipe found")
            self._active = False
            self.result.emit({
                "success": False,
                "error": f"No recipe matches: '{prompt[:50]}...' in {context} context",
                "detail": "Try a more specific description or different context.",
            })
            return
        
        self.progress.emit("selecting_recipe", 15, f"Using recipe: {recipe['name']}")
        
        # Send to Houdini via MCP tool chain
        self._send_hda_request(recipe, options)
    
    def _select_recipe(self, prompt: str, context: str) -> dict:
        """Match prompt to best recipe. Simple keyword matching for now."""
        prompt_lower = prompt.lower()
        
        # Score each recipe by keyword overlap
        best = None
        best_score = 0
        for key, recipe in HDA_RECIPES.items():
            if recipe.get("context", "SOP") != context:
                continue
            
            # Score by keyword presence
            score = 0
            desc_words = recipe["description"].lower().split()
            for word in desc_words:
                if len(word) > 3 and word in prompt_lower:
                    score += 1
            
            name_words = recipe["name"].lower().replace("_", " ").split()
            for word in name_words:
                if word in prompt_lower:
                    score += 2  # Name matches weighted higher
            
            if score > best_score:
                best_score = score
                best = recipe
        
        return best
    
    def _send_hda_request(self, recipe: dict, options: dict):
        """Send the HDA build request through the bridge."""
        # This calls the hda_package MCP tool via WebSocket
        request = {
            "tool": "hda_package",
            "params": {
                "recipe": recipe,
                "include_help": options.get("include_help", True),
                "add_to_toolbar": options.get("add_to_toolbar", False),
            }
        }
        self._bridge.send(request)
    
    def _on_bridge_progress(self, data: dict):
        """Handle progress updates from Houdini."""
        self.progress.emit(
            data.get("stage", ""),
            data.get("progress_pct", 0),
            data.get("detail", ""),
        )
    
    def _on_bridge_result(self, data: dict):
        """Handle completion from Houdini."""
        self._active = False
        self.result.emit(data)
    
    def cancel(self):
        """Cancel in-progress HDA creation."""
        if self._active:
            self._bridge.send({"tool": "hda_cancel"})
            self._active = False
```
