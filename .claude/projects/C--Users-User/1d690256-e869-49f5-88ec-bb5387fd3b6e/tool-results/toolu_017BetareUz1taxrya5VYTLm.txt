     1→"""
     2→Synapse Recipe Registry
     3→
     4→Pre-built, pre-approved multi-step operation templates.
     5→Recipes execute at Tier 0 speed by expanding triggers into
     6→sequences of SynapseCommands without any LLM involvement.
     7→
     8→Artists can register custom recipes via registry.register().
     9→"""
    10→
    11→import json
    12→import re
    13→from dataclasses import dataclass, field
    14→from typing import Dict, List, Optional, Tuple, Any
    15→
    16→from ..core.protocol import SynapseCommand
    17→from ..core.gates import GateLevel
    18→from ..core.determinism import deterministic_uuid

    20→
    21→@dataclass
    22→class RecipeStep:
    23→    """A single step in a recipe template."""
    24→    action: str                        # CommandType value
    25→    payload_template: Dict[str, Any]   # Values may contain {placeholders}
    26→    gate_level: GateLevel = GateLevel.REVIEW
    27→    output_var: str = ""               # If set, response data stored as ${var}.{field}
    28→
    29→    def instantiate(self, params: Dict[str, str]) -> SynapseCommand:
    30→        """Fill placeholders and return a SynapseCommand."""
    31→        payload = {}
    32→        for key, value in sorted(self.payload_template.items()):
    33→            if isinstance(value, str) and "{" in value:
    34→                payload[key] = value.format(**params)
    35→            else:
    36→                payload[key] = value
    37→
    38→        return SynapseCommand(
    39→            type=self.action,
    40→            id=deterministic_uuid(
    41→                f"recipe:{self.action}:{json.dumps(payload, sort_keys=True, default=str)}", "cmd"
    42→            ),
    43→            payload=payload,
    44→        )
    45→
    46→    def instantiate_with_vars(self, variables: Dict[str, str]) -> SynapseCommand:
    47→        """Fill placeholders using accumulated variables (supports $var.field syntax)."""
    48→        payload = {}
    49→        for key, value in sorted(self.payload_template.items()):
    50→            if isinstance(value, str):
    51→                # Replace $var.field references first
    52→                resolved = value
    53→                for var_key in sorted(variables.keys(), key=len, reverse=True):
    54→                    resolved = resolved.replace(var_key, str(variables[var_key]))
    55→                # Then standard {param} placeholders
    56→                if "{" in resolved:
    57→                    resolved = resolved.format(**{
    58→                        k: v for k, v in sorted(variables.items())
    59→                        if not k.startswith("$")
    60→                    })
    61→                payload[key] = resolved
    62→            else:
    63→                payload[key] = value
    64→
    65→        return SynapseCommand(
    66→            type=self.action,
    67→            id=deterministic_uuid(
    68→                f"recipe:{self.action}:{json.dumps(payload, sort_keys=True, default=str)}", "cmd"
    69→            ),
    70→            payload=payload,
    71→        )

    74→@dataclass
    75→class Recipe:
    76→    """A pre-built multi-step operation template."""
    77→    name: str
    78→    description: str
    79→    triggers: List[str]                # Regex patterns (case-insensitive)
    80→    parameters: List[str]              # Named capture groups expected
    81→    steps: List[RecipeStep]
    82→    gate_level: GateLevel = GateLevel.REVIEW
    83→    category: str = "general"
    84→
    85→    _compiled: List["re.Pattern[str]"] = field(default_factory=list, repr=False)
    86→
    87→    def __post_init__(self):
    88→        self._compiled = [
    89→            re.compile(t, re.IGNORECASE) for t in self.triggers
    90→        ]
    91→
    92→    def match(self, text: str) -> Optional[Dict[str, str]]:
    93→        """
    94→        Try to match text against trigger patterns.
    95→
    96→        Returns extracted parameters on match, None otherwise.
    97→        """
    98→        text = text.strip()
    99→        for pattern in self._compiled:
   100→            m = pattern.match(text)
   101→            if m:
   102→                params = m.groupdict()
   103→                # Fill defaults for missing optional params
   104→                for p in self.parameters:
   105→                    if p not in params or params[p] is None:
   106→                        params[p] = ""
   107→                return params
   108→        return None
   109→
   110→    def instantiate(self, params: Dict[str, str]) -> List[SynapseCommand]:
   111→        """Fill templates with extracted params, return command list."""
   112→        return [step.instantiate(params) for step in self.steps]
   113→
   114→    def execute(
   115→        self,
   116→        params: Dict[str, str],
   117→        command_fn,
   118→    ) -> List[Tuple[SynapseCommand, Any]]:
   119→        """Execute recipe with data flow between steps.
   120→
   121→        Each step can declare output_var. After execution, response data
   122→        fields become ${var}.{field} variables available to later steps.
   123→
   124→        Args:
   125→            params: Initial parameters from trigger match.
   126→            command_fn: Callable(SynapseCommand) -> SynapseResponse.
   127→
   128→        Returns:
   129→            List of (command, response) tuples.
   130→        """
   131→        variables: Dict[str, str] = dict(params)
   132→        results: List[Tuple[SynapseCommand, Any]] = []
   133→        for step in self.steps:
   134→            cmd = step.instantiate_with_vars(variables)
   135→            resp = command_fn(cmd)
   136→            results.append((cmd, resp))
   137→            if step.output_var and hasattr(resp, "data") and isinstance(resp.data, dict):
   138→                for k, v in sorted(resp.data.items()):  # sorted: He2025
   139→                    variables[f"${step.output_var}.{k}"] = str(v)
   140→        return results
   141→
   142→
   143→class RecipeRegistry:
   144→    """
   145→    Registry of pre-built recipes.
   146→
   147→    Matches input text against all registered recipes.
   148→    Built-in recipes are registered on construction.
   149→    """
   150→
   151→    def __init__(self, include_builtins: bool = True):
   152→        self._recipes: List[Recipe] = []
   153→        if include_builtins:
   154→            self._register_builtins()
   155→
   156→    def register(self, recipe: Recipe):
   157→        """Register a custom recipe."""
   158→        self._recipes.append(recipe)
   159→
   160→    def match(self, text: str) -> Optional[Tuple[Recipe, Dict[str, str]]]:
   161→        """
   162→        Try to match text against all registered recipes.
   163→
   164→        Returns (recipe, params) on match, None otherwise.
   165→        """
   166→        text = text.strip()
   167→        for recipe in self._recipes:
   168→            params = recipe.match(text)
   169→            if params is not None:
   170→                return (recipe, params)
   171→        return None
   172→
   173→    @property
   174→    def recipes(self) -> List[Recipe]:
   175→        """List of registered recipes."""
   176→        return list(self._recipes)
