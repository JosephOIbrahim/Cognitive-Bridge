    17→    - get_running_loop() everywhere (deprecated get_event_loop removed)
    18→
    19→Architecture:
    20→    Claude Desktop  <-[stdio/JSON-RPC]->  mcp_server.py  <-[WebSocket]->  Synapse (Houdini)
    21→
    22→Install: pip install mcp websockets
    23→Run: python mcp_server.py

    52→def _get_auth_key() -> str | None:
    53→    """Get API key for authenticating with Synapse server.
    54→
    55→    Sources (checked in order):
    56→    1. SYNAPSE_API_KEY environment variable
    57→    2. ~/.synapse/auth.key file (first non-empty, non-comment line)
    58→    3. None (auth disabled)
    59→    """
    60→    env_key = os.environ.get("SYNAPSE_API_KEY", "").strip()
    61→    if env_key:
    62→        return env_key
    63→    key_path = os.path.join(os.path.expanduser("~"), ".synapse", "auth.key")
    64→    try:
    65→        if os.path.exists(key_path):
    66→            with open(key_path, encoding="utf-8") as f:
    67→                for line in f:
    68→                    line = line.strip()
    69→                    if line and not line.startswith("#"):
    70→                        return line
    71→    except OSError:
    72→        pass
    73→    return None
    74→

    76→async def _auth_handshake(ws) -> None:
    77→    """Handle auth handshake if server requires it.
    78→
    79→    v2 optimization: If no local key is configured, skip the 2s recv wait
    80→    entirely. If the server requires auth, the first command will fail with
    81→    a clear error — acceptable trade-off for eliminating 2s dead time on
    82→    every reconnect when auth is disabled (the common case).
    83→    """
    84→    key = _get_auth_key()
    85→    if not key:
    86→        # No key configured — skip the recv wait entirely (saves ~2s)
    87→        return
    88→
    89→    # Key exists — wait for auth_required from server
    90→    try:
    91→        raw = await asyncio.wait_for(ws.recv(), timeout=2.0)
    92→        msg = _loads(raw)
    93→    except (asyncio.TimeoutError, Exception):
    94→        # Server didn't send auth_required — auth not enabled server-side
    95→        return
    96→
    97→    if msg.get("type") != "auth_required":
    98→        logger.warning("Expected auth_required, got %s (auth may be disabled)", msg.get("type"))
    99→        return
   100→
   101→    await ws.send(_dumps({
   102→        "type": "authenticate",
   103→        "id": "auth-handshake",
   104→        "payload": {"key": key},
   105→    }))
   106→
   107→    # Wait for auth_success or auth_failed
   108→    try:
   109→        raw = await asyncio.wait_for(ws.recv(), timeout=5.0)
   110→        response = _loads(raw)
   111→    except asyncio.TimeoutError:
   112→        raise ConnectionError("Synapse server didn't respond to authentication within 5s")
   113→
   114→    if response.get("type") == "auth_failed":
   115→        raise ConnectionError(
   116→            f"Authentication failed: {response.get('error', 'invalid API key')}. "
   117→            "Check your SYNAPSE_API_KEY or ~/.synapse/auth.key"
   118→        )
   119→
   120→    if response.get("type") == "auth_success":
   121→        logger.info("Authenticated with Synapse server")
   122→        return
   123→
   124→    logger.warning("Unexpected auth response type: %s", response.get("type"))
   125→

   187→
   188→_ws_connection = None
   189→_ws_lock = asyncio.Lock()
   190→_pending: dict[str, asyncio.Future] = {}
   191→_recv_task: asyncio.Task | None = None
   192→
   193→
   194→def _is_connected() -> bool:
   195→    """Check if the WebSocket connection is open."""
   196→    try:
   197→        return _ws_connection is not None and _ws_connection.state == websockets.connection.State.OPEN
   198→    except (AttributeError, Exception):
   199→        return False
   200→
   201→
   202→async def _recv_loop():
   203→    """Background task: read messages and dispatch to pending futures by ID."""
   204→    global _ws_connection, _recv_task
   205→    try:
   206→        while _ws_connection is not None:
   207→            try:
   208→                raw = await _ws_connection.recv()
   209→            except Exception as e:
   210→                # Connection lost — signal all pending futures
   211→                logger.warning("Recv loop: connection lost (%s)", e)
   212→                _signal_all_pending(ConnectionError(f"Connection lost: {e}"))
   213→                break
   214→
   215→            try:
   216→                response = _loads(raw)
   217→            except Exception:
   218→                continue  # Malformed message — skip
   219→
   220→            resp_id = response.get("id")
   221→            future = _pending.pop(resp_id, None) if resp_id else None
   222→            if future and not future.done():
   223→                future.set_result(response)
   224→            elif resp_id:
   225→                logger.warning("No pending future for response ID %s (discarding)", resp_id)
   226→    finally:
   227→        _recv_task = None
   228→
   229→
   230→def _signal_all_pending(exc: Exception):
   231→    """Signal all pending futures with an exception (connection lost)."""
   232→    for fut in _pending.values():
   233→        if not fut.done():
   234→            fut.set_exception(exc)
   235→    _pending.clear()
   236→
   237→
   238→def _start_recv_loop():
   239→    """Ensure the background recv loop is running."""
   240→    global _recv_task
   241→    if _recv_task is None or _recv_task.done():
   242→        _recv_task = asyncio.get_running_loop().create_task(_recv_loop())
   243→
   244→
   245→async def _get_connection():
   246→    """Get or create a persistent WebSocket connection.
   247→
   248→    v2: Lock-free fast path — volatile check before acquiring the lock.
   249→    The common case (connection alive) skips the lock entirely.
   250→    Double-check inside lock for safety.
   251→    """
   252→    global _ws_connection, _recv_task
   253→
   254→    # Lock-free fast path: if connected, return immediately
   255→    if _is_connected():
   256→        return _ws_connection
   257→
   258→    async with _ws_lock:
   259→        # Double-check after acquiring lock
   260→        if _is_connected():
   261→            return _ws_connection
   262→
   263→        # Force cleanup of stale state — cancel lingering recv task
   264→        # to prevent the race where _start_recv_loop() sees the old
   265→        # task as "not done" and skips creating a new recv loop.
   266→        _ws_connection = None
   267→        if _recv_task is not None and not _recv_task.done():
   268→            _recv_task.cancel()
   269→            try:
   270→                await _recv_task
   271→            except (asyncio.CancelledError, Exception):
   272→                pass
   273→        _recv_task = None
   274→        # Clear any orphaned futures from the dead connection
   275→        _signal_all_pending(ConnectionError("Connection recycled"))
   276→
   277→        last_error = None
   278→        for attempt in range(MAX_RETRIES):
   279→            try:
   280→                _ws_connection = await websockets.connect(
   281→                    SYNAPSE_URL,
   282→                    open_timeout=3.0,
   283→                    close_timeout=5.0,
   284→                    ping_interval=None,
   285→                    compression=None,
   286→                    max_size=None,  # Skip frame size validation on localhost
   287→                )
   288→                await _auth_handshake(_ws_connection)
   289→                _start_recv_loop()
   290→                logger.info("Connected to Synapse at %s", SYNAPSE_URL)
   291→                return _ws_connection
   292→            except Exception as e:
   293→                last_error = e
   294→                if attempt < MAX_RETRIES - 1:
   295→                    await asyncio.sleep(RETRY_DELAY * (attempt + 1))
   296→
   297→        raise ConnectionError(
   298→            f"Couldn't connect to Synapse at {SYNAPSE_URL} after {MAX_RETRIES} attempts: {last_error}\n\n"
   299→            "Houdini might not be running, or the Synapse server hasn't started yet. "
   300→            "Launch Houdini and start the Synapse server from the Python Panel."
   301→        )
   302→
   303→
   304→async def send_command(cmd_type: str, payload: dict | None = None) -> dict:
   305→    """
   306→    Send a SynapseCommand over WebSocket and return the response data.
   307→
   308→    Supports true parallel dispatch — multiple concurrent send_command
   309→    calls share a single recv loop that routes responses by ID.
   310→    """
   311→    command_id = _cmd_id(cmd_type, payload)
   312→    command = {
   313→        "type": cmd_type,
   314→        "id": command_id,
   315→        "payload": payload or {},
   316→        "sequence": 0,
   317→        "timestamp": time.time(),
   318→        "protocol_version": PROTOCOL_VERSION,
   319→    }
   320→
   321→    cmd_timeout = _SLOW_COMMANDS.get(cmd_type, COMMAND_TIMEOUT)
   322→    last_err = None
   323→
   324→    for _attempt in range(2):  # One transparent retry on connection failure
   325→        ws = await _get_connection()
   326→
   327→        # Register a future for this command's response
   328→        loop = asyncio.get_running_loop()
   329→        future: asyncio.Future = loop.create_future()
   330→        _pending[command_id] = future
   331→
   332→        try:
   333→            await ws.send(_dumps(command))
   334→
   335→            # Direct wait on future set — avoids asyncio.wait_for's internal task overhead
   336→            done, _ = await asyncio.wait({future}, timeout=cmd_timeout)
   337→            if not done:
   338→                # Timeout — future still pending
   339→                _pending.pop(command_id, None)
   340→                future.cancel()
   341→                raise asyncio.TimeoutError()
   342→            response = future.result()
   343→            break  # Success
   344→
   345→        except asyncio.TimeoutError:
   346→            _pending.pop(command_id, None)
   347→            # Close stale connection on timeout
   348→            global _ws_connection
   349→            try:
   350→                if ws:
   351→                    await ws.close()
   352→            except Exception:
   353→                pass
   354→            _ws_connection = None
   355→            raise TimeoutError(
   356→                f"The {cmd_type} command took too long to respond \u2014 "
   357→                "Houdini may be busy with a heavy operation"
   358→            )
   359→        except ConnectionError:
   360→            # Future was signaled by _recv_loop disconnect
   361→            _pending.pop(command_id, None)
   362→            try:
   363→                if ws:
   364→                    await ws.close()
   365→            except Exception:
   366→                pass
   367→            _ws_connection = None
   368→            last_err = ConnectionError(f"Connection lost during {cmd_type}")
   369→            logger.warning("Connection lost during %s, reconnecting...", cmd_type)
   370→        except Exception as e:
   371→            _pending.pop(command_id, None)
   372→            try:
   373→                if ws:
   374→                    await ws.close()
   375→            except Exception:
   376→                pass
   377→            _ws_connection = None
   378→            last_err = e
   379→            logger.warning("Connection lost during %s, reconnecting... (%s)", cmd_type, e)
   380→    else:
   381→        raise ConnectionError(
   382→            f"Lost connection while sending {cmd_type} and couldn't reconnect: {last_err}"
   383→        )
   384→
   385→    if not response.get("success", False):
   386→        error_msg = response.get("error", "Something went wrong on the Synapse side")
   387→        data = response.get("data") or {}
   388→        if isinstance(data, dict) and "retry_after" in data:
   389→            error_msg += f" (retry after {data['retry_after']}s)"
   390→        raise RuntimeError(error_msg)
   391→
   392→    return response.get("data", {})

  2049→def _execute_python_payload(args: dict) -> dict:
  2050→    # MCP uses 'code', Synapse handler resolves 'content'
  2051→    p = {"content": args["code"]}
  2052→    if "dry_run" in args:
  2053→        p["dry_run"] = args["dry_run"]
  2054→    if "atomic" in args:
  2055→        p["atomic"] = args["atomic"]

  2060→    p = {}
  2061→    if "node" in args:
  2062→        p["node"] = args["node"]
  2063→    return p

  2065→
  2066→def _decide_payload(args: dict) -> dict:
  2067→    p = {"decision": args["decision"]}
  2068→    if "reasoning" in args:
  2069→        p["reasoning"] = args["reasoning"]
  2070→    if "alternatives" in args:
  2071→        alt = args["alternatives"]
  2072→        p["alternatives"] = [a.strip() for a in alt.split(",") if a.strip()] if isinstance(alt, str) else alt
  2073→    return p

  2077→    p = {"content": args["content"]}

  2079→        p["memory_type"] = args["memory_type"]

  2175→@server.call_tool()
  2176→async def call_tool(name: str, arguments: dict):
  2177→    """Dispatch MCP tool call to Synapse via WebSocket."""
  2178→    if name not in TOOL_DISPATCH:
  2179→        return [TextContent(type="text", text=f"I don't recognize the tool '{name}' \u2014 check the available tools list")]
  2180→
  2181→    cmd_type, build_payload = TOOL_DISPATCH[name]
  2182→
  2183→    try:
  2184→        payload = build_payload(arguments)
  2185→        data = await send_command(cmd_type, payload)
  2186→
  2187→        # Image-producing tools: read file and return as ImageContent
  2188→        if name in ("houdini_capture_viewport", "houdini_render"):
  2189→            import base64
  2190→            image_path = data.get("image_path", "")
  2191→            try:
  2192→                with open(image_path, "rb") as f:
  2193→                    b64 = base64.b64encode(f.read()).decode("utf-8")
  2194→                mime = "image/jpeg" if data.get("format") == "jpeg" else "image/png"
  2195→                meta = {
  2196→                    "width": data.get("width"),
  2197→                    "height": data.get("height"),
  2198→                    "format": data.get("format"),
  2199→                }
  2200→                # Include render-specific metadata
  2201→                if "engine" in data:
  2202→                    meta["rop"] = data.get("rop")
  2203→                    meta["rop_type"] = data.get("rop_type")
  2204→                    meta["engine"] = data.get("engine")
  2205→                return [
  2206→                    ImageContent(type="image", data=b64, mimeType=mime),
  2207→                    TextContent(type="text", text=_dumps_str(meta)),
  2208→                ]
