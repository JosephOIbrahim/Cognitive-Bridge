     3→Synapse MCP Server v2 — Bridge between Claude Desktop and Houdini via WebSocket.
     4→
     5→Connects Claude Desktop (stdio/JSON-RPC) to the Synapse WebSocket server

    10→    - Auth handshake: skip 2s recv wait when no local key configured

    15→    - Fire-and-forget warmup: stdio server starts accepting immediately
    16→    - max_size=None: skip frame validation on localhost
    17→    - get_running_loop() everywhere (deprecated get_event_loop removed)
    18→
    19→Architecture:
    20→    Claude Desktop  <-[stdio/JSON-RPC]->  mcp_server.py  <-[WebSocket]->  Synapse (Houdini)
    21→
    22→Install: pip install mcp websockets
    23→Run: python mcp_server.py
    24→"""

    52→def _get_auth_key() -> str | None:
    53→    """Get API key for authenticating with Synapse server.
    54→
    55→    Sources (checked in order):
    56→    1. SYNAPSE_API_KEY environment variable
    57→    2. ~/.synapse/auth.key file (first non-empty, non-comment line)
    58→    3. None (auth disabled)
    59→    """
    60→    env_key = os.environ.get("SYNAPSE_API_KEY", "").strip()
    61→    if env_key:
    62→        return env_key
    63→    key_path = os.path.join(os.path.expanduser("~"), ".synapse", "auth.key")
    64→    try:
    65→        if os.path.exists(key_path):
    66→            with open(key_path, encoding="utf-8") as f:
    67→                for line in f:
    68→                    line = line.strip()
    69→                    if line and not line.startswith("#"):
    70→                        return line
    71→    except OSError:
    72→        pass
    73→    return None

    76→async def _auth_handshake(ws) -> None:
    77→    """Handle auth handshake if server requires it.
    78→
    79→    v2 optimization: If no local key is configured, skip the 2s recv wait
    80→    entirely. If the server requires auth, the first command will fail with
    81→    a clear error — acceptable trade-off for eliminating 2s dead time on
    82→    every reconnect when auth is disabled (the common case).
    83→    """
    84→    key = _get_auth_key()
    85→    if not key:
    86→        # No key configured — skip the recv wait entirely (saves ~2s)
    87→        return
    88→
    89→    # Key exists — wait for auth_required from server
    90→    try:
    91→        raw = await asyncio.wait_for(ws.recv(), timeout=2.0)
    92→        msg = _loads(raw)
    93→    except (asyncio.TimeoutError, Exception):
    94→        # Server didn't send auth_required — auth not enabled server-side
    95→        return
    96→
    97→    if msg.get("type") != "auth_required":
    98→        logger.warning("Expected auth_required, got %s (auth may be disabled)", msg.get("type"))
    99→        return
   100→
   101→    await ws.send(_dumps({
   102→        "type": "authenticate",
   103→        "id": "auth-handshake",
   104→        "payload": {"key": key},
   105→    }))
   106→
   107→    # Wait for auth_success or auth_failed
   108→    try:
   109→        raw = await asyncio.wait_for(ws.recv(), timeout=5.0)
   110→        response = _loads(raw)
   111→    except asyncio.TimeoutError:
   112→        raise ConnectionError("Synapse server didn't respond to authentication within 5s")
   113→
   114→    if response.get("type") == "auth_failed":
   115→        raise ConnectionError(
   116→            f"Authentication failed: {response.get('error', 'invalid API key')}. "
   117→            "Check your SYNAPSE_API_KEY or ~/.synapse/auth.key"
   118→        )
   119→
   120→    if response.get("type") == "auth_success":
   121→        logger.info("Authenticated with Synapse server")
   122→        return
   123→
   124→    logger.warning("Unexpected auth response type: %s", response.get("type"))

   179→
   180→_ws_connection = None
   181→_ws_lock = asyncio.Lock()
   182→_pending: dict[str, asyncio.Future] = {}
   183→_recv_task: asyncio.Task | None = None
   184→
   185→
   186→def _is_connected() -> bool:
   187→    """Check if the WebSocket connection is open."""
   188→    try:
   189→        return _ws_connection is not None and _ws_connection.state == websockets.connection.State.OPEN
   190→    except (AttributeError, Exception):
   191→        return False
   192→
   193→
   194→async def _recv_loop():
   195→    """Background task: read messages and dispatch to pending futures by ID."""
   196→    global _ws_connection, _recv_task
   197→    try:
   198→        while _ws_connection is not None:
   199→            try:
   200→                raw = await _ws_connection.recv()
   201→            except Exception as e:
   202→                # Connection lost — signal all pending futures
   203→                logger.warning("Recv loop: connection lost (%s)", e)
   204→                _signal_all_pending(ConnectionError(f"Connection lost: {e}"))
   205→                break
   206→
   207→            try:
   208→                response = _loads(raw)
   209→            except Exception:
   210→                continue  # Malformed message — skip
   211→
   212→            resp_id = response.get("id")
   213→            future = _pending.pop(resp_id, None) if resp_id else None
   214→            if future and not future.done():
   215→                future.set_result(response)
   216→            elif resp_id:
   217→                logger.warning("No pending future for response ID %s (discarding)", resp_id)
   218→    finally:
   219→        _recv_task = None
   220→
   221→
   222→def _signal_all_pending(exc: Exception):
   223→    """Signal all pending futures with an exception (connection lost)."""
   224→    for fut in _pending.values():
   225→        if not fut.done():
   226→            fut.set_exception(exc)
   227→    _pending.clear()
   228→
   229→
   230→def _start_recv_loop():
   231→    """Ensure the background recv loop is running."""
   232→    global _recv_task
   233→    if _recv_task is None or _recv_task.done():
   234→        _recv_task = asyncio.get_running_loop().create_task(_recv_loop())
   235→
   236→
   237→async def _get_connection():
   238→    """Get or create a persistent WebSocket connection.
   239→
   240→    v2: Lock-free fast path — volatile check before acquiring the lock.
   241→    The common case (connection alive) skips the lock entirely.
   242→    Double-check inside lock for safety.
   243→    """
   244→    global _ws_connection, _recv_task
   245→
   246→    # Lock-free fast path: if connected, return immediately
   247→    if _is_connected():
   248→        return _ws_connection
   249→
   250→    async with _ws_lock:
   251→        # Double-check after acquiring lock
   252→        if _is_connected():
   253→            return _ws_connection
   254→
   255→        # Force cleanup of stale state — cancel lingering recv task
   256→        # to prevent the race where _start_recv_loop() sees the old
   257→        # task as "not done" and skips creating a new recv loop.
   258→        _ws_connection = None
   259→        if _recv_task is not None and not _recv_task.done():
   260→            _recv_task.cancel()
   261→            try:
   262→                await _recv_task
   263→            except (asyncio.CancelledError, Exception):
   264→                pass
   265→        _recv_task = None
   266→        # Clear any orphaned futures from the dead connection
   267→        _signal_all_pending(ConnectionError("Connection recycled"))
   268→
   269→        last_error = None
   270→        for attempt in range(MAX_RETRIES):
   271→            try:
   272→                _ws_connection = await websockets.connect(
   273→                    SYNAPSE_URL,
   274→                    open_timeout=3.0,
   275→                    close_timeout=1.0,
   276→                    ping_interval=None,
   277→                    compression=None,
   278→                    max_size=None,  # Skip frame size validation on localhost
   279→                )
   280→                await _auth_handshake(_ws_connection)
   281→                _start_recv_loop()
   282→                logger.info("Connected to Synapse at %s", SYNAPSE_URL)
   283→                return _ws_connection
   284→            except Exception as e:
   285→                last_error = e
   286→                if attempt < MAX_RETRIES - 1:
   287→                    await asyncio.sleep(RETRY_DELAY * (attempt + 1))
   288→
   289→        raise ConnectionError(
   290→            f"Couldn't connect to Synapse at {SYNAPSE_URL} after {MAX_RETRIES} attempts: {last_error}\n\n"
   291→            "Houdini might not be running, or the Synapse server hasn't started yet. "
   292→            "Launch Houdini and start the Synapse server from the Python Panel."
   293→        )

   296→async def send_command(cmd_type: str, payload: dict | None = None) -> dict:
   297→    """
   298→    Send a SynapseCommand over WebSocket and return the response data.
   299→
   300→    Supports true parallel dispatch — multiple concurrent send_command
   301→    calls share a single recv loop that routes responses by ID.
   302→    """
   303→    command_id = _cmd_id(cmd_type, payload)
   304→    command = {
   305→        "type": cmd_type,
   306→        "id": command_id,
   307→        "payload": payload or {},
   308→        "sequence": 0,
   309→        "timestamp": time.time(),
   310→        "protocol_version": PROTOCOL_VERSION,
   311→    }
   312→
   313→    cmd_timeout = _SLOW_COMMANDS.get(cmd_type, COMMAND_TIMEOUT)
   314→    last_err = None
   315→
   316→    for _attempt in range(2):  # One transparent retry on connection failure
   317→        ws = await _get_connection()
   318→
   319→        # Register a future for this command's response
   320→        loop = asyncio.get_running_loop()
   321→        future: asyncio.Future = loop.create_future()
   322→        _pending[command_id] = future
   323→
   324→        try:
   325→            await ws.send(_dumps(command))
   326→
   327→            # Direct wait on future set — avoids asyncio.wait_for's internal task overhead
   328→            done, _ = await asyncio.wait({future}, timeout=cmd_timeout)
   329→            if not done:
   330→                # Timeout — future still pending
   331→                _pending.pop(command_id, None)
   332→                future.cancel()
   333→                raise asyncio.TimeoutError()
   334→            response = future.result()
   335→            break  # Success
   336→
   337→        except asyncio.TimeoutError:
   338→            _pending.pop(command_id, None)
   339→            # Close stale connection on timeout
   340→            global _ws_connection
   341→            try:
   342→                if ws:
   343→                    await ws.close()
   344→            except Exception:
   345→                pass
   346→            _ws_connection = None
   347→            raise TimeoutError(
   348→                f"The {cmd_type} command took too long to respond \u2014 "
   349→                "Houdini may be busy with a heavy operation"
   350→            )
   351→        except ConnectionError:
   352→            # Future was signaled by _recv_loop disconnect
   353→            _pending.pop(command_id, None)
   354→            try:
   355→                if ws:
   356→                    await ws.close()
   357→            except Exception:
   358→                pass
   359→            _ws_connection = None
   360→            last_err = ConnectionError(f"Connection lost during {cmd_type}")
   361→            logger.warning("Connection lost during %s, reconnecting...", cmd_type)
   362→        except Exception as e:
   363→            _pending.pop(command_id, None)
   364→            try:
   365→                if ws:
   366→                    await ws.close()
   367→            except Exception:
   368→                pass
   369→            _ws_connection = None
   370→            last_err = e
   371→            logger.warning("Connection lost during %s, reconnecting... (%s)", cmd_type, e)
   372→    else:
   373→        raise ConnectionError(
   374→            f"Lost connection while sending {cmd_type} and couldn't reconnect: {last_err}"
   375→        )
   376→
   377→    if not response.get("success", False):
   378→        error_msg = response.get("error", "Something went wrong on the Synapse side")
   379→        data = response.get("data") or {}
   380→        if isinstance(data, dict) and "retry_after" in data:
   381→            error_msg += f" (retry after {data['retry_after']}s)"
   382→        raise RuntimeError(error_msg)
   383→
   384→    return response.get("data", {})

  2024→def _execute_python_payload(args: dict) -> dict:

  2041→def _decide_payload(args: dict) -> dict:
  2042→    p = {"decision": args["decision"]}
  2043→    if "reasoning" in args:

  2046→        alt = args["alternatives"]
  2047→        p["alternatives"] = [a.strip() for a in alt.split(",") if a.strip()] if isinstance(alt, str) else alt

  2148→@server.call_tool()
  2149→async def call_tool(name: str, arguments: dict):
  2150→    """Dispatch MCP tool call to Synapse via WebSocket."""
  2151→    if name not in TOOL_DISPATCH:
  2152→        return [TextContent(type="text", text=f"I don't recognize the tool '{name}' \u2014 check the available tools list")]
  2153→
  2154→    cmd_type, build_payload = TOOL_DISPATCH[name]
  2155→
  2156→    try:
  2157→        payload = build_payload(arguments)
  2158→        data = await send_command(cmd_type, payload)
  2159→
  2160→        # Image-producing tools: read file and return as ImageContent
  2161→        if name in ("houdini_capture_viewport", "houdini_render"):
  2162→            import base64
  2163→            image_path = data.get("image_path", "")
  2164→            try:
  2165→                with open(image_path, "rb") as f:
  2166→                    b64 = base64.b64encode(f.read()).decode("utf-8")
  2167→                mime = "image/jpeg" if data.get("format") == "jpeg" else "image/png"
  2168→                meta = {
  2169→                    "width": data.get("width"),
  2170→                    "height": data.get("height"),
  2171→                    "format": data.get("format"),
  2172→                }
  2173→                # Include render-specific metadata
  2174→                if "engine" in data:
  2175→                    meta["rop"] = data.get("rop")
  2176→                    meta["rop_type"] = data.get("rop_type")
  2177→                    meta["engine"] = data.get("engine")
  2178→                return [
  2179→                    ImageContent(type="image", data=b64, mimeType=mime),
  2180→                    TextContent(type="text", text=_dumps_str(meta)),
  2181→                ]
  2182→            except FileNotFoundError:
  2183→                return [TextContent(type="text", text=(
  2184→                    f"The capture ran but the image file wasn't found at {image_path} \u2014 "
  2185→                    "it may have been cleaned up or the path changed"
  2186→                ))]
  2187→
  2188→        return [TextContent(type="text", text=_dumps_str(data))]
  2189→    except ConnectionError as e:
  2190→        return [TextContent(type="text", text=(
  2191→            f"Couldn't reach Synapse \u2014 {e}"
  2192→        ))]
  2193→    except RuntimeError as e:
  2194→        return [TextContent(type="text", text=f"Synapse hit a snag: {e}")]
  2195→    except Exception as e:
  2196→        logger.exception("Unexpected error in tool %s", name)
  2197→        return [TextContent(type="text", text=f"Something unexpected happened: {e}")]
