     1→"""
     2→Canonical Tool Registry -- Single Source of Truth
     3→
     4→Every MCP tool definition lives here. Both transports import from this module:
     5→- stdio bridge (mcp_server.py) generates Tool() objects from TOOL_DEFS
     6→- Streamable HTTP (mcp/tools.py) builds JSON tool definitions from TOOL_DEFS
     7→
     8→Adding a new tool:
     9→1. Add handler in the appropriate handlers_*.py file
    10→2. Add entry to TOOL_DEFS in this file
    11→3. Both transports pick it up automatically
    12→
    13→No hou dependency. No imports from synapse.server or synapse.mcp.tools.
    14→Only json and orjson (optional).
    15→"""
    16→
    17→import json
    18→from typing import Any
    19→
    20→try:
    21→    import orjson
    22→
    23→    def _dumps_str(obj) -> str:
    24→        return orjson.dumps(obj, option=orjson.OPT_SORT_KEYS).decode()
    25→except ImportError:
    26→
    27→    def _dumps_str(obj) -> str:
    28→        return json.dumps(obj, sort_keys=True)
    29→
    30→
    31→# =========================================================================
    32→# Dispatch ID counter (He2025: sequential, not uuid4)
    33→# =========================================================================
    34→
    35→_call_counter = 0
    36→
    37→
    38→def _next_call_id(tool_name: str) -> str:
    39→    global _call_counter
    40→    _call_counter += 1
    41→    return f"mcp-{tool_name}-{_call_counter}"
    42→
    43→
    44→# =========================================================================
    45→# Payload builders (transform MCP arguments to handler payload)
    46→# =========================================================================
    47→
    48→
    49→def _passthrough(_args: dict) -> dict:
    50→    return {}
    51→
    52→
    53→def _identity(args: dict) -> dict:
    54→    return dict(args)
    55→
    56→
    57→def _execute_python_payload(args: dict) -> dict:
    58→    p = {"content": args["code"]}
    59→    if "dry_run" in args:
    60→        p["dry_run"] = args["dry_run"]
    61→    if "atomic" in args:
    62→        p["atomic"] = args["atomic"]
    63→    return p
    64→
    65→
    66→def _stage_info_payload(args: dict) -> dict:
    67→    p = {}
    68→    if "node" in args:
    69→        p["node"] = args["node"]
    70→    return p
    71→
    72→
    73→def _decide_payload(args: dict) -> dict:
    74→    p = {"decision": args["decision"]}
    75→    if "reasoning" in args:
    76→        p["reasoning"] = args["reasoning"]
    77→    if "alternatives" in args:
    78→        alt = args["alternatives"]
    79→        p["alternatives"] = (
    80→            [a.strip() for a in alt.split(",") if a.strip()]
    81→            if isinstance(alt, str) else alt
    82→        )
    83→    return p
    84→
    85→
    86→def _add_memory_payload(args: dict) -> dict:
    87→    p = {"content": args["content"]}
    88→    if "memory_type" in args:
    89→        p["memory_type"] = args["memory_type"]
    90→    if "tags" in args:
    91→        p["tags"] = args["tags"]
    92→    return p
    93→
    94→
    95→def _filter_keys(keys):
    96→    """Return a payload builder that passes through only the specified keys."""
    97→    def _builder(args: dict) -> dict:
    98→        return {k: args[k] for k in keys if k in args}
    99→    return _builder
   100→
   101→
   102→def _network_explain_payload(args: dict) -> dict:
   103→    """Rename root_path -> node for the network_explain handler."""
   104→    return {**{k: v for k, v in args.items() if k != "root_path"}, "node": args["root_path"]}
   105→
   106→
   107→def _delete_node_payload(args: dict) -> dict:
   108→    return {"node": args["node"]}
