   181→# =============================================================================
   182→# COMMAND HANDLER REGISTRY
   183→# =============================================================================
   184→
   185→class CommandHandlerRegistry:
   186→    """
   187→    Registry for command handlers.
   188→
   189→    Allows registering handler functions for specific command types.
   190→    Supports both built-in and custom handlers.
   191→    """
   192→
   193→    def __init__(self):
   194→        self._handlers: Dict[str, Callable] = {}
   195→
   196→    def register(self, command_type: str, handler: Callable):
   197→        """Register a handler for a command type."""
   198→        self._handlers[command_type] = handler
   199→
   200→    def get(self, command_type: str) -> Optional[Callable]:
   201→        """Get handler for a command type (expects pre-normalized type)."""
   202→        return self._handlers.get(command_type)
   203→
   204→    def has(self, command_type: str) -> bool:
   205→        """Check if a handler exists for a command type."""
   206→        return self.get(command_type) is not None
   207→
   208→    @property
   209→    def registered_types(self):
   210→        """Get list of registered command types."""
   211→        return list(self._handlers.keys())

   217→class SynapseHandler(NodeHandlerMixin, UsdHandlerMixin, RenderHandlerMixin, TopsHandlerMixin, MaterialHandlerMixin, MemoryHandlerMixin, HdaHandlerMixin, CopsHandlerMixin):
   218→    """
   219→    Main command handler for the Synapse server.
   220→
   221→    Routes commands to appropriate handler functions and manages
   222→    integration with the session tracker / memory bridge.
   223→    """
   224→
   225→    def __init__(self):
   226→        self._registry = CommandHandlerRegistry()
   227→        self._session_id: Optional[str] = None
   228→        self._user_id: str = ""
   229→        self._bridge = None
   230→        self._register_handlers()
   231→
   232→    def set_session_id(self, session_id: str):
   233→        """Set the current session ID for action logging."""
   234→        self._session_id = session_id
   235→
   236→    def set_user_id(self, user_id: str):
   237→        """Set the current user ID for per-user audit trails."""
   238→        self._user_id = user_id
   239→
   240→    def _get_bridge(self):
   241→        """Lazy-load the bridge to avoid circular imports."""
   242→        if self._bridge is None:
   243→            from ..session.tracker import get_bridge
   244→            self._bridge = get_bridge()
   245→        return self._bridge
   246→
   247→    def handle(self, command: SynapseCommand) -> SynapseResponse:
   248→        """
   249→        Handle an incoming command.
   250→
   251→        Args:
   252→            command: The command to handle
   253→
   254→        Returns:
   255→            SynapseResponse with the result
   256→        """
   257→        try:
   258→            # Fast path: try raw type first (aliases are pre-populated in registry)
   259→            cmd_type = command.type
   260→            handler = self._registry.get(cmd_type)
   261→            if handler is None:
   262→                # Fallback: normalize for any aliases not in registry
   263→                cmd_type = normalize_command_type(command.type)
   264→                handler = self._registry.get(cmd_type)
   265→
   266→            if handler is None:
   267→                return SynapseResponse(
   268→                    id=command.id,
   269→                    success=False,
   270→                    error=f"I don't recognize the command '{command.type}' \u2014 try get_help to see what's available",
   271→                    sequence=command.sequence,
   272→                )
   273→
   274→            result = handler(command.payload)
   275→
   276→            # Log action asynchronously — don't block the response path
   277→            if cmd_type not in _READ_ONLY_COMMANDS:
   278→                self._submit_logs(cmd_type, command.payload, result)
   279→
   280→            return SynapseResponse(
   281→                id=command.id,
   282→                success=True,
   283→                data=result,
   284→                sequence=command.sequence,
   285→            )
   286→
   287→        except SynapseUserError as e:
   288→            # User errors: bad input, missing node, bad parm. Don't trip CB.
   289→            return SynapseResponse(
   290→                id=command.id,
   291→                success=False,
   292→                error=str(e),
   293→                sequence=command.sequence,
   294→            )
   295→        except ValueError as e:
   296→            # Legacy ValueError path -- still user error, don't trip CB.
   297→            return SynapseResponse(
   298→                id=command.id,
   299→                success=False,
   300→                error=str(e),
   301→                sequence=command.sequence,
   302→            )
   303→        except SynapseServiceError as e:
   304→            # Service errors: Houdini down, execution crash. DO trip CB.
   305→            return SynapseResponse(
   306→                id=command.id,
   307→                success=False,
   308→                error=str(e),
   309→                sequence=command.sequence,
   310→            )
   311→        except Exception as e:
   312→            return SynapseResponse(
   313→                id=command.id,
   314→                success=False,
   315→                error=f"Hit a snag processing that request: {e}",
   316→                sequence=command.sequence,
   317→            )
   318→
   319→    def _submit_logs(self, cmd_type: str, payload: Dict, result: Any):
   320→        """Submit bridge + audit log in a single executor call."""
   321→        bridge = self._get_bridge()
   322→        sid = self._session_id
   323→        uid = self._user_id
   324→        category = _CMD_CATEGORY.get(cmd_type, AuditCategory.SYNAPSE)
   325→        output = result if isinstance(result, dict) else {}
   326→
   327→        def _do_log():
   328→            if bridge and sid:
   329→                bridge.log_action(f"Executed: {cmd_type}", session_id=sid)
   330→            audit_log().log(
   331→                operation=cmd_type,
   332→                message=f"Executed {cmd_type}",
   333→                level=AuditLevel.AGENT_ACTION,
   334→                category=category,
   335→                user_id=uid,
   336→                input_data=payload,
   337→                output_data=output,
   338→            )
   339→
   340→        _log_executor.submit(_do_log)

   591→
   592→    def _handle_batch_commands(self, payload: Dict) -> Dict:
   593→        """Execute a batch of commands in declared order.
   594→
   595→        Payload:
   596→            commands: list of {type: str, payload: dict}
   597→            atomic: bool (default True) -- wrap in single undo group
   598→            stop_on_error: bool (default False) -- halt on first error
   599→        """
   600→        commands = payload.get("commands")
   601→        if not commands or not isinstance(commands, list):
   602→            raise ValueError("'commands' must be a non-empty list")
   603→
   604→        atomic = payload.get("atomic", True)
   605→        stop_on_error = payload.get("stop_on_error", False)
   606→
   607→        from .main_thread import run_on_main, _SLOW_TIMEOUT
   608→
   609→        def _on_main():
   610→            results: list = []
   611→            statuses: list = []
   612→            errors: list = []
   613→
   614→            def _run_batch():
   615→                for i, cmd_spec in enumerate(commands):
   616→                    cmd_type = cmd_spec.get("type", "")
   617→                    cmd_payload = cmd_spec.get("payload", {})
   618→                    handler = self._registry.get(cmd_type)
   619→
   620→                    if handler is None:
   621→                        err = f"Step {i}: unknown command '{cmd_type}'"
   622→                        errors.append(err)
   623→                        statuses.append("error")
   624→                        results.append(None)
   625→                        if stop_on_error:
   626→                            break
   627→                        continue
   628→
   629→                    try:
   630→                        result = handler(cmd_payload)
   631→                        results.append(result)
   632→                        statuses.append("ok")
   633→                        errors.append(None)
   634→                    except Exception as e:
   635→                        results.append(None)
   636→                        statuses.append("error")
   637→                        errors.append(f"Step {i}: {e}")
   638→                        if stop_on_error:
   639→                            break
   640→
   641→            if atomic and HOU_AVAILABLE:
   642→                with hou.undos.group("synapse_batch"):
   643→                    _run_batch()
   644→            else:
   645→                _run_batch()
   646→
   647→            return {
   648→                "results": results,
   649→                "statuses": statuses,
   650→                "errors": errors,
   651→            }
   652→
   653→        return run_on_main(_on_main, timeout=_SLOW_TIMEOUT)

   706→    def _handle_set_parm(self, payload: Dict) -> Dict:
   707→        """Handle set_parm command."""
   708→        if not HOU_AVAILABLE:
   709→            raise HoudiniUnavailableError()
   710→
   711→        node_path = resolve_param(payload, "node")
   712→        parm_name = resolve_param(payload, "parm")
   713→        value = resolve_param(payload, "value")
   714→
   715→        # Guard: reject NaN/Inf before they reach Houdini (they persist and corrupt)
   716→        if isinstance(value, float) and (math.isnan(value) or math.isinf(value)):
   717→            bad = "NaN" if math.isnan(value) else "Infinity"
   718→            raise ValueError(
   719→                f"Can't set {parm_name} to {bad} — that would corrupt the "
   720→                "parameter. Use a finite number instead."
   721→            )
   722→        if isinstance(value, (list, tuple)):
   723→            for i, v in enumerate(value):
   724→                if isinstance(v, float) and (math.isnan(v) or math.isinf(v)):
   725→                    bad = "NaN" if math.isnan(v) else "Infinity"
   726→                    raise ValueError(
   727→                        f"Can't set {parm_name} — element {i} is {bad}, which "
   728→                        "would corrupt the parameter. Use finite numbers instead."
   729→                    )
   730→
   731→        from .main_thread import run_on_main
   732→
   733→        def _on_main():
   734→            nonlocal parm_name
   735→            node = hou.node(node_path)
   736→            if node is None:
   737→                raise NodeNotFoundError(node_path)
   738→
   739→            parm = node.parm(parm_name)
   740→            # USD alias fallback -- resolve human-readable name to encoded parm
   741→            if parm is None:
   742→                usd_encoded = USD_PARM_ALIASES.get(parm_name.lower())
   743→                if usd_encoded:
   744→                    parm = node.parm(usd_encoded)
   745→                    if parm is not None:
   746→                        parm_name = usd_encoded
   747→            if parm is not None:
   748→                # If value is a list/tuple but we found a scalar parm, try the
   749→                # tuple parm instead — this lets callers set color (R,G,B) in one
   750→                # call using the base name (e.g., "base_color" with [1,0,0]).
   751→                if isinstance(value, (list, tuple)):
   752→                    parm_tuple = node.parmTuple(parm_name)
   753→                    if parm_tuple is not None and len(parm_tuple) > 1:
   754→                        parm_tuple.set(value)
   755→                        return {"node": node_path, "parm": parm_name, "value": value}
   756→                parm.set(value)
   757→                result = {"node": node_path, "parm": parm_name, "value": value}
   758→                # Lighting Law: soft warning when intensity > 1.0 on light nodes
   759→                if ("intensity" in parm_name.lower()
   760→                        and isinstance(value, (int, float)) and value > 1.0):
   761→                    try:
   762→                        if "light" in node.type().name().lower():
   763→                            result["warning"] = (
   764→                                "Intensity is usually 1.0 — consider using "
   765→                                "exposure for brightness control (Lighting Law)"
   766→                            )
   767→                    except Exception:
   768→                        pass  # Node type check is best-effort
   769→                return result
   770→
   771→            # Try as parm tuple
   772→            parm_tuple = node.parmTuple(parm_name)
   773→            if parm_tuple is not None:
   774→                if isinstance(value, (list, tuple)):
   775→                    parm_tuple.set(value)
   776→                else:
   777→                    parm_tuple.set([value] * len(parm_tuple))
   778→                result = {"node": node_path, "parm": parm_name, "value": value}
   779→                # Lighting Law: soft warning when intensity > 1.0 on light nodes
   780→                if ("intensity" in parm_name.lower()
   781→                        and isinstance(value, (int, float)) and value > 1.0):
   782→                    try:
   783→                        if "light" in node.type().name().lower():
   784→                            result["warning"] = (
   785→                                "Intensity is usually 1.0 — consider using "
   786→                                "exposure for brightness control (Lighting Law)"
   787→                            )
   788→                    except Exception:
   789→                        pass
   790→                return result
   791→
   792→            hint = _suggest_parms(node, parm_name)
   793→            raise ParameterError(node_path, parm_name, suggestion=hint.strip() if hint else "")
   794→
   795→        return run_on_main(_on_main)

   841→    def _handle_execute_python(self, payload: Dict) -> Dict:
   842→        """
   843→        Handle execute_python command.
   844→
   845→        Executes Python code in Houdini's runtime environment.
   846→        This is a standard DCC scripting pattern for automation.
   847→
   848→        Options:
   849→            dry_run (bool): Compile-only syntax check -- no execution.
   850→            atomic (bool): Wrap in undo group with rollback (default True).
   851→        """
   852→        if not HOU_AVAILABLE:
   853→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
   854→
   855→        code = resolve_param(payload, "content")
   856→        dry_run = resolve_param_with_default(payload, "dry_run", False)
   857→        atomic = resolve_param_with_default(payload, "atomic", True)
   858→
   859→        # Compile first -- catches SyntaxError for both dry_run and real runs
   860→        try:
   861→            compiled = compile(code, "<synapse_exec>", "exec")
   862→        except SyntaxError as e:
   863→            if dry_run:
   864→                return {"valid": False, "error": str(e), "dry_run": True}
   865→            raise
   866→
   867→        if dry_run:
   868→            return {"valid": True, "dry_run": True}
   869→
   870→        # Build execution namespace with Houdini access + idempotent guards
   871→        exec_globals = {"hou": hou, "__builtins__": __builtins__}
   872→        try:
   873→            from .guards import GUARD_FUNCTIONS
   874→            exec_globals.update(GUARD_FUNCTIONS)
   875→        except ImportError:
   876→            pass
   877→        exec_locals: dict = {}
   878→
   879→        from .main_thread import run_on_main, _SLOW_TIMEOUT
   880→
   881→        def _on_main():
   882→            # Execute inside undo group with smart rollback:
   883→            # - Coding bugs (NameError, SyntaxError, TypeError, AttributeError)
   884→            #   -> auto-rollback, since the script is broken and partial state is bad
   885→            # - Operational errors (render timeout, file not found, hou.OperationFailed)
   886→            #   -> keep mutations, since node creation/wiring may have succeeded
   887→            if atomic:
   888→                _needs_rollback = False
   889→                _rollback_exc = None
   890→                with hou.undos.group("synapse_execute"):
   891→                    try:
   892→                        _run_compiled(compiled, exec_globals, exec_locals)
   893→                    except _ROLLBACK_ERRORS as e:
   894→                        # Don't call performUndo() inside the undo group —
   895→                        # that raises "Cannot undo within an undo group".
   896→                        # Instead, flag for rollback after the group closes.
   897→                        _needs_rollback = True
   898→                        _rollback_exc = e
   899→                # Undo group is now closed — safe to roll back
   900→                if _needs_rollback:
   901→                    try:
   902→                        hou.undos.performUndo()
   903→                    except Exception:
   904→                        pass  # Best effort — group may already be empty
   905→                    raise _rollback_exc
   906→            else:
   907→                _run_compiled(compiled, exec_globals, exec_locals)
   908→
   909→            # Try to extract a result variable
   910→            result = exec_locals.get("result", "executed")
   911→
   912→            return {
   913→                "executed": True,
   914→                "result": str(result) if result else "executed",
   915→            }
   916→
   917→        return run_on_main(_on_main, timeout=_SLOW_TIMEOUT)

   919→    def _handle_execute_vex(self, payload: Dict) -> Dict:
   920→        """
   921→        Handle execute_vex command.
   922→
   923→        Creates a temporary attribwrangle node, sets the VEX snippet,
   924→        optionally wires an input geometry, and returns the node path.
   925→        Includes VEX error diagnosis when compilation or cooking fails.
   926→        """
   927→        if not HOU_AVAILABLE:
   928→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
   929→
   930→        snippet = resolve_param(payload, "snippet")
   931→        run_over = resolve_param_with_default(payload, "run_over", "Points")
   932→        input_node = resolve_param_with_default(payload, "input_node", None)
   933→
   934→        # Pre-execution lint check (no hou.* needed)
   935→        from ..routing.vex_diagnostics import (
   936→            lint_vex_snippet, diagnose_vex_error, format_diagnosis,
   937→        )
   938→        lint_issues = lint_vex_snippet(snippet)
   939→        warnings = []
   940→        if lint_issues:
   941→            warnings = [
   942→                {"line": d.line_number, "message": d.symptom, "fix": d.fix}
   943→                for d in lint_issues
   944→            ]
   945→
   946→        from .main_thread import run_on_main, _SLOW_TIMEOUT
   947→
   948→        def _on_main():
   949→            # Find or create a working SOP context
   950→            parent = None
   951→            if input_node:
   952→                src = hou.node(input_node)
   953→                if src is not None:
   954→                    parent = src.parent()
   955→
   956→            if parent is None:
   957→                # Default to /obj -- create a temp geo container
   958→                obj = hou.node("/obj")
   959→                parent = obj.createNode("geo", "synapse_vex_temp")
   960→
   961→            wrangle = parent.createNode("attribwrangle", "synapse_vex")
   962→            wrangle.parm("snippet").set(snippet)
   963→
   964→            # Map run_over string to class menu value
   965→            run_over_map = {
   966→                "detail": 0, "points": 1, "vertices": 2, "primitives": 3,
   967→            }
   968→            class_val = run_over_map.get(run_over.lower(), 1)
   969→            wrangle.parm("class").set(class_val)
   970→
   971→            # Wire input if provided
   972→            if input_node:
   973→                src = hou.node(input_node)
   974→                if src is not None:
   975→                    wrangle.setInput(0, src)
   976→
   977→            wrangle.setDisplayFlag(True)
   978→            wrangle.setRenderFlag(True)
   979→
   980→            # Try to cook and capture any VEX errors
   981→            cook_errors = []
   982→            diagnosis_text = ""
   983→            try:
   984→                wrangle.cook(force=True)
   985→            except Exception as cook_exc:
   986→                error_msg = str(cook_exc)
   987→                cook_errors.append(error_msg)
   988→                # Run pattern-based diagnosis
   989→                diagnoses = diagnose_vex_error(error_msg, snippet, wrangle.path())
   990→                if diagnoses:
   991→                    diagnosis_text = format_diagnosis(diagnoses, snippet)
   992→
   993→            # Also check node error state (Houdini may set errors without raising)
   994→            node_errors = []
   995→            try:
   996→                if wrangle.errors():
   997→                    node_errors = list(wrangle.errors())
   998→                    if not diagnosis_text and node_errors:
   999→                        all_errors = " | ".join(node_errors)
  1000→                        diagnoses = diagnose_vex_error(all_errors, snippet, wrangle.path())
  1001→                        if diagnoses:
  1002→                            diagnosis_text = format_diagnosis(diagnoses, snippet)
  1003→            except Exception:
  1004→                pass
  1005→
  1006→            # Clean up temp nodes on failure when no input was provided
  1007→            if (cook_errors or node_errors) and not input_node:
  1008→                try:
  1009→                    temp_parent = wrangle.parent()
  1010→                    if temp_parent and temp_parent.name().startswith("synapse_vex_temp"):
  1011→                        temp_parent.destroy()
  1012→                        wrangle = None  # Node no longer exists
  1013→                except Exception:
  1014→                    pass  # Best-effort cleanup
  1015→
  1016→            result = {
  1017→                "node": wrangle.path() if wrangle else "(cleaned up)",
  1018→                "snippet": snippet,
  1019→                "run_over": run_over,
  1020→                "class": class_val,
  1021→            }
  1022→
  1023→            if warnings:
  1024→                result["warnings"] = warnings
  1025→            if cook_errors or node_errors:
  1026→                result["errors"] = cook_errors + node_errors
  1027→            if diagnosis_text:
  1028→                result["diagnosis"] = diagnosis_text
  1029→
  1030→            return result
  1031→
  1032→        return run_on_main(_on_main, timeout=_SLOW_TIMEOUT)

  1236→    def _handle_autonomous_render(self, payload: Dict) -> Dict:
  1237→        """Run the autonomous render pipeline: Plan -> Validate -> Execute -> Evaluate -> Report.
  1238→
  1239→        Payload:
  1240→            intent (str, required): Natural-language render intent (e.g. 'render frames 1-48').
  1241→            max_iterations (int, optional): Max re-render attempts (default: 3).
  1242→            quality_threshold (float, optional): Minimum quality score 0.0-1.0 (default: 0.85).
  1243→
  1244→        Returns:
  1245→            Serialized RenderReport with plan, evaluation, decisions, and success flag.
  1246→        """
  1247→        import asyncio
  1248→        from dataclasses import asdict
  1249→
  1250→        intent = payload.get("intent")
  1251→        if not intent:
  1252→            raise ValueError(
  1253→                "We need an intent to plan the render \u2014 "
  1254→                "e.g. 'render frames 1-48' or 'render turntable at 50mm'"
  1255→            )
  1256→
  1257→        max_iterations = int(payload.get("max_iterations", 3))
  1258→        quality_threshold = float(payload.get("quality_threshold", 0.85))
  1259→
  1260→        # Late import to keep autonomy optional (no import at module level)
  1261→        from ..autonomy import (
  1262→            AutonomousDriver,
  1263→            RenderPlanner,
  1264→            PreFlightValidator,
  1265→            RenderEvaluator,
  1266→        )
  1267→
  1268→        # Adapter: bridge the sync handler registry to the async interface
  1269→        # expected by the autonomy modules.
  1270→        class _HandlerAdapter:
  1271→            """Adapts the sync handler registry to the async interface expected by autonomy modules."""
  1272→
  1273→            def __init__(self, registry):
  1274→                self._registry = registry
  1275→
  1276→            async def call(self, tool_name, params):
  1277→                handler = self._registry.get(tool_name)
  1278→                if handler is None:
  1279→                    raise ValueError(f"Couldn't find handler: {tool_name}")
  1280→                result = handler(params)
  1281→                if asyncio.iscoroutine(result):
  1282→                    return await result
  1283→                return result
  1284→
  1285→        adapter = _HandlerAdapter(self._registry)
  1286→
  1287→        # Optionally wire up memory for decision persistence
  1288→        memory = None
  1289→        try:
  1290→            from ..memory.store import get_synapse_memory
  1291→            memory = get_synapse_memory()
  1292→        except Exception:
  1293→            pass
  1294→
  1295→        planner = RenderPlanner()
  1296→        validator = PreFlightValidator(adapter)
  1297→        evaluator = RenderEvaluator()
  1298→
  1299→        driver = AutonomousDriver(
  1300→            planner=planner,
  1301→            validator=validator,
  1302→            evaluator=evaluator,
  1303→            handler_interface=adapter,
  1304→            memory_system=memory,
  1305→            max_iterations=max_iterations,
  1306→        )
  1307→
  1308→        # The autonomy pipeline is async; run it from this sync handler context.
  1309→        try:
  1310→            loop = asyncio.get_running_loop()
  1311→        except RuntimeError:
  1312→            loop = None
  1313→
  1314→        if loop and loop.is_running():
  1315→            # Already inside an event loop (e.g. hwebserver) — schedule as a task.
  1316→            import concurrent.futures
  1317→            future: concurrent.futures.Future = concurrent.futures.Future()
  1318→
  1319→            async def _run():
  1320→                try:
  1321→                    result = await driver.execute(intent)
  1322→                    future.set_result(result)
  1323→                except Exception as exc:
  1324→                    future.set_exception(exc)
  1325→
  1326→            loop.create_task(_run())
  1327→            report = future.result(timeout=600)
  1328→        else:
  1329→            report = asyncio.run(driver.execute(intent))
  1330→
  1331→        # Serialize the dataclass report to a plain dict
  1332→        def _serialize(obj):
  1333→            """Convert dataclass / enum / datetime to JSON-safe dict."""
  1334→            if hasattr(obj, "__dataclass_fields__"):
  1335→                return {k: _serialize(v) for k, v in asdict(obj).items()}
  1336→            if isinstance(obj, dict):
  1337→                return {k: _serialize(v) for k, v in obj.items()}
  1338→            if isinstance(obj, list):
  1339→                return [_serialize(v) for v in obj]
  1340→            if hasattr(obj, "value"):  # Enum
  1341→                return obj.value
  1342→            return obj
  1343→
  1344→        return _serialize(report)
