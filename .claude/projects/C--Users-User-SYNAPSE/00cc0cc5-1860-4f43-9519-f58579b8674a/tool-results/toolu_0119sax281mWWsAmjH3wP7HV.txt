    78→def _jsonrpc(method, params=None, msg_id=1):
    79→    """Build a JSON-RPC 2.0 request as bytes."""
    80→    msg = {"jsonrpc": "2.0", "method": method, "id": msg_id}
    81→    if params is not None:
    82→        msg["params"] = params
    83→    return json.dumps(msg, sort_keys=True).encode("utf-8")
    84→
    85→
    86→def _notification(method, params=None):
    87→    """Build a JSON-RPC 2.0 notification (no id)."""
    88→    msg = {"jsonrpc": "2.0", "method": method}
    89→    if params is not None:
    90→        msg["params"] = params
    91→    return json.dumps(msg, sort_keys=True).encode("utf-8")
    92→
    93→
    94→def _parse_response(body):
    95→    """Parse response bytes to dict."""
    96→    return json.loads(body)
