   106→class StatusDotAnimator(QtCore.QObject):
   107→    """Pulses the status dot opacity for 'executing' state."""
   108→
   109→    color_changed = QtCore.Signal(str)
   110→
   111→    def __init__(self, parent=None):
   112→        super().__init__(parent)
   113→        self._timer = QtCore.QTimer(self)
   114→        self._timer.setInterval(600)
   115→        self._timer.timeout.connect(self._tick)
   116→        self._on = True
   117→        self._color = SIGNAL if _HAS_DESIGN else "#00D4FF"
   118→        self._active = False
   119→
   120→    def start(self, color):
   121→        self._color = color
   122→        self._active = True
   123→        self._on = True
   124→        self._timer.start()
   125→
   126→    def stop(self, color):
   127→        self._timer.stop()
   128→        self._active = False
   129→        self._color = color
   130→        self.color_changed.emit(color)
   131→
   132→    def _tick(self):
   133→        self._on = not self._on
   134→        if self._on:
   135→            self.color_changed.emit(self._color)
   136→        else:
   137→            alpha_color = SLATE if _HAS_DESIGN else "#555555"
   138→            self.color_changed.emit(alpha_color)

   141→# ── Fallback ClaudeWorker (text-only, no tools) ──────────
   142→
   143→class _FallbackClaudeWorker(QtCore.QThread):
   144→    """Minimal text-only Claude streaming worker.
   145→
   146→    Used when synapse.panel.claude_worker cannot be imported.
   147→    Streams text tokens but does NOT support tool use.
   148→    """
   149→
   150→    token_received = QtCore.Signal(str)
   151→    stream_done = QtCore.Signal()
   152→    stream_error = QtCore.Signal(str)
   153→    # Stubs -- never emitted, but wired so the panel doesn't crash
   154→    tool_requested = QtCore.Signal(object)
   155→    tool_status = QtCore.Signal(str, str)
   156→
   157→    _API_HOST = "api.anthropic.com"
   158→    _API_PATH = "/v1/messages"
   159→    _API_VERSION = "2023-06-01"
   160→    _MODEL = "claude-sonnet-4-20250514"
   161→    _MAX_TOKENS = 4096
   162→
   163→    def __init__(self, messages, system_prompt="", parent=None):
   164→        super().__init__(parent)
   165→        self._messages = copy.deepcopy(messages)
   166→        self._system = system_prompt
   167→        self._abort = False
   168→
   169→    def abort(self):
   170→        self._abort = True
   171→
   172→    def get_messages(self):
   173→        return copy.deepcopy(self._messages)
   174→
   175→    def run(self):
   176→        try:
   177→            api_key = os.environ.get("ANTHROPIC_API_KEY", "")
   178→            if not api_key:
   179→                self.stream_error.emit("ANTHROPIC_API_KEY not set in environment")
   180→                return
   181→
   182→            body = {
   183→                "model": self._MODEL,
   184→                "max_tokens": self._MAX_TOKENS,
   185→                "stream": True,
   186→                "messages": self._messages,
   187→            }
   188→            if self._system:
   189→                body["system"] = self._system
   190→
   191→            payload = json.dumps(body).encode("utf-8")
   192→            ctx = ssl.create_default_context()
   193→            conn = http.client.HTTPSConnection(
   194→                self._API_HOST, timeout=60, context=ctx
   195→            )
   196→
   197→            try:
   198→                conn.request(
   199→                    "POST", self._API_PATH, body=payload,
   200→                    headers={
   201→                        "Content-Type": "application/json",
   202→                        "x-api-key": api_key,
   203→                        "anthropic-version": self._API_VERSION,
   204→                    },
   205→                )
   206→                response = conn.getresponse()
   207→                if response.status != 200:
   208→                    error_body = response.read().decode("utf-8", errors="replace")
   209→                    self.stream_error.emit(
   210→                        "API error {}: {}".format(response.status, error_body[:200])
   211→                    )
   212→                    return
   213→
   214→                self._parse_stream(response)
   215→                if not self._abort:
   216→                    self.stream_done.emit()
   217→            finally:
   218→                conn.close()
   219→
   220→        except Exception as exc:
   221→            self.stream_error.emit(str(exc))
   222→
   223→    def _parse_stream(self, response):
   224→        buf = ""
   225→        while True:
   226→            if self._abort:
   227→                return
   228→            chunk = response.read(4096)
   229→            if not chunk:
   230→                break
   231→            buf += chunk.decode("utf-8", errors="replace")
   232→            while "\n" in buf:
   233→                line, buf = buf.split("\n", 1)
   234→                line = line.strip()
   235→                if line.startswith("data:"):
   236→                    data_str = line[5:].strip()
   237→                    if data_str and data_str != "[DONE]":
   238→                        try:
   239→                            data = json.loads(data_str)
   240→                            delta = data.get("delta", {})
   241→                            text = delta.get("text", "")
   242→                            if text:
   243→                                self.token_received.emit(text)
   244→                        except (json.JSONDecodeError, KeyError):
   245→                            pass

   546→    def _set_status(self, state, detail=""):
   547→        """Update the status bar to reflect a new state."""
   548→        self._status = state
   549→
   550→        if _HAS_DESIGN and state in STATES:
   551→            state_info = STATES[state]
   552→            color = state_info["color"]
   553→            label = state_info["label"]
   554→        else:
   555→            color = "#00D4FF"
   556→            label = state.capitalize()
   557→
   558→        self._status_label.setText(label)
   559→        self._status_detail.setText(detail)
   560→
   561→        if state == "executing":
   562→            self._dot_animator.start(color)
   563→        else:
   564→            self._dot_animator.stop(color)
   565→            self._dot_label.setStyleSheet(f"color: {color};")
   566→            self._status_label.setStyleSheet(f"color: {color};")
   567→
   568→        # Update connect button text
   569→        if state in ("connected", "idle", "executing"):
   570→            self._connect_btn.setText("Disconnect")
   571→        else:
   572→            self._connect_btn.setText("Connect")

   587→    @staticmethod
   588→    def _copy_text(text):
   589→        """Copy text to clipboard via clip.exe (Windows) or Qt fallback."""
   590→        import platform, subprocess
   591→        if platform.system() == "Windows":
   592→            try:
   593→                p = subprocess.run(
   594→                    ["clip"], input=text.encode("utf-8"),
   595→                    creationflags=0x08000000, timeout=5,
   596→                )
   597→                return p.returncode == 0
   598→            except Exception:
   599→                pass
   600→        try:
   601→            app = QtWidgets.QApplication.instance() or QtGui.QGuiApplication.instance()
   602→            if app:
   603→                app.clipboard().setText(text)
   604→                return True
   605→        except Exception:
   606→            pass
   607→        return False

   794→    def _do_fix(self):
   795→        """Apply auto-fixes from the last /diagnose run."""
   796→        if self._last_diagnosis is None:
   797→            self._sys("No diagnosis to fix. Run /diagnose first.")
   798→            return
   799→        fixable = [i for i in self._last_diagnosis.issues if i.fix_available and i.fix_fn is not None]
   800→        if not fixable:
   801→            self._sys("No fixable issues in the last diagnosis.")
   802→            self._last_diagnosis = None
   803→            return
   804→        self._log_activity("Applying {} auto-fix(es)...".format(len(fixable)), "action")
   805→        try:
   806→            from synapse.panel.scene_doctor import apply_fixes
   807→            result = apply_fixes(self._last_diagnosis.issues)
   808→            msg = "Fixed {}, failed {}, skipped {}".format(
   809→                result["fixed"], result["failed"], result["skipped"]
   810→            )
   811→            self._sys(msg)
   812→            self._log_activity(msg, "success" if result["failed"] == 0 else "warning")
   813→            # Log to journal
   814→            try:
   815→                from synapse.panel.session_journal import get_journal
   816→                get_journal().log_command("/fix", msg)
   817→            except Exception:
   818→                pass
   819→            self._last_diagnosis = None
   820→            # Re-run diagnose to show remaining issues
   821→            self._do_diagnose("")
   822→        except ImportError:
   823→            self._sys("Scene doctor unavailable (synapse.panel.scene_doctor not found)")
   824→        except Exception as e:
   825→            self._sys("Fix error: " + str(e))

  2050→        """Handle /bookmark commands."""
  2051→        try:

  2053→                get_bookmark_manager, parse_bookmark_command,

  2064→
  2065→        if parsed["action"] == "list":
  2066→            bookmarks = manager.get_all()
  2067→            html = format_bookmarks_html(bookmarks)
  2068→            self._chat_display.append(html)
  2069→
  2070→        elif parsed["action"] == "add":

  2078→                elif isinstance(last.get("content"), list):
  2079→                    for block in last["content"]:
  2080→                        if isinstance(block, dict) and block.get("type") == "text":
  2081→                            preview = block["text"][:100]
  2082→                            break
  2083→
  2084→            bookmark = manager.add(parsed["label"], msg_idx, preview)
  2085→            self._sys("Bookmarked: {}".format(bookmark.label))
  2086→
  2087→            # Log to journal
  2088→            try:
  2089→                from synapse.panel.session_journal import get_journal
  2090→                get_journal().log_command("/bookmark", bookmark.label)
  2091→            except Exception:
  2092→                pass
  2093→
  2094→        elif parsed["action"] == "search":
  2095→            results = manager.search(parsed["query"])
  2096→            html = format_bookmarks_html(results, parsed["query"])
  2097→            self._chat_display.append(html)
  2098→
  2099→        elif parsed["action"] == "remove":
  2100→            if manager.remove(parsed["id"]):
  2101→                self._sys("Bookmark removed")
  2102→            else:
  2103→                self._sys("Bookmark not found: " + parsed["id"])
  2104→
  2105→        sb = self._chat_display.verticalScrollBar()
