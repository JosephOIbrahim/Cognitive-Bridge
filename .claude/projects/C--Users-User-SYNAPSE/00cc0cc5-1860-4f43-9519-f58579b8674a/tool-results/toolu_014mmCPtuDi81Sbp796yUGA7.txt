   106→class StatusDotAnimator(QtCore.QObject):
   107→    """Pulses the status dot opacity for 'executing' state."""
   108→
   109→    color_changed = QtCore.Signal(str)
   110→
   111→    def __init__(self, parent=None):
   112→        super().__init__(parent)
   113→        self._timer = QtCore.QTimer(self)
   114→        self._timer.setInterval(600)
   115→        self._timer.timeout.connect(self._tick)
   116→        self._on = True
   117→        self._color = SIGNAL if _HAS_DESIGN else "#00D4FF"
   118→        self._active = False

   120→    def start(self, color):
   121→        self._color = color
   122→        self._active = True
   123→        self._on = True
   124→        self._timer.start()
   125→
   126→    def stop(self, color):
   127→        self._timer.stop()
   128→        self._active = False
   129→        self._color = color
   130→        self.color_changed.emit(color)
   131→
   132→    def _tick(self):
   133→        self._on = not self._on
   134→        if self._on:
   135→            self.color_changed.emit(self._color)
   136→        else:
   137→            alpha_color = SLATE if _HAS_DESIGN else "#555555"
   138→            self.color_changed.emit(alpha_color)

   141→# ── Fallback ClaudeWorker (text-only, no tools) ──────────
   142→
   143→class _FallbackClaudeWorker(QtCore.QThread):
   144→    """Minimal text-only Claude streaming worker.
   145→
   146→    Used when synapse.panel.claude_worker cannot be imported.
   147→    Streams text tokens but does NOT support tool use.
   148→    """
   149→
   150→    token_received = QtCore.Signal(str)
   151→    stream_done = QtCore.Signal()
   152→    stream_error = QtCore.Signal(str)
   153→    # Stubs -- never emitted, but wired so the panel doesn't crash
   154→    tool_requested = QtCore.Signal(object)
   155→    tool_status = QtCore.Signal(str, str)
   156→
   157→    _API_HOST = "api.anthropic.com"
   158→    _API_PATH = "/v1/messages"
   159→    _API_VERSION = "2023-06-01"
   160→    _MODEL = "claude-sonnet-4-20250514"
   161→    _MAX_TOKENS = 4096
   162→
   163→    def __init__(self, messages, system_prompt="", parent=None):
   164→        super().__init__(parent)
   165→        self._messages = copy.deepcopy(messages)
   166→        self._system = system_prompt
   167→        self._abort = False
   168→
   169→    def abort(self):
   170→        self._abort = True
   171→
   172→    def get_messages(self):
   173→        return copy.deepcopy(self._messages)
   174→
   175→    def run(self):
   176→        try:
   177→            api_key = os.environ.get("ANTHROPIC_API_KEY", "")
   178→            if not api_key:
   179→                self.stream_error.emit("ANTHROPIC_API_KEY not set in environment")
   180→                return
   181→
   182→            body = {
   183→                "model": self._MODEL,
   184→                "max_tokens": self._MAX_TOKENS,
   185→                "stream": True,
   186→                "messages": self._messages,
   187→            }
   188→            if self._system:
   189→                body["system"] = self._system
   190→
   191→            payload = json.dumps(body).encode("utf-8")
   192→            ctx = ssl.create_default_context()
   193→            conn = http.client.HTTPSConnection(
   194→                self._API_HOST, timeout=60, context=ctx
   195→            )
   196→
   197→            try:
   198→                conn.request(
   199→                    "POST", self._API_PATH, body=payload,
   200→                    headers={
   201→                        "Content-Type": "application/json",
   202→                        "x-api-key": api_key,
   203→                        "anthropic-version": self._API_VERSION,
   204→                    },
   205→                )
   206→                response = conn.getresponse()
   207→                if response.status != 200:
   208→                    error_body = response.read().decode("utf-8", errors="replace")
   209→                    self.stream_error.emit(
   210→                        "API error {}: {}".format(response.status, error_body[:200])
   211→                    )
   212→                    return
   213→
   214→                self._parse_stream(response)
   215→                if not self._abort:
   216→                    self.stream_done.emit()
   217→            finally:
   218→                conn.close()
   219→
   220→        except Exception as exc:
   221→            self.stream_error.emit(str(exc))
   222→
   223→    def _parse_stream(self, response):
   224→        buf = ""
   225→        while True:
   226→            if self._abort:
   227→                return
   228→            chunk = response.read(4096)
   229→            if not chunk:
   230→                break
   231→            buf += chunk.decode("utf-8", errors="replace")
   232→            while "\n" in buf:
   233→                line, buf = buf.split("\n", 1)
   234→                line = line.strip()
   235→                if line.startswith("data:"):
   236→                    data_str = line[5:].strip()
   237→                    if data_str and data_str != "[DONE]":
   238→                        try:
   239→                            data = json.loads(data_str)
   240→                            delta = data.get("delta", {})
   241→                            text = delta.get("text", "")
   242→                            if text:
   243→                                self.token_received.emit(text)
   244→                        except (json.JSONDecodeError, KeyError):
   245→                            pass

   530→    # ── WS Path Copy ─────────────────────────────────────

   532→    def _on_copy_ws_path(self):
   533→        """Copy the WebSocket URL to clipboard."""
   534→        if self._copy_text(_WS_URL):
   535→            self._log_activity("Copied: {}".format(_WS_URL), "success")
   536→        else:
   537→            self._log_activity("Couldn't copy WS path to clipboard", "error")

   539→    @staticmethod
   540→    def _copy_text(text):
   541→        """Copy text to clipboard via clip.exe (Windows) or Qt fallback."""
   542→        import platform, subprocess
   543→        if platform.system() == "Windows":
   544→            try:
   545→                p = subprocess.run(
   546→                    ["clip"], input=text.encode("utf-8"),
   547→                    creationflags=0x08000000, timeout=5,
   548→                )
   549→                return p.returncode == 0
   550→            except Exception:
   551→                pass
   552→        try:
   553→            app = QtWidgets.QApplication.instance() or QtGui.QGuiApplication.instance()
   554→            if app:
   555→                app.clipboard().setText(text)
   556→                return True
   557→        except Exception:
   558→            pass
   559→        return False

   561→    # ── Activity Log ─────────────────────────────────────
   562→
   563→    def _log_activity(self, message, level="info"):
   564→        """Add a timestamped entry to the activity log."""
   565→        timestamp = time.strftime("%H:%M:%S")
   566→
   567→        color_map = {
   568→            "info": SILVER if _HAS_DESIGN else "#999999",
   569→            "success": GROW if _HAS_DESIGN else "#00E676",
   570→            "warning": WARN if _HAS_DESIGN else "#FFAB00",
   571→            "error": ERROR if _HAS_DESIGN else "#FF3D71",
   572→            "action": SIGNAL if _HAS_DESIGN else "#00D4FF",
   573→        }
   574→        color = color_map.get(level, color_map["info"])
   575→

   733→    def _do_fix(self):
   734→        """Apply auto-fixes from the last /diagnose run."""
   735→        if self._last_diagnosis is None:
   736→            self._sys("No diagnosis to fix. Run /diagnose first.")
   737→            return
   738→        fixable = [i for i in self._last_diagnosis.issues if i.fix_available and i.fix_fn is not None]
   739→        if not fixable:
   740→            self._sys("No fixable issues in the last diagnosis.")
   741→            self._last_diagnosis = None
   742→            return
   743→        self._log_activity("Applying {} auto-fix(es)...".format(len(fixable)), "action")
   744→        try:
   745→            from synapse.panel.scene_doctor import apply_fixes
   746→            result = apply_fixes(self._last_diagnosis.issues)
   747→            msg = "Fixed {}, failed {}, skipped {}".format(
   748→                result["fixed"], result["failed"], result["skipped"]
   749→            )
   750→            self._sys(msg)
   751→            self._log_activity(msg, "success" if result["failed"] == 0 else "warning")
   752→            # Log to journal
   753→            try:
   754→                from synapse.panel.session_journal import get_journal
   755→                get_journal().log_command("/fix", msg)
   756→            except Exception:
   757→                pass
   758→            self._last_diagnosis = None
   759→            # Re-run diagnose to show remaining issues
   760→            self._do_diagnose("")
   761→        except ImportError:
   762→            self._sys("Scene doctor unavailable (synapse.panel.scene_doctor not found)")
   763→        except Exception as e:
   764→            self._sys("Fix error: " + str(e))

   939→    def _route_cmd(self, text):
   940→        """Route slash commands to their handlers."""

   943→        arg = parts[1].strip() if len(parts) > 1 else ""
   944→
   945→        if cmd == "/hda":
   946→            if not arg:
   947→                self._sys("Usage: /hda &lt;describe what you want to build&gt;")
   948→                return
   949→            # Display user message
   950→            user_color = SIGNAL if _HAS_DESIGN else "#00D4FF"
   951→            user_html = (
   952→                '<div style="margin: 8px 0 4px 0;">'
   953→                '<b style="color: {};">You</b></div>'
   954→                '<div style="margin: 0 0 8px 10px;">{}</div>'
   955→            ).format(user_color, arg.replace("\n", "<br>"))
   956→            self._chat_display.append(user_html)
   957→            self._do_hda(arg)
   958→        elif cmd == "/diagnose":
   959→            self._do_diagnose(arg)
   960→        elif cmd == "/fix":
   961→            self._do_fix()
   962→        elif cmd == "/preflight":
   963→            self._do_preflight(arg)
   964→        elif cmd == "/journal":
   965→            self._do_journal(arg)
   966→        elif cmd == "/login":
   967→            self._do_shot_login()
   968→        elif cmd == "/help":
   969→            self._show_help()
