   106→class StatusDotAnimator(QtCore.QObject):
   107→    """Pulses the status dot opacity for 'executing' state."""
   108→
   109→    color_changed = QtCore.Signal(str)
   110→
   111→    def __init__(self, parent=None):
   112→        super().__init__(parent)
   113→        self._timer = QtCore.QTimer(self)
   114→        self._timer.setInterval(600)
   115→        self._timer.timeout.connect(self._tick)
   116→        self._on = True
   117→        self._color = SIGNAL if _HAS_DESIGN else "#00D4FF"
   118→        self._active = False
   119→
   120→    def start(self, color):
   121→        self._color = color
   122→        self._active = True
   123→        self._on = True
   124→        self._timer.start()
   125→
   126→    def stop(self, color):
   127→        self._timer.stop()
   128→        self._active = False
   129→        self._color = color
   130→        self.color_changed.emit(color)
   131→
   132→    def _tick(self):
   133→        self._on = not self._on
   134→        if self._on:
   135→            self.color_changed.emit(self._color)
   136→        else:
   137→            alpha_color = SLATE if _HAS_DESIGN else "#555555"
   138→            self.color_changed.emit(alpha_color)

   141→# ── Fallback ClaudeWorker (text-only, no tools) ──────────
   142→
   143→class _FallbackClaudeWorker(QtCore.QThread):
   144→    """Minimal text-only Claude streaming worker.
   145→
   146→    Used when synapse.panel.claude_worker cannot be imported.
   147→    Streams text tokens but does NOT support tool use.
   148→    """
   149→
   150→    token_received = QtCore.Signal(str)
   151→    stream_done = QtCore.Signal()
   152→    stream_error = QtCore.Signal(str)
   153→    # Stubs -- never emitted, but wired so the panel doesn't crash
   154→    tool_requested = QtCore.Signal(object)
   155→    tool_status = QtCore.Signal(str, str)
   156→
   157→    _API_HOST = "api.anthropic.com"
   158→    _API_PATH = "/v1/messages"
   159→    _API_VERSION = "2023-06-01"
   160→    _MODEL = "claude-sonnet-4-20250514"
   161→    _MAX_TOKENS = 4096
   162→
   163→    def __init__(self, messages, system_prompt="", parent=None):
   164→        super().__init__(parent)
   165→        self._messages = copy.deepcopy(messages)
   166→        self._system = system_prompt
   167→        self._abort = False
   168→
   169→    def abort(self):
   170→        self._abort = True
   171→
   172→    def get_messages(self):
   173→        return copy.deepcopy(self._messages)
   174→
   175→    def run(self):
   176→        try:
   177→            api_key = os.environ.get("ANTHROPIC_API_KEY", "")
   178→            if not api_key:
   179→                self.stream_error.emit("ANTHROPIC_API_KEY not set in environment")
   180→                return
   181→
   182→            body = {
   183→                "model": self._MODEL,
   184→                "max_tokens": self._MAX_TOKENS,
   185→                "stream": True,
   186→                "messages": self._messages,
   187→            }
   188→            if self._system:
   189→                body["system"] = self._system
   190→
   191→            payload = json.dumps(body).encode("utf-8")
   192→            ctx = ssl.create_default_context()
   193→            conn = http.client.HTTPSConnection(
   194→                self._API_HOST, timeout=60, context=ctx
   195→            )
   196→
   197→            try:
   198→                conn.request(
   199→                    "POST", self._API_PATH, body=payload,
   200→                    headers={
   201→                        "Content-Type": "application/json",
   202→                        "x-api-key": api_key,
   203→                        "anthropic-version": self._API_VERSION,
   204→                    },
   205→                )
   206→                response = conn.getresponse()
   207→                if response.status != 200:
   208→                    error_body = response.read().decode("utf-8", errors="replace")
   209→                    self.stream_error.emit(
   210→                        "API error {}: {}".format(response.status, error_body[:200])
   211→                    )
   212→                    return
   213→
   214→                self._parse_stream(response)
   215→                if not self._abort:
   216→                    self.stream_done.emit()
   217→            finally:
   218→                conn.close()
   219→
   220→        except Exception as exc:
   221→            self.stream_error.emit(str(exc))
   222→
   223→    def _parse_stream(self, response):
   224→        buf = ""
   225→        while True:
   226→            if self._abort:
   227→                return
   228→            chunk = response.read(4096)
   229→            if not chunk:
   230→                break
   231→            buf += chunk.decode("utf-8", errors="replace")
   232→            while "\n" in buf:
   233→                line, buf = buf.split("\n", 1)
   234→                line = line.strip()
   235→                if line.startswith("data:"):
   236→                    data_str = line[5:].strip()
   237→                    if data_str and data_str != "[DONE]":
   238→                        try:
   239→                            data = json.loads(data_str)
   240→                            delta = data.get("delta", {})
   241→                            text = delta.get("text", "")
   242→                            if text:
   243→                                self.token_received.emit(text)
   244→                        except (json.JSONDecodeError, KeyError):
   245→                            pass

   502→    def _set_status(self, state, detail=""):
   503→        """Update the status bar to reflect a new state."""
   504→        self._status = state
   505→
   506→        if _HAS_DESIGN and state in STATES:
   507→            state_info = STATES[state]
   508→            color = state_info["color"]

   514→        self._status_label.setText(label)
   515→        self._status_detail.setText(detail)
   516→
   517→        if state == "executing":
   518→            self._dot_animator.start(color)
   519→        else:
   520→            self._dot_animator.stop(color)
   521→            self._dot_label.setStyleSheet(f"color: {color};")
   522→            self._status_label.setStyleSheet(f"color: {color};")
   523→
   524→        # Update connect button text
   525→        if state in ("connected", "idle", "executing"):
   526→            self._connect_btn.setText("Disconnect")
   527→        else:
   528→            self._connect_btn.setText("Connect")

   534→    # ── WS Path Copy ─────────────────────────────────────

   536→    def _on_copy_ws_path(self):
   537→        """Copy the WebSocket URL to clipboard."""
   538→        if self._copy_text(_WS_URL):
   539→            self._log_activity("Copied: {}".format(_WS_URL), "success")
   540→        else:
   541→            self._log_activity("Couldn't copy WS path to clipboard", "error")
   542→
   543→    @staticmethod
   544→    def _copy_text(text):
   545→        """Copy text to clipboard via clip.exe (Windows) or Qt fallback."""
   546→        import platform, subprocess
   547→        if platform.system() == "Windows":
   548→            try:
   549→                p = subprocess.run(
   550→                    ["clip"], input=text.encode("utf-8"),
   551→                    creationflags=0x08000000, timeout=5,
   552→                )
   553→                return p.returncode == 0
   554→            except Exception:
   555→                pass
   556→        try:
   557→            app = QtWidgets.QApplication.instance() or QtGui.QGuiApplication.instance()
   558→            if app:
   559→                app.clipboard().setText(text)
   560→                return True
   561→        except Exception:
   562→            pass
   563→        return False

   565→    # ── Activity Log ─────────────────────────────────────

   567→    def _log_activity(self, message, level="info"):
   568→        """Add a timestamped entry to the activity log."""
   569→        timestamp = time.strftime("%H:%M:%S")
   570→
   571→        color_map = {
   572→            "info": SILVER if _HAS_DESIGN else "#999999",
   573→            "success": GROW if _HAS_DESIGN else "#00E676",

   737→    def _do_fix(self):
   738→        """Apply auto-fixes from the last /diagnose run."""
   739→        if self._last_diagnosis is None:
   740→            self._sys("No diagnosis to fix. Run /diagnose first.")
   741→            return
   742→        fixable = [i for i in self._last_diagnosis.issues if i.fix_available and i.fix_fn is not None]
   743→        if not fixable:
   744→            self._sys("No fixable issues in the last diagnosis.")
   745→            self._last_diagnosis = None
   746→            return
   747→        self._log_activity("Applying {} auto-fix(es)...".format(len(fixable)), "action")
   748→        try:
   749→            from synapse.panel.scene_doctor import apply_fixes
   750→            result = apply_fixes(self._last_diagnosis.issues)
   751→            msg = "Fixed {}, failed {}, skipped {}".format(
   752→                result["fixed"], result["failed"], result["skipped"]
   753→            )
   754→            self._sys(msg)
   755→            self._log_activity(msg, "success" if result["failed"] == 0 else "warning")
   756→            # Log to journal
   757→            try:
   758→                from synapse.panel.session_journal import get_journal
   759→                get_journal().log_command("/fix", msg)
   760→            except Exception:
   761→                pass
   762→            self._last_diagnosis = None
   763→            # Re-run diagnose to show remaining issues
   764→            self._do_diagnose("")
   765→        except ImportError:
   766→            self._sys("Scene doctor unavailable (synapse.panel.scene_doctor not found)")
   767→        except Exception as e:
   768→            self._sys("Fix error: " + str(e))

   898→    def _route_cmd(self, text):
   899→        """Route slash commands to their handlers."""
   900→        parts = text.split(None, 1)
   901→        cmd = parts[0].lower()
   902→        arg = parts[1].strip() if len(parts) > 1 else ""
   903→
   904→        if cmd == "/hda":
   905→            if not arg:
   906→                self._sys("Usage: /hda &lt;describe what you want to build&gt;")
   907→                return
   908→            # Display user message
   909→            user_color = SIGNAL if _HAS_DESIGN else "#00D4FF"
   910→            user_html = (
   911→                '<div style="margin: 8px 0 4px 0;">'
   912→                '<b style="color: {};">You</b></div>'
   913→                '<div style="margin: 0 0 8px 10px;">{}</div>'
   914→            ).format(user_color, arg.replace("\n", "<br>"))
   915→            self._chat_display.append(user_html)
   916→            self._do_hda(arg)
   917→        elif cmd == "/diagnose":
   918→            self._do_diagnose(arg)
   919→        elif cmd == "/fix":
   920→            self._do_fix()
   921→        elif cmd == "/preflight":
   922→            self._do_preflight(arg)
   923→        elif cmd == "/journal":
   924→            self._do_journal(arg)
   925→        elif cmd == "/login":
   926→            self._do_shot_login()
   927→        elif cmd == "/explain":
   928→            self._do_explain(arg)
   929→        elif cmd == "/trace":
   930→            self._do_trace(arg)

   932→            self._do_vex(arg)
   933→        elif cmd == "/recipes" or cmd == "/recipe":
   934→            self._do_recipes(arg)
   935→        elif cmd == "/help":
   936→            self._show_help()
   937→        else:
   938→            self._sys("Unknown command: {}. Type /help for available commands.".format(cmd))

  1032→    def _do_explain(self, arg=""):
  1033→        """Explain a selected node or network data flow via Claude."""
  1034→        try:
  1035→            from synapse.panel.explain_mode import (
  1036→                gather_node_context, gather_network_context,
  1037→                build_explain_prompt, build_explain_messages,
  1038→            )
  1039→        except ImportError as e:
  1040→            self._sys("Explain mode unavailable: " + str(e))
  1041→            return
  1042→
  1043→        arg = arg.strip()
  1044→
  1045→        # Determine mode: network path (starts with /) or node
  1046→        is_network = False
  1047→        if arg.startswith("/"):
  1048→            # Check if this resolves to a network (container with children)
  1049→            try:
  1050→                node = hou.node(arg)
  1051→                if node is not None and len(node.children()) > 0:
  1052→                    is_network = True
  1053→            except Exception:
  1054→                pass
  1055→
  1056→        if is_network:
  1057→            context = gather_network_context(arg)
  1058→            mode = "network"
  1059→        else:
  1060→            # Node mode: use arg as path, or fall back to selection
  1061→            node_path = arg
  1062→            if not node_path:
  1063→                try:
  1064→                    sel = hou.selectedNodes()
  1065→                    if sel:
  1066→                        node_path = sel[0].path()
  1067→                except Exception:
  1068→                    pass
  1069→            if not node_path:
  1070→                self._sys("Select a node first, or type /explain &lt;node_path&gt;")
  1071→                return
  1072→            context = gather_node_context(node_path)
  1073→            mode = "node"
  1074→
  1075→        if context.get("error"):
  1076→            self._sys(context["error"])
  1077→            return
  1078→
  1079→        # Display user intent
  1080→        user_color = SIGNAL if _HAS_DESIGN else "#00D4FF"
  1081→        target = context.get("network_path") or context.get("node_path", "")
  1082→        self._chat_display.append(
  1083→            '<div style="margin: 8px 0 4px 0;">'
  1084→            '<b style="color: {};">You</b></div>'
  1085→            '<div style="margin: 0 0 8px 10px;">/explain {}</div>'.format(
  1086→                user_color, target,
  1087→            )
  1088→        )
  1089→
  1090→        # Build prompts and launch worker
  1091→        explain_prompt = build_explain_prompt(context, mode)
  1092→        messages = build_explain_messages(context, mode)
  1093→
  1094→        # Separate system message from user messages
  1095→        sys_content = explain_prompt
  1096→        user_messages = [m for m in messages if m["role"] != "system"]
  1097→
  1098→        try:
  1099→            from synapse.panel.system_prompt import build_system_prompt
  1100→            base = build_system_prompt(self._gather_context())
  1101→        except ImportError:
  1102→            base = "You are SYNAPSE, an AI co-pilot inside Houdini 21."
  1103→        system_prompt = self._build_memory_system_prompt(base + "\n\n" + sys_content)
  1104→
  1105→        self._launch_worker(user_messages, system_prompt)
  1106→
  1107→        # Log to journal
  1108→        try:
  1109→            from synapse.panel.session_journal import get_journal
  1110→            get_journal().log_command("/explain", "{} ({})".format(target, mode))
