   647→# =============================================================================
   648→# Command Parsing
   649→# =============================================================================
   650→
   651→def parse_recipe_command(text: str) -> dict:
   652→    """Parse a /recipes command into an action dict.
   653→
   654→    Returns:
   655→        dict with keys: action, category, recipe, query (depending on action)
   656→
   657→    Examples:
   658→        "/recipes"                              -> {"action": "list", ...}
   659→        "/recipes scatter"                      -> {"action": "category", "category": "scatter", ...}
   660→        "/recipes scatter uniform_scatter"      -> {"action": "detail", ...}
   661→        "/recipes build scatter uniform_scatter" -> {"action": "build", ...}
   662→        "/recipes search noise"                 -> {"action": "search", "query": "noise"}
   663→    """
   664→    # Strip the command prefix
   665→    stripped = text.strip()
   666→    if stripped.lower().startswith("/recipes"):
   667→        stripped = stripped[len("/recipes"):].strip()
   668→    elif stripped.lower().startswith("/recipe"):
   669→        stripped = stripped[len("/recipe"):].strip()
   670→
   671→    if not stripped:
   672→        return {"action": "list", "category": "", "recipe": ""}
   673→
   674→    parts = stripped.split()
   675→
   676→    # /recipes search <query>
   677→    if parts[0].lower() == "search" and len(parts) >= 2:
   678→        return {"action": "search", "query": " ".join(parts[1:]), "category": "", "recipe": ""}
   679→
   680→    # /recipes build <category> <recipe>
   681→    if parts[0].lower() == "build" and len(parts) >= 3:
   682→        return {"action": "build", "category": parts[1].lower(), "recipe": parts[2].lower()}
   683→
   684→    # /recipes <category> <recipe>
   685→    if len(parts) >= 2:
   686→        return {"action": "detail", "category": parts[0].lower(), "recipe": parts[1].lower()}
   687→
   688→    # /recipes <category>
   689→    return {"action": "category", "category": parts[0].lower(), "recipe": ""}

   713→def search_recipes(query: str) -> list:
   714→    """Search all recipes by title, description, explanation, and tips.
   715→
   716→    Args:
   717→        query: Search string (case-insensitive).
   718→
   719→    Returns:
   720→        List of (category, recipe_name, title, match_context) tuples.
   721→    """
   722→    results = []
   723→    q = query.lower()
   724→
   725→    for cat_name in sorted(RECIPES.keys()):
   726→        cat = RECIPES[cat_name]
   727→        for recipe_name in sorted(cat.keys()):
   728→            recipe = cat[recipe_name]
   729→            matches = []
   730→
   731→            if q in recipe["title"].lower():
   732→                matches.append(("title", recipe["title"]))
   733→            if q in recipe["description"].lower():
   734→                matches.append(("description", recipe["description"]))
   735→            if q in recipe["explanation"].lower():
   736→                matches.append(("explanation", recipe["explanation"]))
   737→            for tip in recipe.get("tips", []):
   738→                if q in tip.lower():
   739→                    matches.append(("tip", tip))
   740→                    break  # one tip match is enough
   741→
   742→            if matches:
   743→                # Use the first match as context
   744→                match_type, match_text = matches[0]
   745→                results.append((cat_name, recipe_name, recipe["title"], match_text))
   746→
   747→    return results

   933→def build_recipe_messages(category: str, recipe_name: str, context: dict) -> list:
   934→    """Build Claude API messages that instruct Claude to construct the recipe.
   935→
   936→    Args:
   937→        category: Category name.
   938→        recipe_name: Recipe name.
   939→        context: Current network context dict (e.g., {"parent_path": "/obj/geo1"}).
   940→
   941→    Returns:
   942→        List of message dicts for the Claude API, or empty list if recipe not found.
   943→    """
   944→    recipe = get_recipe(category, recipe_name)
   945→    if recipe is None:
   946→        return []
   947→
   948→    parent_path = context.get("parent_path", "/obj/geo1")
   949→
   950→    # Build node descriptions
   951→    node_lines = []
   952→    for i, node in enumerate(recipe["nodes"]):
   953→        parm_desc = ""
   954→        if node.get("parms"):
   955→            parm_items = []
   956→            for k, v in sorted(node["parms"].items()):
   957→                parm_items.append(f"  {k} = {repr(v)}")
   958→            parm_desc = "\n" + "\n".join(parm_items)
   959→        node_lines.append(
   960→            f"  {i + 1}. Create node type='{node['type']}' name='{node['name']}'"
   961→            f"{parm_desc}"
   962→        )
   963→
   964→    # Build connection descriptions
   965→    conn_lines = []
   966→    for conn in recipe["connections"]:
   967→        src, dst, idx = conn[0], conn[1], conn[2]
   968→        conn_lines.append(f"  - Connect {src} output 0 -> {dst} input {idx}")
   969→
   970→    instruction = (
   971→        f"Build the '{recipe['title']}' recipe in {parent_path}.\n\n"
   972→        f"Context: {recipe['context']} network\n\n"
   973→        f"Create these nodes:\n"
   974→        + "\n".join(node_lines)
   975→        + "\n\n"
   976→        f"Make these connections:\n"
   977→        + "\n".join(conn_lines)
   978→        + "\n\n"
   979→        f"Use houdini_create_node, houdini_set_parm, and houdini_connect_nodes tools. "
   980→        f"Set the display flag on the last node.\n\n"
   981→        f"Key parameters to set: {', '.join(recipe['key_parms'])}"
   982→    )
   983→
   984→    return [
   985→        {
   986→            "role": "user",
   987→            "content": instruction,
   988→        }
   989→    ]
