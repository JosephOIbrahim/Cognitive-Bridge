   105→
   106→class _InputKeyFilter(QtCore.QObject):
   107→    """Enter sends chat, Shift+Enter inserts newline."""
   108→
   109→    def __init__(self, text_edit, panel, parent=None):
   110→        super().__init__(parent or text_edit)
   111→        self._text_edit = text_edit
   112→        self._panel = panel
   113→
   114→    def eventFilter(self, obj, event):
   115→        if obj is self._text_edit and event.type() == QtCore.QEvent.KeyPress:
   116→            if event.key() in (QtCore.Qt.Key_Return, QtCore.Qt.Key_Enter):
   117→                if event.modifiers() & QtCore.Qt.ShiftModifier:
   118→                    return False  # Let Shift+Enter insert newline
   119→                self._panel._on_send_chat()
   120→                return True
   121→        return super().eventFilter(obj, event)

   126→class StatusDotAnimator(QtCore.QObject):
   127→    """Pulses the status dot opacity for 'executing' state."""
   128→
   129→    color_changed = QtCore.Signal(str)

   131→    def __init__(self, parent=None):
   132→        super().__init__(parent)

   135→        self._timer.timeout.connect(self._tick)
   136→        self._on = True
   137→        self._color = SIGNAL if _HAS_DESIGN else "#00D4FF"
   138→        self._active = False

   140→    def start(self, color):
   141→        self._color = color
   142→        self._active = True
   143→        self._on = True
   144→        self._timer.start()
   145→
   146→    def stop(self, color):
   147→        self._timer.stop()
   148→        self._active = False
   149→        self._color = color
   150→        self.color_changed.emit(color)
   151→
   152→    def _tick(self):
   153→        self._on = not self._on
   154→        if self._on:
   155→            self.color_changed.emit(self._color)
   156→        else:
   157→            alpha_color = SLATE if _HAS_DESIGN else "#555555"
   158→            self.color_changed.emit(alpha_color)

   161→# ── Fallback ClaudeWorker (text-only, no tools) ──────────
   162→
   163→class _FallbackClaudeWorker(QtCore.QThread):
   164→    """Minimal text-only Claude streaming worker.
   165→
   166→    Used when synapse.panel.claude_worker cannot be imported.
   167→    Streams text tokens but does NOT support tool use.
   168→    """
   169→
   170→    token_received = QtCore.Signal(str)
   171→    stream_done = QtCore.Signal()
   172→    stream_error = QtCore.Signal(str)
   173→    # Stubs -- never emitted, but wired so the panel doesn't crash
   174→    tool_requested = QtCore.Signal(object)
   175→    tool_status = QtCore.Signal(str, str)
   176→
   177→    _API_HOST = "api.anthropic.com"
   178→    _API_PATH = "/v1/messages"
   179→    _API_VERSION = "2023-06-01"
   180→    _MODEL = "claude-sonnet-4-20250514"
   181→    _MAX_TOKENS = 4096
   182→
   183→    def __init__(self, messages, system_prompt="", parent=None):
   184→        super().__init__(parent)
   185→        self._messages = copy.deepcopy(messages)
   186→        self._system = system_prompt
   187→        self._abort = False
   188→
   189→    def abort(self):
   190→        self._abort = True
   191→
   192→    def get_messages(self):
   193→        return copy.deepcopy(self._messages)
   194→
   195→    def run(self):
   196→        try:
   197→            api_key = os.environ.get("ANTHROPIC_API_KEY", "")
   198→            if not api_key:
   199→                self.stream_error.emit("ANTHROPIC_API_KEY not set in environment")
   200→                return
   201→
   202→            body = {
   203→                "model": self._MODEL,
   204→                "max_tokens": self._MAX_TOKENS,
   205→                "stream": True,
   206→                "messages": self._messages,
   207→            }
   208→            if self._system:
   209→                body["system"] = self._system
   210→
   211→            payload = json.dumps(body).encode("utf-8")
   212→            ctx = ssl.create_default_context()
   213→            conn = http.client.HTTPSConnection(
   214→                self._API_HOST, timeout=60, context=ctx
   215→            )
   216→
   217→            try:
   218→                conn.request(
   219→                    "POST", self._API_PATH, body=payload,
   220→                    headers={
   221→                        "Content-Type": "application/json",
   222→                        "x-api-key": api_key,
   223→                        "anthropic-version": self._API_VERSION,
   224→                    },
   225→                )
   226→                response = conn.getresponse()
   227→                if response.status != 200:
   228→                    error_body = response.read().decode("utf-8", errors="replace")
   229→                    self.stream_error.emit(
   230→                        "API error {}: {}".format(response.status, error_body[:200])
   231→                    )
   232→                    return
   233→
   234→                self._parse_stream(response)
   235→                if not self._abort:
   236→                    self.stream_done.emit()
   237→            finally:
   238→                conn.close()
   239→
   240→        except Exception as exc:
   241→            self.stream_error.emit(str(exc))
   242→
   243→    def _parse_stream(self, response):
   244→        buf = ""
   245→        while True:
   246→            if self._abort:
   247→                return
   248→            chunk = response.read(4096)
   249→            if not chunk:
   250→                break
   251→            buf += chunk.decode("utf-8", errors="replace")
   252→            while "\n" in buf:
   253→                line, buf = buf.split("\n", 1)
   254→                line = line.strip()
   255→                if line.startswith("data:"):
   256→                    data_str = line[5:].strip()
   257→                    if data_str and data_str != "[DONE]":
   258→                        try:
   259→                            data = json.loads(data_str)
   260→                            delta = data.get("delta", {})
   261→                            text = delta.get("text", "")
   262→                            if text:
   263→                                self.token_received.emit(text)
   264→                        except (json.JSONDecodeError, KeyError):
   265→                            pass

   589→    # ── Message Deduplication ─────────────────────────────

   591→    def _should_show_message(self, msg):
   592→        """Return True if this message hasn't been shown in the last 5 seconds."""
   593→        now = time.time()
   594→        h = hashlib.md5(msg.encode("utf-8", errors="replace")).hexdigest()
   595→        last = self._msg_dedup.get(h, 0)
   596→        if now - last < 5.0:
   597→            return False
   598→        self._msg_dedup[h] = now
   599→        # Prune old entries
   600→        if len(self._msg_dedup) > 100:
   601→            cutoff = now - 10.0
   602→            self._msg_dedup = {
   603→                k: v for k, v in self._msg_dedup.items() if v > cutoff
   604→            }
   605→        return True

   650→    @staticmethod
   651→    def _copy_text(text):
   652→        """Copy text to clipboard via clip.exe (Windows) or Qt fallback."""
   653→        import platform, subprocess
   654→        if platform.system() == "Windows":
   655→            try:
   656→                p = subprocess.run(
   657→                    ["clip"], input=text.encode("utf-8"),
   658→                    creationflags=0x08000000, timeout=5,
   659→                )
   660→                return p.returncode == 0
   661→            except Exception:
   662→                pass
   663→        try:
   664→            app = QtWidgets.QApplication.instance() or QtGui.QGuiApplication.instance()
   665→            if app:
   666→                app.clipboard().setText(text)
   667→                return True
   668→        except Exception:
   669→            pass
   670→        return False

   857→    def _do_fix(self):
   858→        """Apply auto-fixes from the last /diagnose run."""
   859→        if self._last_diagnosis is None:
   860→            self._sys("No diagnosis to fix. Run /diagnose first.")
   861→            return
   862→        fixable = [i for i in self._last_diagnosis.issues if i.fix_available and i.fix_fn is not None]
   863→        if not fixable:
   864→            self._sys("No fixable issues in the last diagnosis.")
   865→            self._last_diagnosis = None
   866→            return
   867→        self._log_activity("Applying {} auto-fix(es)...".format(len(fixable)), "action")
   868→        try:
   869→            from synapse.panel.scene_doctor import apply_fixes
   870→            result = apply_fixes(self._last_diagnosis.issues)
   871→            msg = "Fixed {}, failed {}, skipped {}".format(
   872→                result["fixed"], result["failed"], result["skipped"]
   873→            )
   874→            self._sys(msg)
   875→            self._log_activity(msg, "success" if result["failed"] == 0 else "warning")
   876→            # Log to journal
   877→            try:
   878→                from synapse.panel.session_journal import get_journal
   879→                get_journal().log_command("/fix", msg)
   880→            except Exception:
   881→                pass
   882→            self._last_diagnosis = None
   883→            # Re-run diagnose to show remaining issues
   884→            self._do_diagnose("")
   885→        except ImportError:
   886→            self._sys("Scene doctor unavailable (synapse.panel.scene_doctor not found)")
   887→        except Exception as e:
   888→            self._sys("Fix error: " + str(e))
