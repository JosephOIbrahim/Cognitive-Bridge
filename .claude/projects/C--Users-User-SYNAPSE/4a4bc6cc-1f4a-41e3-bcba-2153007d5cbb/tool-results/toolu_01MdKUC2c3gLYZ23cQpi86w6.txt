   106→class StatusDotAnimator(QtCore.QObject):
   107→    """Pulses the status dot opacity for 'executing' state."""
   108→
   109→    color_changed = QtCore.Signal(str)
   110→
   111→    def __init__(self, parent=None):
   112→        super().__init__(parent)
   113→        self._timer = QtCore.QTimer(self)
   114→        self._timer.setInterval(600)
   115→        self._timer.timeout.connect(self._tick)
   116→        self._on = True
   117→        self._color = SIGNAL if _HAS_DESIGN else "#00D4FF"
   118→        self._active = False
   119→
   120→    def start(self, color):
   121→        self._color = color
   122→        self._active = True
   123→        self._on = True
   124→        self._timer.start()
   125→
   126→    def stop(self, color):
   127→        self._timer.stop()
   128→        self._active = False
   129→        self._color = color
   130→        self.color_changed.emit(color)
   131→
   132→    def _tick(self):
   133→        self._on = not self._on
   134→        if self._on:
   135→            self.color_changed.emit(self._color)
   136→        else:
   137→            alpha_color = SLATE if _HAS_DESIGN else "#555555"
   138→            self.color_changed.emit(alpha_color)

   141→# ── Fallback ClaudeWorker (text-only, no tools) ──────────
   142→
   143→class _FallbackClaudeWorker(QtCore.QThread):
   144→    """Minimal text-only Claude streaming worker.
   145→
   146→    Used when synapse.panel.claude_worker cannot be imported.
   147→    Streams text tokens but does NOT support tool use.
   148→    """
   149→
   150→    token_received = QtCore.Signal(str)
   151→    stream_done = QtCore.Signal()
   152→    stream_error = QtCore.Signal(str)
   153→    # Stubs -- never emitted, but wired so the panel doesn't crash
   154→    tool_requested = QtCore.Signal(object)
   155→    tool_status = QtCore.Signal(str, str)
   156→
   157→    _API_HOST = "api.anthropic.com"
   158→    _API_PATH = "/v1/messages"
   159→    _API_VERSION = "2023-06-01"
   160→    _MODEL = "claude-sonnet-4-20250514"
   161→    _MAX_TOKENS = 4096
   162→
   163→    def __init__(self, messages, system_prompt="", parent=None):
   164→        super().__init__(parent)
   165→        self._messages = copy.deepcopy(messages)
   166→        self._system = system_prompt
   167→        self._abort = False
   168→
   169→    def abort(self):
   170→        self._abort = True
   171→
   172→    def get_messages(self):
   173→        return copy.deepcopy(self._messages)
   174→
   175→    def run(self):
   176→        try:
   177→            api_key = os.environ.get("ANTHROPIC_API_KEY", "")
   178→            if not api_key:
   179→                self.stream_error.emit("ANTHROPIC_API_KEY not set in environment")
   180→                return
   181→
   182→            body = {
   183→                "model": self._MODEL,
   184→                "max_tokens": self._MAX_TOKENS,
   185→                "stream": True,
   186→                "messages": self._messages,
   187→            }
   188→            if self._system:
   189→                body["system"] = self._system
   190→
   191→            payload = json.dumps(body).encode("utf-8")
   192→            ctx = ssl.create_default_context()
   193→            conn = http.client.HTTPSConnection(
   194→                self._API_HOST, timeout=60, context=ctx
   195→            )
   196→
   197→            try:
   198→                conn.request(
   199→                    "POST", self._API_PATH, body=payload,
   200→                    headers={
   201→                        "Content-Type": "application/json",
   202→                        "x-api-key": api_key,
   203→                        "anthropic-version": self._API_VERSION,
   204→                    },
   205→                )
   206→                response = conn.getresponse()
   207→                if response.status != 200:
   208→                    error_body = response.read().decode("utf-8", errors="replace")
   209→                    self.stream_error.emit(
   210→                        "API error {}: {}".format(response.status, error_body[:200])
   211→                    )
   212→                    return
   213→
   214→                self._parse_stream(response)
   215→                if not self._abort:
   216→                    self.stream_done.emit()
   217→            finally:
   218→                conn.close()
   219→
   220→        except Exception as exc:
   221→            self.stream_error.emit(str(exc))
   222→
   223→    def _parse_stream(self, response):
   224→        buf = ""
   225→        while True:
   226→            if self._abort:
   227→                return
   228→            chunk = response.read(4096)
   229→            if not chunk:
   230→                break
   231→            buf += chunk.decode("utf-8", errors="replace")
   232→            while "\n" in buf:
   233→                line, buf = buf.split("\n", 1)
   234→                line = line.strip()
   235→                if line.startswith("data:"):
   236→                    data_str = line[5:].strip()
   237→                    if data_str and data_str != "[DONE]":
   238→                        try:
   239→                            data = json.loads(data_str)
   240→                            delta = data.get("delta", {})
   241→                            text = delta.get("text", "")
   242→                            if text:
   243→                                self.token_received.emit(text)
   244→                        except (json.JSONDecodeError, KeyError):
   245→                            pass

   549→    # ── Message Deduplication ─────────────────────────────

   551→    def _should_show_message(self, msg):
   552→        """Return True if this message hasn't been shown in the last 5 seconds."""
   553→        now = time.time()
   554→        h = hashlib.md5(msg.encode("utf-8", errors="replace")).hexdigest()
   555→        last = self._msg_dedup.get(h, 0)
   556→        if now - last < 5.0:
   557→            return False
   558→        self._msg_dedup[h] = now
   559→        # Prune old entries
   560→        if len(self._msg_dedup) > 100:
   561→            cutoff = now - 10.0
   562→            self._msg_dedup = {

   610→    @staticmethod
   611→    def _copy_text(text):
   612→        """Copy text to clipboard via clip.exe (Windows) or Qt fallback."""
   613→        import platform, subprocess
   614→        if platform.system() == "Windows":
   615→            try:
   616→                p = subprocess.run(
   617→                    ["clip"], input=text.encode("utf-8"),
   618→                    creationflags=0x08000000, timeout=5,
   619→                )
   620→                return p.returncode == 0
   621→            except Exception:
   622→                pass
   623→        try:
   624→            app = QtWidgets.QApplication.instance() or QtGui.QGuiApplication.instance()
   625→            if app:
   626→                app.clipboard().setText(text)
   627→                return True
   628→        except Exception:
   629→            pass
   630→        return False

   817→    def _do_fix(self):
   818→        """Apply auto-fixes from the last /diagnose run."""
   819→        if self._last_diagnosis is None:
   820→            self._sys("No diagnosis to fix. Run /diagnose first.")
   821→            return
   822→        fixable = [i for i in self._last_diagnosis.issues if i.fix_available and i.fix_fn is not None]
   823→        if not fixable:
   824→            self._sys("No fixable issues in the last diagnosis.")
   825→            self._last_diagnosis = None
   826→            return
   827→        self._log_activity("Applying {} auto-fix(es)...".format(len(fixable)), "action")
   828→        try:
   829→            from synapse.panel.scene_doctor import apply_fixes
   830→            result = apply_fixes(self._last_diagnosis.issues)
   831→            msg = "Fixed {}, failed {}, skipped {}".format(
   832→                result["fixed"], result["failed"], result["skipped"]
   833→            )
   834→            self._sys(msg)
   835→            self._log_activity(msg, "success" if result["failed"] == 0 else "warning")
   836→            # Log to journal
   837→            try:
   838→                from synapse.panel.session_journal import get_journal
   839→                get_journal().log_command("/fix", msg)
   840→            except Exception:
   841→                pass
   842→            self._last_diagnosis = None
   843→            # Re-run diagnose to show remaining issues
   844→            self._do_diagnose("")
   845→        except ImportError:
   846→            self._sys("Scene doctor unavailable (synapse.panel.scene_doctor not found)")
   847→        except Exception as e:
   848→            self._sys("Fix error: " + str(e))

  2076→    def _do_bookmark(self, arg=""):
  2077→        """Handle /bookmark commands."""
  2078→        try:
  2079→            from synapse.panel.bookmarks import (
  2080→                get_bookmark_manager, parse_bookmark_command,
  2081→                format_bookmarks_html,
  2082→            )
  2083→        except ImportError:
  2084→            self._sys("Bookmarks not available")
  2085→            return
  2086→
  2087→        parsed = parse_bookmark_command(
  2088→            "/bookmark " + arg if arg else "/bookmark"
  2089→        )
  2090→        manager = get_bookmark_manager()
  2091→
  2092→        if parsed["action"] == "list":
  2093→            bookmarks = manager.get_all()
  2094→            html = format_bookmarks_html(bookmarks)
  2095→            self._chat_display.append(html)
  2096→
  2097→        elif parsed["action"] == "add":
  2098→            # Bookmark the last assistant message
  2099→            preview = ""
  2100→            msg_idx = len(self._messages) - 1
  2101→            if self._messages:
  2102→                last = self._messages[-1]
  2103→                if isinstance(last.get("content"), str):
  2104→                    preview = last["content"][:100]
  2105→                elif isinstance(last.get("content"), list):
  2106→                    for block in last["content"]:
  2107→                        if isinstance(block, dict) and block.get("type") == "text":
  2108→                            preview = block["text"][:100]
  2109→                            break
  2110→
  2111→            bookmark = manager.add(parsed["label"], msg_idx, preview)
  2112→            self._sys("Bookmarked: {}".format(bookmark.label))
  2113→
  2114→            # Log to journal
  2115→            try:
  2116→                from synapse.panel.session_journal import get_journal
  2117→                get_journal().log_command("/bookmark", bookmark.label)
  2118→            except Exception:
  2119→                pass
  2120→
  2121→        elif parsed["action"] == "search":
  2122→            results = manager.search(parsed["query"])
  2123→            html = format_bookmarks_html(results, parsed["query"])
  2124→            self._chat_display.append(html)
  2125→
  2126→        elif parsed["action"] == "remove":
  2127→            if manager.remove(parsed["id"]):
  2128→                self._sys("Bookmark removed")
  2129→            else:
  2130→                self._sys("Bookmark not found: " + parsed["id"])
  2131→
  2132→        sb = self._chat_display.verticalScrollBar()
  2133→        sb.setValue(sb.maximum())
