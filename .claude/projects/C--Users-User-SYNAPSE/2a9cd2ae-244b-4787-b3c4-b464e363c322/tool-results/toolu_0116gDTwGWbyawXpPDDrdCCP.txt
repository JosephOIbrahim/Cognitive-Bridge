     3→Synapse MCP Server v2 — Bridge between Claude Desktop and Houdini via WebSocket.
     4→
     5→Connects Claude Desktop (stdio/JSON-RPC) to the Synapse WebSocket server

    10→    - Auth handshake: skip 2s recv wait when no local key configured

    15→    - Fire-and-forget warmup: stdio server starts accepting immediately
    16→    - max_size=None: skip frame validation on localhost
    17→    - get_running_loop() everywhere (deprecated get_event_loop removed)
    18→
    19→Architecture:
    20→    Claude Desktop  <-[stdio/JSON-RPC]->  mcp_server.py  <-[WebSocket]->  Synapse (Houdini)
    21→
    22→Install: pip install mcp websockets
    23→Run: python mcp_server.py
    24→"""

    52→def _get_auth_key() -> str | None:
    53→    """Get API key for authenticating with Synapse server.
    54→
    55→    Sources (checked in order):
    56→    1. SYNAPSE_API_KEY environment variable
    57→    2. ~/.synapse/auth.key file (first non-empty, non-comment line)
    58→    3. None (auth disabled)
    59→    """
    60→    env_key = os.environ.get("SYNAPSE_API_KEY", "").strip()
    61→    if env_key:
    62→        return env_key
    63→    key_path = os.path.join(os.path.expanduser("~"), ".synapse", "auth.key")
    64→    try:
    65→        if os.path.exists(key_path):
    66→            with open(key_path, encoding="utf-8") as f:
    67→                for line in f:
    68→                    line = line.strip()
    69→                    if line and not line.startswith("#"):
    70→                        return line
    71→    except OSError:
    72→        pass
    73→    return None

    76→async def _auth_handshake(ws) -> None:
    77→    """Handle auth handshake if server requires it.
    78→
    79→    v2 optimization: If no local key is configured, skip the 2s recv wait
    80→    entirely. If the server requires auth, the first command will fail with
    81→    a clear error — acceptable trade-off for eliminating 2s dead time on
    82→    every reconnect when auth is disabled (the common case).
    83→    """
    84→    key = _get_auth_key()
    85→    if not key:
    86→        # No key configured — skip the recv wait entirely (saves ~2s)
    87→        return
    88→
    89→    # Key exists — wait for auth_required from server
    90→    try:
    91→        raw = await asyncio.wait_for(ws.recv(), timeout=2.0)
    92→        msg = _loads(raw)
    93→    except (asyncio.TimeoutError, Exception):
    94→        # Server didn't send auth_required — auth not enabled server-side
    95→        return
    96→
    97→    if msg.get("type") != "auth_required":
    98→        logger.warning("Expected auth_required, got %s (auth may be disabled)", msg.get("type"))
    99→        return
   100→
   101→    await ws.send(_dumps({
   102→        "type": "authenticate",
   103→        "id": "auth-handshake",
   104→        "payload": {"key": key},
   105→    }))
   106→
   107→    # Wait for auth_success or auth_failed
   108→    try:
   109→        raw = await asyncio.wait_for(ws.recv(), timeout=5.0)
   110→        response = _loads(raw)
   111→    except asyncio.TimeoutError:
   112→        raise ConnectionError("Synapse server didn't respond to authentication within 5s")
   113→
   114→    if response.get("type") == "auth_failed":
   115→        raise ConnectionError(
   116→            f"Authentication failed: {response.get('error', 'invalid API key')}. "
   117→            "Check your SYNAPSE_API_KEY or ~/.synapse/auth.key"
   118→        )
   119→
   120→    if response.get("type") == "auth_success":
   121→        logger.info("Authenticated with Synapse server")
   122→        return
   123→
   124→    logger.warning("Unexpected auth response type: %s", response.get("type"))

   175→
   176→_ws_connection = None
   177→_ws_lock = asyncio.Lock()
   178→_pending: dict[str, asyncio.Future] = {}
   179→_recv_task: asyncio.Task | None = None
   180→
   181→
   182→def _is_connected() -> bool:
   183→    """Check if the WebSocket connection is open."""
   184→    try:
   185→        return _ws_connection is not None and _ws_connection.state == websockets.connection.State.OPEN
   186→    except (AttributeError, Exception):
   187→        return False
   188→
   189→
   190→async def _recv_loop():
   191→    """Background task: read messages and dispatch to pending futures by ID."""
   192→    global _ws_connection, _recv_task
   193→    try:
   194→        while _ws_connection is not None:
   195→            try:
   196→                raw = await _ws_connection.recv()
   197→            except Exception as e:
   198→                # Connection lost — signal all pending futures
   199→                logger.warning("Recv loop: connection lost (%s)", e)
   200→                _signal_all_pending(ConnectionError(f"Connection lost: {e}"))
   201→                break
   202→
   203→            try:
   204→                response = _loads(raw)
   205→            except Exception:
   206→                continue  # Malformed message — skip
   207→
   208→            resp_id = response.get("id")
   209→            future = _pending.pop(resp_id, None) if resp_id else None
   210→            if future and not future.done():
   211→                future.set_result(response)
   212→            elif resp_id:
   213→                logger.warning("No pending future for response ID %s (discarding)", resp_id)
   214→    finally:
   215→        _recv_task = None
   216→
   217→
   218→def _signal_all_pending(exc: Exception):
   219→    """Signal all pending futures with an exception (connection lost)."""
   220→    for fut in _pending.values():
   221→        if not fut.done():
   222→            fut.set_exception(exc)
   223→    _pending.clear()
   224→
   225→
   226→def _start_recv_loop():
   227→    """Ensure the background recv loop is running."""
   228→    global _recv_task
   229→    if _recv_task is None or _recv_task.done():
   230→        _recv_task = asyncio.get_running_loop().create_task(_recv_loop())
   231→
   232→
   233→async def _get_connection():
   234→    """Get or create a persistent WebSocket connection.
   235→
   236→    v2: Lock-free fast path — volatile check before acquiring the lock.
   237→    The common case (connection alive) skips the lock entirely.
   238→    Double-check inside lock for safety.
   239→    """
   240→    global _ws_connection, _recv_task
   241→
   242→    # Lock-free fast path: if connected, return immediately
   243→    if _is_connected():
   244→        return _ws_connection
   245→
   246→    async with _ws_lock:
   247→        # Double-check after acquiring lock
   248→        if _is_connected():
   249→            return _ws_connection
   250→
   251→        # Force cleanup of stale state — cancel lingering recv task
   252→        # to prevent the race where _start_recv_loop() sees the old
   253→        # task as "not done" and skips creating a new recv loop.
   254→        _ws_connection = None
   255→        if _recv_task is not None and not _recv_task.done():
   256→            _recv_task.cancel()
   257→            try:
   258→                await _recv_task
   259→            except (asyncio.CancelledError, Exception):
   260→                pass
   261→        _recv_task = None
   262→        # Clear any orphaned futures from the dead connection
   263→        _signal_all_pending(ConnectionError("Connection recycled"))
   264→
   265→        last_error = None
   266→        for attempt in range(MAX_RETRIES):
   267→            try:
   268→                _ws_connection = await websockets.connect(
   269→                    SYNAPSE_URL,
   270→                    open_timeout=3.0,
   271→                    close_timeout=1.0,
   272→                    ping_interval=None,
   273→                    compression=None,
   274→                    max_size=None,  # Skip frame size validation on localhost
   275→                )
   276→                await _auth_handshake(_ws_connection)
   277→                _start_recv_loop()
   278→                logger.info("Connected to Synapse at %s", SYNAPSE_URL)
   279→                return _ws_connection
   280→            except Exception as e:
   281→                last_error = e
   282→                if attempt < MAX_RETRIES - 1:
   283→                    await asyncio.sleep(RETRY_DELAY * (attempt + 1))
   284→
   285→        raise ConnectionError(
   286→            f"Couldn't connect to Synapse at {SYNAPSE_URL} after {MAX_RETRIES} attempts: {last_error}\n\n"
   287→            "Houdini might not be running, or the Synapse server hasn't started yet. "
   288→            "Launch Houdini and start the Synapse server from the Python Panel."
   289→        )

   292→async def send_command(cmd_type: str, payload: dict | None = None) -> dict:
   293→    """
   294→    Send a SynapseCommand over WebSocket and return the response data.
   295→
   296→    Supports true parallel dispatch — multiple concurrent send_command
   297→    calls share a single recv loop that routes responses by ID.
   298→    """
   299→    command_id = _cmd_id(cmd_type, payload)
   300→    command = {
   301→        "type": cmd_type,
   302→        "id": command_id,
   303→        "payload": payload or {},
   304→        "sequence": 0,
   305→        "timestamp": time.time(),
   306→        "protocol_version": PROTOCOL_VERSION,
   307→    }
   308→
   309→    cmd_timeout = _SLOW_COMMANDS.get(cmd_type, COMMAND_TIMEOUT)
   310→    last_err = None
   311→
   312→    for _attempt in range(2):  # One transparent retry on connection failure
   313→        ws = await _get_connection()
   314→
   315→        # Register a future for this command's response
   316→        loop = asyncio.get_running_loop()
   317→        future: asyncio.Future = loop.create_future()
   318→        _pending[command_id] = future
   319→
   320→        try:
   321→            await ws.send(_dumps(command))
   322→
   323→            # Direct wait on future set — avoids asyncio.wait_for's internal task overhead
   324→            done, _ = await asyncio.wait({future}, timeout=cmd_timeout)
   325→            if not done:
   326→                # Timeout — future still pending
   327→                _pending.pop(command_id, None)
   328→                future.cancel()
   329→                raise asyncio.TimeoutError()
   330→            response = future.result()
   331→            break  # Success
   332→
   333→        except asyncio.TimeoutError:
   334→            _pending.pop(command_id, None)
   335→            # Close stale connection on timeout
   336→            global _ws_connection
   337→            try:
   338→                if ws:
   339→                    await ws.close()
   340→            except Exception:
   341→                pass
   342→            _ws_connection = None
   343→            raise TimeoutError(
   344→                f"The {cmd_type} command took too long to respond \u2014 "
   345→                "Houdini may be busy with a heavy operation"
   346→            )
   347→        except ConnectionError:
   348→            # Future was signaled by _recv_loop disconnect
   349→            _pending.pop(command_id, None)
   350→            try:
   351→                if ws:
   352→                    await ws.close()
   353→            except Exception:
   354→                pass
   355→            _ws_connection = None
   356→            last_err = ConnectionError(f"Connection lost during {cmd_type}")
   357→            logger.warning("Connection lost during %s, reconnecting...", cmd_type)
   358→        except Exception as e:
   359→            _pending.pop(command_id, None)
   360→            try:
   361→                if ws:
   362→                    await ws.close()
   363→            except Exception:
   364→                pass
   365→            _ws_connection = None
   366→            last_err = e
   367→            logger.warning("Connection lost during %s, reconnecting... (%s)", cmd_type, e)
   368→    else:
   369→        raise ConnectionError(
   370→            f"Lost connection while sending {cmd_type} and couldn't reconnect: {last_err}"
   371→        )
   372→
   373→    if not response.get("success", False):
   374→        error_msg = response.get("error", "Something went wrong on the Synapse side")
   375→        data = response.get("data") or {}
   376→        if isinstance(data, dict) and "retry_after" in data:
   377→            error_msg += f" (retry after {data['retry_after']}s)"
   378→        raise RuntimeError(error_msg)
   379→
   380→    return response.get("data", {})

  1870→def _execute_python_payload(args: dict) -> dict:
  1871→    # MCP uses 'code', Synapse handler resolves 'content'
  1872→    p = {"content": args["code"]}
  1873→    if "dry_run" in args:
  1874→        p["dry_run"] = args["dry_run"]

  1881→    p = {}

  1887→def _decide_payload(args: dict) -> dict:
  1888→    p = {"decision": args["decision"]}
  1889→    if "reasoning" in args:
  1890→        p["reasoning"] = args["reasoning"]
  1891→    if "alternatives" in args:
  1892→        alt = args["alternatives"]
  1893→        p["alternatives"] = [a.strip() for a in alt.split(",") if a.strip()] if isinstance(alt, str) else alt
  1894→    return p

  1898→    p = {"content": args["content"]}

  1900→        p["memory_type"] = args["memory_type"]
  1901→    if "tags" in args:
  1902→        p["tags"] = args["tags"]
  1903→    return p

  1989→@server.call_tool()
  1990→async def call_tool(name: str, arguments: dict):
  1991→    """Dispatch MCP tool call to Synapse via WebSocket."""
  1992→    if name not in TOOL_DISPATCH:
  1993→        return [TextContent(type="text", text=f"I don't recognize the tool '{name}' \u2014 check the available tools list")]
  1994→
  1995→    cmd_type, build_payload = TOOL_DISPATCH[name]
  1996→
  1997→    try:
  1998→        payload = build_payload(arguments)
  1999→        data = await send_command(cmd_type, payload)
  2000→
  2001→        # Image-producing tools: read file and return as ImageContent
  2002→        if name in ("houdini_capture_viewport", "houdini_render"):
  2003→            import base64
  2004→            image_path = data.get("image_path", "")
  2005→            try:
  2006→                with open(image_path, "rb") as f:
  2007→                    b64 = base64.b64encode(f.read()).decode("utf-8")
  2008→                mime = "image/jpeg" if data.get("format") == "jpeg" else "image/png"
  2009→                meta = {
  2010→                    "width": data.get("width"),
  2011→                    "height": data.get("height"),
  2012→                    "format": data.get("format"),
  2013→                }
  2014→                # Include render-specific metadata
  2015→                if "engine" in data:
  2016→                    meta["rop"] = data.get("rop")
  2017→                    meta["rop_type"] = data.get("rop_type")
  2018→                    meta["engine"] = data.get("engine")
  2019→                return [
  2020→                    ImageContent(type="image", data=b64, mimeType=mime),
  2021→                    TextContent(type="text", text=_dumps_str(meta)),
  2022→                ]
  2023→            except FileNotFoundError:
  2024→                return [TextContent(type="text", text=(
  2025→                    f"The capture ran but the image file wasn't found at {image_path} \u2014 "
  2026→                    "it may have been cleaned up or the path changed"
  2027→                ))]
  2028→
  2029→        return [TextContent(type="text", text=_dumps_str(data))]
  2030→    except ConnectionError as e:
  2031→        return [TextContent(type="text", text=(
  2032→            f"Couldn't reach Synapse \u2014 {e}"
  2033→        ))]
  2034→    except RuntimeError as e:
  2035→        return [TextContent(type="text", text=f"Synapse hit a snag: {e}")]
  2036→    except Exception as e:
  2037→        logger.exception("Unexpected error in tool %s", name)
  2038→        return [TextContent(type="text", text=f"Something unexpected happened: {e}")]
  2039→

  2045→async def _warmup():
  2046→    """Pre-connect to Synapse on startup — makes first tool call instant."""
  2047→    global _ws_connection
  2048→    try:
  2049→        _ws_connection = await websockets.connect(
  2050→            SYNAPSE_URL,
  2051→            open_timeout=3.0,
  2052→            close_timeout=1.0,
  2053→            ping_interval=None,
  2054→            compression=None,
  2055→            max_size=None,
  2056→        )
  2057→        await _auth_handshake(_ws_connection)
  2058→        _start_recv_loop()
  2059→        logger.info("Warmup: connected to %s", SYNAPSE_URL)
  2060→    except Exception:
  2061→        logger.info("Warmup: Synapse not available yet (will retry on first tool call)")

  2064→def _atexit_cleanup():
  2065→    """Close the persistent WebSocket on interpreter shutdown."""
  2066→    global _ws_connection
  2067→    if _ws_connection is not None:
  2068→        try:
  2069→            loop = asyncio.get_event_loop()
  2070→            if not loop.is_closed():
  2071→                loop.run_until_complete(_ws_connection.close())
  2072→        except Exception:
  2073→            pass
  2074→        _ws_connection = None
  2075→    _signal_all_pending(ConnectionError("MCP server shutting down"))
  2076→
  2077→atexit.register(_atexit_cleanup)
