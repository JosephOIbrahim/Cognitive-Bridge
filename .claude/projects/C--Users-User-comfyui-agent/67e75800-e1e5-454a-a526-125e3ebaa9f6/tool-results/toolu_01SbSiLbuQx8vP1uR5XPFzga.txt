    64→def _get_source_tier(title: str, url: str) -> str:
    65→    """Determine source tier: 'core', 'partner', or 'community'."""
    66→    url_lower = (url or "").lower()
    67→    title_lower = (title or "").lower()
    68→
    69→    # Check core
    70→    if url_lower in _CORE_URLS or "comfyanonymous" in url_lower:
    71→        return "core"
    72→
    73→    # Check partner nodes
    74→    for partner_info in PARTNER_NODES.values():
    75→        if partner_info["url"].lower() == url_lower:
    76→            return "partner"
    77→        if any(kw in title_lower for kw in partner_info["keywords"]):
    78→            return "partner"
    79→
    80→    return "community"

   260→def _load_custom_nodes() -> list[dict]:
   261→    """Load and cache custom-node-list.json."""
   262→    if _cache["custom_nodes"] is not None:
   263→        return _cache["custom_nodes"]
   264→
   265→    path = _MANAGER_DIR / "custom-node-list.json"
   266→    if not path.exists():
   267→        return []
   268→
   269→    try:
   270→        data = json.loads(path.read_text(encoding="utf-8"))
   271→        _cache["custom_nodes"] = data.get("custom_nodes", [])
   272→        _freshness["custom_nodes_loaded_at"] = time.time()

   274→        _cache["custom_nodes"] = []

   297→def _build_node_to_pack() -> dict[str, dict]:
   298→    """Build reverse index: node_type -> {url, title, node_types}."""
   299→    if _cache["node_to_pack"] is not None:
   300→        return _cache["node_to_pack"]
   301→
   302→    ext_map = _load_extension_map()
   303→    index = {}
   304→    for url, entry in sorted(ext_map.items()):  # He2025: sorted iteration for deterministic collision resolution
   305→        if not isinstance(entry, list) or len(entry) < 2:
   306→            continue
   307→        node_types = entry[0] if isinstance(entry[0], list) else []
   308→        meta = entry[1] if isinstance(entry[1], dict) else {}
   309→        title = meta.get("title_aux", url.split("/")[-1] if "/" in url else url)
   310→
   311→        pack_info = {"url": url, "title": title, "node_count": len(node_types)}
   312→        for nt in node_types:
   313→            index[nt] = pack_info
   314→
   315→    _cache["node_to_pack"] = index
   316→    return index

   361→def _normalize_result(
   362→    name: str,
   363→    result_type: str,
   364→    source: str,
   365→    relevance_score: float,
   366→    installed: bool,
   367→    url: str,
   368→    **extra,
   369→) -> dict:
   370→    """Common result schema for unified discovery."""
   371→    result = {
   372→        "installed": installed,
   373→        "name": name,
   374→        "relevance_score": round(relevance_score, 3),
   375→        "source": source,
   376→        "source_tier": _get_source_tier(name, url),
   377→        "type": result_type,
   378→        "url": url,
   379→    }
   380→    # He2025: sorted keys via sorted extra insertion
   381→    for k in sorted(extra):
   382→        if extra[k] is not None:
   383→            result[k] = extra[k]
   384→    return result

   387→def _search_nodes_unified(query: str, max_results: int) -> list[dict]:
   388→    """Search custom node packs, return normalized results."""
   389→    packs = _load_custom_nodes()
   390→    if not packs:
   391→        return []
   392→
   393→    query_lower = query.lower()
   394→    query_words = query_lower.split()
   395→    scored = []
   396→
   397→    for pack in packs:
   398→        title = pack.get("title", "")
   399→        desc = pack.get("description", "")
   400→        author = pack.get("author", "")
   401→        searchable = f"{title} {desc} {author}".lower()
   402→
   403→        word_hits = sum(1 for w in query_words if w in searchable)
   404→        if word_hits == 0:
   405→            continue
   406→
   407→        score = word_hits / max(len(query_words), 1)
   408→        if query_lower in title.lower():
   409→            score = min(score + 0.5, 1.0)
   410→
   411→        ref = pack.get("reference", "")
   412→        scored.append((score, _normalize_result(
   413→            name=title,
   414→            result_type="node_pack",
   415→            source="registry",
   416→            relevance_score=score,
   417→            installed=_is_pack_installed(ref) if ref else False,
   418→            url=ref,
   419→            author=author,
   420→            description=desc[:200] if desc else None,
   421→        )))
   422→
   423→    scored.sort(key=lambda x: (-x[0], x[1]["name"]))
   424→    return [r for _, r in scored[:max_results]]

   541→def _deduplicate(results: list[dict]) -> list[dict]:
   542→    """Deduplicate by lowercased name. Keep highest-scored, note also_found_on."""
   543→    seen: dict[str, dict] = {}  # lowercase name -> result dict
   544→    for r in results:
   545→        key = r["name"].lower().strip()
   546→        if key in seen:
   547→            existing = seen[key]
   548→            # Track sources
   549→            if r["source"] not in existing.get("also_found_on", []):
   550→                existing.setdefault("also_found_on", []).append(r["source"])
   551→                existing["also_found_on"] = sorted(existing["also_found_on"])
   552→            # Keep higher score
   553→            if r["relevance_score"] > existing["relevance_score"]:
   554→                also = existing.get("also_found_on", [])
   555→                also_with_old = sorted(set(also) | {existing["source"]})
   556→                r["also_found_on"] = [s for s in also_with_old if s != r["source"]]
   557→                if r["also_found_on"]:
   558→                    r["also_found_on"] = sorted(r["also_found_on"])
   559→                else:
   560→                    r.pop("also_found_on", None)
   561→                seen[key] = r
   562→        else:
   563→            seen[key] = r
   564→    return list(seen.values())

   570→def _rank_results(results: list[dict]) -> list[dict]:
   571→    """Rank: installed first, tier (core>partner>community), relevance, name."""
   572→    return sorted(
   573→        results,
   574→        key=lambda r: (
   575→            not r.get("installed", False),            # installed first
   576→            _TIER_RANK.get(r.get("source_tier", "community"), 2),  # tier rank
   577→            -r.get("relevance_score", 0),              # higher score first
   578→            r.get("name", ""),                         # alphabetical tiebreaker
   579→        ),
   580→    )

   584→    """Unified discovery across registry, CivitAI, and HuggingFace."""

   586→    category = tool_input.get("category", "all")
   587→    sources = tool_input.get("sources")
   588→    model_type = tool_input.get("model_type")
   589→    base_model = tool_input.get("base_model")
   590→    sort = tool_input.get("sort", "most_downloaded")
   591→    max_results = tool_input.get("max_results", 5)
   592→
   593→    all_results: list[dict] = []
   594→    sources_searched: list[str] = []
   595→    errors: list[dict] = []
   596→
   597→    # Determine which sources to search
   598→    if sources is None:
   599→        search_registry = True
   600→        search_civitai = category in ("models", "all")
   601→        search_hf = category in ("models", "all")
   602→    else:
   603→        search_registry = "registry" in sources
   604→        search_civitai = "civitai" in sources
   605→        search_hf = "huggingface" in sources
   606→
   607→    # 1. Registry nodes (if applicable)
   608→    if search_registry and category in ("nodes", "all"):
   609→        node_results = _search_nodes_unified(query, max_results)
   610→        all_results.extend(node_results)
   611→        if "registry_nodes" not in sources_searched:
   612→            sources_searched.append("registry_nodes")
   613→
   614→    # 2. Registry models (if applicable)
   615→    if search_registry and category in ("models", "all"):
   616→        model_results = _search_models_unified(query, model_type, max_results)
   617→        all_results.extend(model_results)
   618→        if "registry_models" not in sources_searched:
   619→            sources_searched.append("registry_models")
   620→
   621→    # 3. CivitAI (models only)
   622→    if search_civitai and category in ("models", "all"):
   623→        civitai_results, civitai_err = _search_civitai_unified(
   624→            query, model_type, base_model, sort, max_results,
   625→        )
   626→        all_results.extend(civitai_results)
   627→        sources_searched.append("civitai")
   628→        if civitai_err:
   629→            errors.append({"source": "civitai", "error": civitai_err})
   630→
   631→    # 4. HuggingFace (models only)
   632→    if search_hf and category in ("models", "all"):
   633→        hf_results, hf_err = _search_hf_unified(query, model_type, max_results)
   634→        all_results.extend(hf_results)
   635→        sources_searched.append("huggingface")
   636→        if hf_err:
   637→            errors.append({"source": "huggingface", "error": hf_err})
   638→
   639→    # Deduplicate and rank
   640→    deduped = _deduplicate(all_results)
   641→    ranked = _rank_results(deduped)
   642→
   643→    response: dict = {
   644→        "query": query,
   645→        "category": category,
   646→        "results": ranked,
   647→        "sources_searched": sorted(sources_searched),
   648→        "total": len(ranked),
   649→    }
   650→    if errors:
   651→        response["errors"] = errors
   652→
   653→    return to_json(response)

   671→def _search_by_node_type(query: str, max_results: int) -> str:
   672→    """Find which pack provides a specific node type."""
   673→    index = _build_node_to_pack()
   674→
   675→    if not index:
   676→        return to_json({
   677→            "error": (
   678→                "ComfyUI Manager not found. Install it from "
   679→                "https://github.com/ltdrdata/ComfyUI-Manager"
   680→            ),
   681→        })
   682→
   683→    # Exact match first
   684→    if query in index:
   685→        pack = index[query]
   686→        return to_json({
   687→            "match": "exact",
   688→            "node_type": query,
   689→            "pack": {
   690→                "title": pack["title"],
   691→                "url": pack["url"],
   692→                "node_count": pack["node_count"],
   693→                "installed": _is_pack_installed(pack["url"]),
   694→            },
   695→        })
   696→
   697→    # Fuzzy match — case-insensitive substring, sorted for determinism
   698→    query_lower = query.lower()
   699→    matches = []
   700→    seen_packs = set()
   701→    for nt, pack in sorted(index.items()):
   702→        if query_lower in nt.lower() and pack["url"] not in seen_packs:
   703→            seen_packs.add(pack["url"])
   704→            matches.append({
   705→                "node_type": nt,
   706→                "pack_title": pack["title"],
   707→                "pack_url": pack["url"],
   708→                "installed": _is_pack_installed(pack["url"]),
   709→            })
   710→            if len(matches) >= max_results:
   711→                break
   712→
   713→    if not matches:
   714→        return to_json({
   715→            "match": "none",
   716→            "node_type": query,
   717→            "message": f"No pack found providing '{query}'. Check the spelling.",
   718→            "total_known_types": len(index),
   719→        })
   720→
   721→    return to_json({
   722→        "match": "fuzzy",
   723→        "query": query,
   724→        "results": matches,
   725→        "total_matches": len(matches),
   726→    })

   729→def _search_by_name(query: str, max_results: int) -> str:
   730→    """Search custom node packs by title/description/author."""
   731→    packs = _load_custom_nodes()
   732→
   733→    if not packs:
   734→        return to_json({
   735→            "error": (
   736→                "ComfyUI Manager not found. Install it from "
   737→                "https://github.com/ltdrdata/ComfyUI-Manager"
   738→            ),
   739→        })
   740→
   741→    query_lower = query.lower()
   742→    query_words = query_lower.split()
   743→    scored = []
   744→
   745→    for pack in packs:
   746→        title = pack.get("title", "")
   747→        desc = pack.get("description", "")
   748→        author = pack.get("author", "")
   749→        searchable = f"{title} {desc} {author}".lower()
   750→
   751→        # Score: all words match > some words match
   752→        word_hits = sum(1 for w in query_words if w in searchable)
   753→        if word_hits == 0:
   754→            continue
   755→
   756→        # Boost exact title match
   757→        score = word_hits
   758→        if query_lower in title.lower():
   759→            score += 10
   760→
   761→        ref = pack.get("reference", "")
   762→        scored.append((score, {
   763→            "title": title,
   764→            "author": author,
   765→            "description": desc[:200] if desc else "",
   766→            "url": ref,
   767→            "install_type": pack.get("install_type", ""),
   768→            "installed": _is_pack_installed(ref) if ref else False,
   769→        }))
   770→
   771→    scored.sort(key=lambda x: (-x[0], x[1].get("title", "")))
   772→    results = [r for _, r in scored[:max_results]]
   773→
   774→    return to_json({
   775→        "query": query,
   776→        "results": results,
   777→        "total_matches": len(scored),
   778→        "showing": len(results),
   779→    })

   857→def _search_huggingface(query: str, model_type: str | None, max_results: int) -> str:
   858→    """Search HuggingFace Hub API for models."""
   859→    params = {
   860→        "search": query,
   861→        "sort": "downloads",
   862→        "direction": "-1",
   863→        "limit": max_results,
   864→    }
   865→
   866→    # Map model types to HuggingFace tags
   867→    if model_type:
   868→        type_tags = {
   869→            "checkpoint": "diffusers",
   870→            "lora": "lora",
   871→            "vae": "vae",
   872→            "controlnet": "controlnet",
   873→            "embedding": "textual-inversion",
   874→            "3d": "image-to-3d",
   875→            "image-to-3d": "image-to-3d",
   876→            "text-to-3d": "text-to-3d",
   877→            "text-to-speech": "text-to-speech",
   878→            "tts": "text-to-speech",
   879→            "audio": "text-to-audio",
   880→            "text-to-audio": "text-to-audio",
   881→            "text-to-video": "text-to-video",
   882→            "video": "text-to-video",
   883→            "image-to-video": "image-to-video",
   884→        }
   885→        tag = type_tags.get(model_type.lower())
   886→        if tag:
   887→            params["filter"] = tag
   888→
   889→    if not HUGGINGFACE_LIMITER().acquire(timeout=5.0):
   890→        return to_json({"error": "Rate limited — too many HuggingFace requests. Try again shortly."})
   891→
   892→    try:
   893→        with httpx.Client() as client:
   894→            resp = client.get(
   895→                "https://huggingface.co/api/models",
   896→                params=params,
   897→                timeout=15.0,
   898→            )
   899→            resp.raise_for_status()
   900→            data = resp.json()
   901→    except httpx.ConnectError:
   902→        return to_json({"error": "Could not reach HuggingFace API. Check your internet connection."})
   903→    except httpx.HTTPStatusError as e:
   904→        return to_json({"error": f"HuggingFace API returned {e.response.status_code}."})
   905→    except Exception as e:
   906→        return to_json({"error": f"HuggingFace search failed: {e}"})
   907→
   908→    results = []
   909→    for item in data:
   910→        model_id = item.get("modelId", item.get("id", ""))
   911→        results.append({
   912→            "name": model_id,
   913→            "author": model_id.split("/")[0] if "/" in model_id else "",
   914→            "downloads": item.get("downloads", 0),
   915→            "likes": item.get("likes", 0),
   916→            "tags": item.get("tags", [])[:5],
   917→            "url": f"https://huggingface.co/{model_id}",
   918→            "last_modified": item.get("lastModified", ""),
   919→        })
   920→
   921→    return to_json({
   922→        "source": "huggingface",
   923→        "query": query,
   924→        "type_filter": model_type,
   925→        "results": results,
   926→        "showing": len(results),
   927→    })

   930→def _handle_find_missing_nodes(tool_input: dict) -> str:
   931→    path_str = tool_input.get("path")
   932→
   933→    # Get workflow
   934→    if path_str:
   935→        path = Path(path_str)
   936→        if not path.exists():
   937→            return to_json({"error": f"File not found: {path_str}"})
   938→        try:
   939→            data = json.loads(path.read_text(encoding="utf-8"))
   940→        except json.JSONDecodeError as e:
   941→            return to_json({"error": f"Invalid JSON: {e}"})
   942→
   943→        # Extract API-format nodes
   944→        if "nodes" in data and isinstance(data["nodes"], list):
   945→            prompt_data = data.get("extra", {}).get("prompt")
   946→            if prompt_data and isinstance(prompt_data, dict):
   947→                workflow = {
   948→                    k: v for k, v in prompt_data.items()
   949→                    if isinstance(v, dict) and "class_type" in v
   950→                }
   951→            else:
   952→                # UI-only: extract class_types from nodes array
   953→                workflow = {}
   954→                for node in data["nodes"]:
   955→                    nid = str(node.get("id", ""))
   956→                    ntype = node.get("type", "")
   957→                    if ntype:
   958→                        workflow[nid] = {"class_type": ntype, "inputs": {}}
   959→        else:
   960→            workflow = {
   961→                k: v for k, v in data.items()
   962→                if isinstance(v, dict) and "class_type" in v
   963→            }
   964→    else:
   965→        from .workflow_patch import get_current_workflow
   966→        workflow = get_current_workflow()
   967→        if workflow is None:
   968→            return to_json({
   969→                "error": (
   970→                    "No workflow loaded. Either provide a 'path' or load one "
   971→                    "with apply_workflow_patch first."
   972→                ),
   973→            })
   974→
   975→    # Collect unique class_types
   976→    class_types = set()
   977→    for node in workflow.values():
   978→        ct = node.get("class_type")
   979→        if ct:
   980→            class_types.add(ct)
   981→
   982→    if not class_types:
   983→        return to_json({"error": "No nodes found in workflow."})
   984→
   985→    # Check which are available in live ComfyUI
   986→    available = set()
   987→    try:
   988→        with httpx.Client() as client:
   989→            resp = client.get(f"{COMFYUI_URL}/object_info", timeout=15.0)
   990→            resp.raise_for_status()
   991→            object_info = resp.json()
   992→            available = set(object_info.keys())
   993→    except httpx.ConnectError:
   994→        return to_json({
   995→            "error": "ComfyUI not reachable. Start ComfyUI to check node availability.",
   996→        })
   997→    except Exception as e:
   998→        return to_json({"error": f"Could not fetch node info: {e}"})
   999→
  1000→    # Classify
  1001→    installed = sorted(class_types & available)
  1002→    missing = sorted(class_types - available)
  1003→
  1004→    # Look up missing nodes in extension-node-map
  1005→    index = _build_node_to_pack()
  1006→    missing_report = []
  1007→    packs_needed = {}  # url -> {title, nodes}
  1008→
  1009→    for node_type in missing:
  1010→        pack_info = index.get(node_type)
  1011→        if pack_info:
  1012→            url = pack_info["url"]
  1013→            if url not in packs_needed:
  1014→                packs_needed[url] = {
  1015→                    "title": pack_info["title"],
  1016→                    "url": url,
  1017→                    "installed": _is_pack_installed(url),
  1018→                    "missing_nodes": [],
  1019→                }
  1020→            packs_needed[url]["missing_nodes"].append(node_type)
  1021→
  1022→            missing_report.append({
  1023→                "node_type": node_type,
  1024→                "pack_title": pack_info["title"],
  1025→                "pack_url": url,
  1026→                "pack_installed": _is_pack_installed(url),
  1027→            })
  1028→        else:
  1029→            missing_report.append({
  1030→                "node_type": node_type,
  1031→                "pack_title": None,
  1032→                "pack_url": None,
  1033→                "message": "Not found in ComfyUI Manager registry.",
  1034→            })
  1035→
  1036→    # Summary
  1037→    if not missing:
  1038→        return to_json({
  1039→            "status": "all_installed",
  1040→            "total_node_types": len(class_types),
  1041→            "message": "All node types in this workflow are available.",
  1042→        })
  1043→
  1044→    return to_json({
  1045→        "status": "missing_nodes",
  1046→        "total_node_types": len(class_types),
  1047→        "installed_count": len(installed),
  1048→        "missing_count": len(missing),
  1049→        "missing_nodes": missing_report,
  1050→        "packs_to_install": list(packs_needed.values()),
  1051→        "install_hint": (
  1052→            "Install missing packs via ComfyUI Manager or git clone into "
  1053→            f"{CUSTOM_NODES_DIR}"
  1054→        ),
  1055→    })

  1191→def _file_age_info(path: Path) -> dict:
  1192→    """Get age info for a file. Returns dict with exists, modified_at, age_s, age_human."""
  1193→    if not path.exists():
  1194→        return {"exists": False, "path": str(path)}
  1195→
  1196→    mtime = path.stat().st_mtime
  1197→    age_s = time.time() - mtime
  1198→    age_days = age_s / 86400
  1199→
  1200→    if age_days < 1:
  1201→        age_human = f"{age_s / 3600:.1f} hours"
  1202→    else:
  1203→        age_human = f"{age_days:.1f} days"
  1204→
  1205→    if age_s > _WARN_THRESHOLD:
  1206→        status = "very_stale"
  1207→    elif age_s > _STALE_THRESHOLD:
  1208→        status = "stale"
  1209→    else:
  1210→        status = "fresh"
  1211→
  1212→    return {
  1213→        "exists": True,
  1214→        "path": str(path),
  1215→        "age_seconds": round(age_s),
  1216→        "age_human": age_human,
  1217→        "status": status,
  1218→    }
