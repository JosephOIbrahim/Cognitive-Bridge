    41→def _parse(result: str) -> dict:
    42→    """Parse a tool result JSON string."""
    43→    return json.loads(result)
    44→
    45→
    46→def _make_intent_spec(
    47→    model_id: str = "sdxl-base",
    48→    confidence: float = 0.85,
    49→    *,
    50→    mutations: list[ParameterMutation] | None = None,
    51→    prompt_mutations: list[PromptMutation] | None = None,
    52→    warnings: list[str] | None = None,
    53→    fallback: bool = False,
    54→) -> IntentSpecification:
    55→    """Build a test IntentSpecification."""
    56→    return IntentSpecification(
    57→        model_id=model_id,
    58→        parameter_mutations=mutations or [],
    59→        prompt_mutations=prompt_mutations or [],
    60→        confidence=confidence,
    61→        warnings=warnings or [],
    62→        using_fallback_profile=fallback,
    63→    )
    64→
    65→
    66→def _make_verification(
    67→    decision: str = "accept",
    68→    overall_score: float = 0.85,
    69→    intent_alignment: float = 0.9,
    70→    technical_quality: float = 0.8,
    71→    *,
    72→    refinement_actions: list[RefinementAction] | None = None,
    73→    iteration_count: int = 0,
    74→    max_iterations: int = 3,
    75→    diagnosed_issues: list[str] | None = None,
    76→) -> VerificationResult:
    77→    """Build a test VerificationResult."""
    78→    return VerificationResult(
    79→        overall_score=overall_score,
    80→        intent_alignment=intent_alignment,
    81→        technical_quality=technical_quality,
    82→        decision=decision,
    83→        refinement_actions=refinement_actions or [],
    84→        iteration_count=iteration_count,
    85→        max_iterations=max_iterations,
    86→        diagnosed_issues=diagnosed_issues or [],
    87→    )
