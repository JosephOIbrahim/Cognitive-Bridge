     1→     1→# FORGE Solaris Pattern Extraction Pipeline
     2→     2→## Learn Wiring Patterns from Houdini 21's Own Examples
     3→     3→
     4→     4→> **Mission:** Crawl Houdini 21's built-in example .hip files, extract every 
     5→     5→> Solaris/LOP network topology, cluster them into canonical wiring patterns, 
     6→     6→> feed them into the FORGE corpus as observations, and auto-generate RAG 
     7→     7→> documents from crystallized rules.
     8→     8→>
     9→     9→> **D1 metaphor:** We're filming the relay team's form — studying how SideFX 
    10→    10→> themselves wire Solaris networks — then teaching SYNAPSE to replicate 
    11→    11→> those handoffs.
    12→    12→
    13→    13→---
    14→    14→
    15→    15→## Why This Works
    16→    16→
    17→    17→Houdini ships with ~200 example .hip files under `$HFS/houdini/help/examples/`.
    18→    18→The LOP/Solaris examples show SideFX's own canonical wiring patterns — these 
    19→    19→are ground truth for how Solaris networks should be assembled. We already have 
    20→    20→`network_explain` which extracts topology from any network. The missing piece 
    21→    21→is connecting these two systems through FORGE.
    22→    22→
    23→    23→---
    24→    24→
    25→    25→## Architecture
    26→    26→
    27→    27→```
    28→    28→Phase 1: CRAWL
    29→    29→  $HFS/houdini/help/examples/ → find all .hip/.hipnc with /stage networks
    30→    30→  Output: hip_manifest.json (paths + metadata)
    31→    31→
    32→    32→Phase 2: EXTRACT  
    33→    33→  For each .hip: load → run network_explain on /stage → capture topology
    34→    34→  Output: raw_topologies.json (node types, connections, ordering)
    35→    35→
    36→    36→Phase 3: CLUSTER
    37→    37→  Group topologies by pattern signature (type sequence hash)
    38→    38→  Identify: canonical chains, common fragments, frequency counts
    39→    39→  Output: pattern_clusters.json
    40→    40→
    41→    41→Phase 4: CORPUS INGEST
    42→    42→  Feed clusters into FORGE corpus as OBSERVATIONS
    43→    43→  Clusters seen 3+ times auto-promote to PATTERNS
    44→    44→  Patterns with 5+ recurrences promote to RULES
    45→    45→  Output: forge/corpus/observations/EX-*.json
    46→    46→
    47→    47→Phase 5: RAG GENERATE
    48→    48→  Crystallized RULES → auto-generate RAG markdown documents
    49→    49→  Output: rag/skills/houdini21-reference/solaris_patterns_extracted.md
    50→    50→
    51→    51→Phase 6: VERIFY (FORGE cycle)
    52→    52→  Run FORGE scenarios that test wiring against extracted patterns
    53→    53→  Measure: does SYNAPSE wire like SideFX's examples?
    54→    54→```

   134→   134→def crawl_examples(
   135→   135→    output_path: str = "forge/extraction_data/hip_manifest.json",
   136→   136→    examples_subdir: str = "houdini/help/examples",
   137→   137→    include_non_solaris: bool = False,
   138→   138→) -> HipManifest:
   139→   139→    """Crawl Houdini example directory for .hip files with Solaris content.
   140→   140→
   141→   141→    Must be run inside Houdini (needs hou module).
   142→   142→
   143→   143→    Args:
   144→   144→        output_path: Where to write the manifest JSON.
   145→   145→        examples_subdir: Subdirectory under $HFS to scan.
   146→   146→        include_non_solaris: If True, include .hip files without /stage.
   147→   147→
   148→   148→    Returns:
   149→   149→        HipManifest with discovered files.
   150→   150→    """

   161→   161→    if not os.path.isdir(examples_root):
   162→   162→        # Also check alternative locations
   163→   163→        alt_paths = [

   167→   167→        ]
   168→   168→        for alt in alt_paths:
   169→   169→            if os.path.isdir(alt):
   170→   170→                examples_root = alt
   171→   171→                break
   172→   172→        else:
   173→   173→            print(f"ERROR: No examples directory found under {hfs}")
   174→   174→            return manifest
   175→   175→
   176→   176→    print(f"Scanning: {examples_root}")
   177→   177→
   178→   178→    # Find all .hip and .hipnc files
   179→   179→    hip_files = []
   180→   180→    for root, dirs, files in os.walk(examples_root):
   181→   181→        for f in files:
   182→   182→            if f.endswith((".hip", ".hipnc")):
   183→   183→                hip_files.append(os.path.join(root, f))

   186→   186→
   187→   187→    for i, hip_path in enumerate(hip_files):

   193→   193→

   197→   197→        elif include_non_solaris:
   198→   198→            manifest.entries.append(entry)
   199→   199→
   200→   200→    # Sort by category then filename
   201→   201→    manifest.entries.sort(key=lambda e: (e.category, e.filename))
   202→   202→
   203→   203→    # Save manifest

   208→   208→    print(f"\nCrawl complete:")

   215→   215→
   216→   216→def _scan_hip_file(hip_path: str, examples_root: str) -> HipEntry:
   217→   217→    """Open a .hip file and check for Solaris content."""
   218→   218→    import hou
   219→   219→
   220→   220→    filename = os.path.basename(hip_path)
   221→   221→    rel_path = os.path.relpath(hip_path, examples_root)
   222→   222→    category = rel_path.split(os.sep)[0] if os.sep in rel_path else "root"
   223→   223→
   224→   224→    try:
   225→   225→        hou.hipFile.load(hip_path, suppress_save_prompt=True, ignore_load_warnings=True)
   226→   226→
   227→   227→        stage = hou.node("/stage")
   228→   228→        if stage is None:
   229→   229→            # Check for lopnet anywhere in the scene
   230→   230→            for child in hou.node("/").children():
   231→   231→                if child.type().name() == "lopnet":
   232→   232→                    stage = child
   233→   233→                    break
   234→   234→
   235→   235→        has_stage = stage is not None
   236→   236→        children = stage.children() if stage else []
   237→   237→        lop_types = sorted(set(c.type().name() for c in children))
   238→   238→
   239→   239→        return HipEntry(
   240→   240→            path=hip_path,
   241→   241→            filename=filename,
   242→   242→            category=category,
   243→   243→            has_stage=has_stage,
   244→   244→            stage_child_count=len(children),
   245→   245→            lop_node_types=lop_types,
   246→   246→            file_size_kb=os.path.getsize(hip_path) // 1024,
   247→   247→        )
   248→   248→
   249→   249→    except Exception as e:
   250→   250→        return HipEntry(
   251→   251→            path=hip_path,
   252→   252→            filename=filename,
   253→   253→            category=category,
   254→   254→            has_stage=False,
   255→   255→            stage_child_count=0,
   256→   256→            lop_node_types=[],
   257→   257→            file_size_kb=os.path.getsize(hip_path) // 1024,
   258→   258→            error=str(e)[:200],
   259→   259→        )
   260→   260→```

   412→   412→def _extract_topology(hip_path: str, category: str) -> TopologyRecord:
   413→   413→    """Extract topology from a single .hip file."""
   414→   414→    import hou
   415→   415→
   416→   416→    try:
   417→   417→        hou.hipFile.load(hip_path, suppress_save_prompt=True, ignore_load_warnings=True)
   418→   418→
   419→   419→        stage = hou.node("/stage")
   420→   420→        if stage is None:
   421→   421→            for child in hou.node("/").children():
   422→   422→                if child.type().name() == "lopnet":
   423→   423→                    stage = child
   424→   424→                    break
   425→   425→
   426→   426→        if stage is None or not stage.children():
   427→   427→            return TopologyRecord(
   428→   428→                source_hip=hip_path,
   429→   429→                category=category,
   430→   430→                node_count=0,
   431→   431→                nodes=[],
   432→   432→                chain_signature="empty",
   433→   433→                chain_types=[],
   434→   434→                connections=[],
   435→   435→                has_linear_chain=False,
   436→   436→                branch_count=0,
   437→   437→                display_flag_node="",
   438→   438→                patterns_detected=[],
   439→   439→                extraction_error="No /stage or empty network",
   440→   440→            )
   441→   441→
   442→   442→        children = stage.children()
   443→   443→
   444→   444→        # Topological sort (Kahn's algorithm — same as handlers_node.py)
   445→   445→        sorted_nodes = _topo_sort(children)
   446→   446→
   447→   447→        # Extract node records
   448→   448→        nodes = []
   449→   449→        connections = []
   450→   450→        display_node = ""
   451→   451→        type_set = set()
   452→   452→
   453→   453→        for n in sorted_nodes:
   454→   454→            type_name = n.type().name()
   455→   455→            type_set.add(type_name)
   456→   456→
   457→   457→            inputs_from = [inp.name() for inp in n.inputs() if inp]
   458→   458→            outputs_to = [out.name() for out in n.outputs() if out]
   459→   459→
   460→   460→            for inp in n.inputs():
   461→   461→                if inp:
   462→   462→                    connections.append((inp.name(), n.name()))
   463→   463→
   464→   464→            # Check for primpath parameter
   465→   465→            has_primpath = False
   466→   466→            primpath_value = ""
   467→   467→            try:
   468→   468→                pp = n.parm("primpath")
   469→   469→                if pp:
   470→   470→                    has_primpath = True
   471→   471→                    primpath_value = pp.eval()
   472→   472→            except Exception:
   473→   473→                pass
   474→   474→
   475→   475→            # Get non-default params (lightweight version)
   476→   476→            key_params = _get_key_params(n)
   477→   477→
   478→   478→            is_display = False
   479→   479→            is_render = False
   480→   480→            try:
   481→   481→                is_display = n.isDisplayFlagSet()
   482→   482→                is_render = n.isRenderFlagSet() if hasattr(n, "isRenderFlagSet") else False
   483→   483→            except Exception:
   484→   484→                pass
   485→   485→
   486→   486→            if is_display:
   487→   487→                display_node = n.name()
   488→   488→
   489→   489→            nodes.append(NodeRecord(
   490→   490→                name=n.name(),
   491→   491→                type_name=type_name,
   492→   492→                type_label=n.type().description(),
   493→   493→                position_y=n.position()[1],
   494→   494→                inputs_from=inputs_from,
   495→   495→                outputs_to=outputs_to,
   496→   496→                input_count=len(inputs_from),
   497→   497→                output_count=len(outputs_to),
   498→   498→                is_display_flag=is_display,
   499→   499→                is_render_flag=is_render,
   500→   500→                has_primpath=has_primpath,
   501→   501→                primpath_value=primpath_value,
   502→   502→                key_params=key_params,
   503→   503→            ))
   504→   504→
   505→   505→        # Chain analysis
   506→   506→        chain_types = [n.type_name for n in nodes]
   507→   507→        chain_sig = _compute_signature(chain_types)
   508→   508→
   509→   509→        # Detect linear vs branching
   510→   510→        max_inputs = max((n.input_count for n in nodes), default=0)
   511→   511→        root_count = sum(1 for n in nodes if n.input_count == 0)
   512→   512→        has_linear = max_inputs <= 1 and root_count <= 1
   513→   513→        branch_count = max(1, root_count)
   514→   514→
   515→   515→        # Detect known patterns
   516→   516→        patterns = _detect_patterns(type_set, nodes)
   517→   517→
   518→   518→        return TopologyRecord(
   519→   519→            source_hip=hip_path,
   520→   520→            category=category,
   521→   521→            node_count=len(nodes),
   522→   522→            nodes=nodes,
   523→   523→            chain_signature=chain_sig,
   524→   524→            chain_types=chain_types,
   525→   525→            connections=connections,
   526→   526→            has_linear_chain=has_linear,
   527→   527→            branch_count=branch_count,
   528→   528→            display_flag_node=display_node,
   529→   529→            patterns_detected=patterns,
   530→   530→        )
   531→   531→
   532→   532→    except Exception as e:
   533→   533→        return TopologyRecord(
   534→   534→            source_hip=hip_path,
   535→   535→            category=category,
   536→   536→            node_count=0,
   537→   537→            nodes=[],
   538→   538→            chain_signature="error",
   539→   539→            chain_types=[],
   540→   540→            connections=[],
   541→   541→            has_linear_chain=False,
   542→   542→            branch_count=0,
   543→   543→            display_flag_node="",
   544→   544→            patterns_detected=[],
   545→   545→            extraction_error=str(e)[:300],
   546→   546→        )
   547→   547→
   548→   548→
   549→   549→def _topo_sort(nodes) -> list:
   550→   550→    """Topological sort by input dependencies. Deterministic (name tie-break)."""
   551→   551→    node_set = {id(n) for n in nodes}
   552→   552→    id_map = {id(n): n for n in nodes}
   553→   553→    in_deg = {id(n): 0 for n in nodes}
   554→   554→    adj = {id(n): [] for n in nodes}
   555→   555→
   556→   556→    for n in nodes:
   557→   557→        for inp in n.inputs():
   558→   558→            if inp and id(inp) in node_set:
   559→   559→                in_deg[id(n)] += 1
   560→   560→                adj[id(inp)].append(id(n))
   561→   561→
   562→   562→    queue = sorted([n for n in nodes if in_deg[id(n)] == 0], key=lambda n: n.name())
   563→   563→    result = []
   564→   564→    while queue:
   565→   565→        node = queue.pop(0)
   566→   566→        result.append(node)
   567→   567→        for nid in sorted(adj[id(node)], key=lambda x: id_map[x].name()):
   568→   568→            in_deg[nid] -= 1
   569→   569→            if in_deg[nid] == 0:
   570→   570→                queue.append(id_map[nid])
   571→   571→        queue.sort(key=lambda n: n.name())
   572→   572→
   573→   573→    remaining = sorted([n for n in nodes if n not in result], key=lambda n: n.name())
   574→   574→    result.extend(remaining)
   575→   575→    return result
   576→   576→
   577→   577→
   578→   578→def _get_key_params(node, max_params: int = 10) -> Dict[str, Any]:
   579→   579→    """Get non-default params, limited to max_params for storage efficiency."""
   580→   580→    params = {}
   581→   581→    try:
   582→   582→        for parm in node.parms():
   583→   583→            if len(params) >= max_params:
   584→   584→                break
   585→   585→            try:
   586→   586→                template = parm.parmTemplate()
   587→   587→                type_name = template.type().name() if hasattr(template.type(), "name") else ""
   588→   588→                if "Folder" in type_name or "Separator" in type_name:
   589→   589→                    continue
   590→   590→                current = parm.eval()
   591→   591→                defaults = template.defaultValue()
   592→   592→                default_val = defaults[0] if isinstance(defaults, tuple) and defaults else defaults
   593→   593→                if current != default_val:
   594→   594→                    params[parm.name()] = current
   595→   595→            except Exception:
   596→   596→                continue
   597→   597→    except Exception:
   598→   598→        pass
   599→   599→    return params
   600→   600→
   601→   601→
   602→   602→def _compute_signature(type_list: List[str]) -> str:
   603→   603→    """Compute a hash signature from a sequence of type names."""
   604→   604→    import hashlib
   605→   605→    content = "|".join(type_list)
   606→   606→    return hashlib.sha256(content.encode()).hexdigest()[:16]
   607→   607→
   608→   608→
   609→   609→def _detect_patterns(type_set: Set[str], nodes: list) -> List[str]:
   610→   610→    """Detect known Solaris workflow patterns."""
   611→   611→    detected = []
   612→   612→    for pattern_name, required_types in sorted(_PATTERN_SIGNATURES.items()):
   613→   613→        if required_types.issubset(type_set):
   614→   614→            # Special case: three_point needs 3+ rectlights
   615→   615→            if pattern_name == "three_point_lighting":
   616→   616→                rect_count = sum(1 for n in nodes if n.type_name == "rectlight")
   617→   617→                if rect_count >= 3:
   618→   618→                    detected.append(pattern_name)
   619→   619→            else:
   620→   620→                detected.append(pattern_name)
   621→   621→    return detected
   622→   622→```

   626→   626→## Phase 3: Pattern Clusterer
   627→   627→
   628→   628→### File: `forge/extractors/pattern_clusterer.py`
   629→   629→
   630→   630→```python
   631→   631→"""
   632→   632→FORGE Pattern Clusterer — Group extracted topologies into canonical patterns.
   633→   633→
   634→   634→Takes raw topologies from Phase 2, clusters them by:
   635→   635→  1. Exact chain signature (identical type sequences)
   636→   636→  2. Subsequence similarity (shared fragments)
   637→   637→  3. Functional role clustering (lighting chains, material chains, etc.)
   638→   638→
   639→   639→Outputs canonical patterns with frequency counts, which become
   640→   640→the training data for SYNAPSE's wiring knowledge.
   641→   641→
   642→   642→Runs outside Houdini (pure Python, no hou dependency).
   643→   643→"""
   644→   644→
   645→   645→from __future__ import annotations
   646→   646→
   647→   647→import json
   648→   648→import os
   649→   649→import time
   650→   650→from collections import Counter, defaultdict
   651→   651→from dataclasses import asdict, dataclass, field
   652→   652→from typing import Any, Dict, List, Optional, Set, Tuple
   653→   653→
   654→   654→
   655→   655→@dataclass
   656→   656→class CanonicalPattern:
   657→   657→    """A validated wiring pattern discovered from examples."""
   658→   658→    pattern_id: str
   659→   659→    name: str                        # Human-readable name
   660→   660→    description: str
   661→   661→    chain_types: List[str]           # Canonical type sequence
   662→   662→    frequency: int                   # How many examples use this pattern
   663→   663→    source_hips: List[str]           # Which .hip files contain it
   664→   664→    is_linear: bool                  # Single chain or branching
   665→   665→    functional_role: str             # "lighting", "material", "render", etc.
   666→   666→    connection_rules: List[Dict]     # [{source_type, target_type, input_idx}]
   667→   667→    display_flag_type: str           # Which type typically gets display flag
   668→   668→    common_params: Dict[str, Any]    # Most common non-default param values
   669→   669→    confidence: float                # frequency / total_examples
   670→   670→
   671→   671→
   672→   672→@dataclass
   673→   673→class PatternClusterResult:
   674→   674→    """Output of the clustering phase."""
   675→   675→    cluster_time: str
   676→   676→    total_topologies: int
   677→   677→    unique_signatures: int
   678→   678→    canonical_patterns: List[CanonicalPattern] = field(default_factory=list)
   679→   679→    # Fragments: common subsequences found across multiple patterns
   680→   680→    common_fragments: List[Dict[str, Any]] = field(default_factory=list)
   681→   681→    # Type frequency: how often each node type appears across all examples
   682→   682→    type_frequency: Dict[str, int] = field(default_factory=dict)
   683→   683→    # Connection frequency: how often each connection type appears
   684→   684→    connection_frequency: Dict[str, int] = field(default_factory=dict)
   685→   685→
   686→   686→    def to_dict(self):
   687→   687→        return {
   688→   688→            "cluster_time": self.cluster_time,
   689→   689→            "total_topologies": self.total_topologies,
   690→   690→            "unique_signatures": self.unique_signatures,
   691→   691→            "canonical_patterns": [asdict(p) for p in self.canonical_patterns],
   692→   692→            "common_fragments": self.common_fragments,
   693→   693→            "type_frequency": dict(sorted(
   694→   694→                self.type_frequency.items(), key=lambda x: -x[1]
   695→   695→            )),
   696→   696→            "connection_frequency": dict(sorted(
   697→   697→                self.connection_frequency.items(), key=lambda x: -x[1]
   698→   698→            )),
   699→   699→        }
   700→   700→
   701→   701→
   702→   702→def cluster_patterns(
   703→   703→    topologies_path: str = "forge/extraction_data/raw_topologies.json",
   704→   704→    output_path: str = "forge/extraction_data/pattern_clusters.json",
   705→   705→    min_frequency: int = 2,
   706→   706→) -> PatternClusterResult:
   707→   707→    """Cluster extracted topologies into canonical patterns.
   708→   708→
   709→   709→    Args:
   710→   710→        topologies_path: Input from Phase 2.
   711→   711→        output_path: Where to write pattern_clusters.json.
   712→   712→        min_frequency: Minimum occurrences to be considered a pattern.
   713→   713→
   714→   714→    Returns:
   715→   715→        PatternClusterResult with discovered patterns.
   716→   716→    """
   717→   717→    with open(topologies_path, encoding="utf-8") as f:
   718→   718→        data = json.load(f)
   719→   719→
   720→   720→    topologies = data.get("topologies", [])
   721→   721→    total = len(topologies)
   722→   722→
   723→   723→    result = PatternClusterResult(
   724→   724→        cluster_time=time.strftime("%Y-%m-%dT%H:%M:%SZ"),
   725→   725→        total_topologies=total,
   726→   726→    )
   727→   727→
   728→   728→    if not topologies:
   729→   729→        return result
   730→   730→
   731→   731→    # ── Step 1: Exact signature clustering ──
   732→   732→    sig_groups: Dict[str, List[Dict]] = defaultdict(list)
   733→   733→    for topo in topologies:
   734→   734→        sig_groups[topo["chain_signature"]].append(topo)
   735→   735→
   736→   736→    result.unique_signatures = len(sig_groups)
   737→   737→
   738→   738→    # ── Step 2: Build canonical patterns from frequent signatures ──
   739→   739→    pattern_id = 0
   740→   740→    for sig, group in sorted(sig_groups.items(), key=lambda x: -len(x[1])):
   741→   741→        if len(group) < min_frequency:
   742→   742→            continue
   743→   743→
   744→   744→        pattern_id += 1
   745→   745→        representative = group[0]  # Use first as template
   746→   746→        chain_types = representative["chain_types"]
   747→   747→
   748→   748→        # Determine functional role from node types
   749→   749→        role = _infer_role(set(chain_types))
   750→   750→
   751→   751→        # Extract connection rules from the representative
   752→   752→        conn_rules = []
   753→   753→        for conn in representative.get("connections", []):
   754→   754→            src_name, tgt_name = conn
   755→   755→            # Find types from nodes list
   756→   756→            src_type = _find_type(representative["nodes"], src_name)
   757→   757→            tgt_type = _find_type(representative["nodes"], tgt_name)
   758→   758→            if src_type and tgt_type:
   759→   759→                conn_rules.append({
   760→   760→                    "source_type": src_type,
   761→   761→                    "target_type": tgt_type,
   762→   762→                    "input_index": 0,
   763→   763→                })
   764→   764→
   765→   765→        # Aggregate common params across the group
   766→   766→        common_params = _aggregate_params(group)
   767→   767→
   768→   768→        # Display flag consensus
   769→   769→        display_types = Counter(
   770→   770→            _find_type(t["nodes"], t.get("display_flag_node", ""))
   771→   771→            for t in group
   772→   772→            if t.get("display_flag_node")
   773→   773→        )
   774→   774→        display_flag_type = display_types.most_common(1)[0][0] if display_types else "null"
   775→   775→
   776→   776→        pattern = CanonicalPattern(
   777→   777→            pattern_id=f"EXP-{pattern_id:03d}",
   778→   778→            name=_generate_name(chain_types, role),
   779→   779→            description=f"Canonical {role} chain: {' → '.join(chain_types)}",
   780→   780→            chain_types=chain_types,
   781→   781→            frequency=len(group),
   782→   782→            source_hips=[t["source_hip"] for t in group],
   783→   783→            is_linear=all(t.get("has_linear_chain", False) for t in group),
   784→   784→            functional_role=role,
   785→   785→            connection_rules=conn_rules,
   786→   786→            display_flag_type=display_flag_type,
   787→   787→            common_params=common_params,
   788→   788→            confidence=len(group) / total,
   789→   789→        )
   790→   790→        result.canonical_patterns.append(pattern)
   791→   791→
   792→   792→    # ── Step 3: Extract common fragments (subsequences) ──
   793→   793→    result.common_fragments = _find_common_fragments(topologies, min_frequency)
   794→   794→
   795→   795→    # ── Step 4: Global type and connection frequency ──
   796→   796→    type_counter: Counter = Counter()
   797→   797→    conn_counter: Counter = Counter()
   798→   798→    for topo in topologies:
   799→   799→        for t in topo.get("chain_types", []):
   800→   800→            type_counter[t] += 1
   801→   801→        for conn in topo.get("connections", []):
   802→   802→            src_type = _find_type(topo["nodes"], conn[0])
   803→   803→            tgt_type = _find_type(topo["nodes"], conn[1])
   804→   804→            if src_type and tgt_type:
   805→   805→                conn_counter[f"{src_type} → {tgt_type}"] += 1
   806→   806→
   807→   807→    result.type_frequency = dict(type_counter)
   808→   808→    result.connection_frequency = dict(conn_counter)
   809→   809→
   810→   810→    # Sort patterns by frequency descending
   811→   811→    result.canonical_patterns.sort(key=lambda p: -p.frequency)
   812→   812→
   813→   813→    # Save
   814→   814→    os.makedirs(os.path.dirname(output_path), exist_ok=True)
   815→   815→    with open(output_path, "w", encoding="utf-8") as f:
   816→   816→        json.dump(result.to_dict(), f, indent=2)
   817→   817→
   818→   818→    print(f"\nClustering complete:")
   819→   819→    print(f"  Topologies: {total}")
   820→   820→    print(f"  Unique signatures: {result.unique_signatures}")
   821→   821→    print(f"  Canonical patterns (freq>={min_frequency}): {len(result.canonical_patterns)}")
   822→   822→    print(f"  Common fragments: {len(result.common_fragments)}")
   823→   823→    print(f"  Top 5 node types: {type_counter.most_common(5)}")
   824→   824→    print(f"  Top 5 connections: {conn_counter.most_common(5)}")
   825→   825→
   826→   826→    return result
   827→   827→
   828→   828→
   829→   829→def _infer_role(type_set: Set[str]) -> str:
   830→   830→    """Infer functional role from node types."""
   831→   831→    if {"rectlight", "domelight"} & type_set:
   832→   832→        return "lighting"
   833→   833→    if {"materiallibrary", "assignmaterial"} & type_set:
   834→   834→        return "material"
   835→   835→    if {"karmarenderproperties", "usdrender"} & type_set:
   836→   836→        return "render"
   837→   837→    if {"camera"} & type_set and not ({"rectlight", "materiallibrary"} & type_set):
   838→   838→        return "camera"
   839→   839→    if {"sopcreate", "sopimport"} & type_set and not ({"camera", "rectlight"} & type_set):
   840→   840→        return "geometry"
   841→   841→    if {"reference", "sublayer"} & type_set:
   842→   842→        return "composition"
   843→   843→    if len(type_set) >= 4:
   844→   844→        return "scene_assembly"
   845→   845→    return "general"
   846→   846→
   847→   847→
   848→   848→def _generate_name(chain_types: List[str], role: str) -> str:
   849→   849→    """Generate a human-readable pattern name."""
   850→   850→    type_str = "_".join(dict.fromkeys(chain_types))  # Deduplicated, ordered
   851→   851→    return f"{role}_{type_str}"[:80]
   852→   852→
   853→   853→
   854→   854→def _find_type(nodes: List[Dict], name: str) -> Optional[str]:
   855→   855→    """Find node type by name in a nodes list."""
   856→   856→    for n in nodes:
   857→   857→        if n["name"] == name:
   858→   858→            return n["type_name"]
   859→   859→    return None
   860→   860→
   861→   861→
   862→   862→def _aggregate_params(group: List[Dict]) -> Dict[str, Any]:
   863→   863→    """Find most common non-default params across a group of topologies."""
   864→   864→    param_values: Dict[str, Counter] = defaultdict(Counter)
   865→   865→    for topo in group:
   866→   866→        for node in topo.get("nodes", []):
   867→   867→            for k, v in node.get("key_params", {}).items():
   868→   868→                param_values[k][str(v)] += 1
   869→   869→
   870→   870→    # Keep params that appear in >50% of the group
   871→   871→    threshold = len(group) / 2
   872→   872→    common = {}
   873→   873→    for param, counter in sorted(param_values.items()):
   874→   874→        top_value, top_count = counter.most_common(1)[0]
   875→   875→        if top_count >= threshold:
   876→   876→            common[param] = top_value
   877→   877→    return common
   878→   878→
   879→   879→
   880→   880→def _find_common_fragments(
   881→   881→    topologies: List[Dict], min_freq: int
   882→   882→) -> List[Dict[str, Any]]:
   883→   883→    """Find common 2-4 node subsequences across topologies."""
   884→   884→    fragment_counter: Counter = Counter()
   885→   885→    fragment_sources: Dict[str, List[str]] = defaultdict(list)
   886→   886→
   887→   887→    for topo in topologies:
   888→   888→        chain = topo.get("chain_types", [])
   889→   889→        seen = set()
   890→   890→        # Extract all 2-4 length subsequences
   891→   891→        for window_size in range(2, min(5, len(chain) + 1)):
   892→   892→            for i in range(len(chain) - window_size + 1):
   893→   893→                fragment = tuple(chain[i : i + window_size])
   894→   894→                frag_key = " → ".join(fragment)
   895→   895→                if frag_key not in seen:
   896→   896→                    seen.add(frag_key)
   897→   897→                    fragment_counter[frag_key] += 1
   898→   898→                    fragment_sources[frag_key].append(topo["source_hip"])
   899→   899→
   900→   900→    # Return fragments above threshold
   901→   901→    fragments = []
   902→   902→    for frag_key, count in fragment_counter.most_common(50):
   903→   903→        if count >= min_freq:
   904→   904→            fragments.append({
   905→   905→                "fragment": frag_key,
   906→   906→                "types": frag_key.split(" → "),
   907→   907→                "frequency": count,
   908→   908→                "confidence": count / len(topologies),
   909→   909→                "source_count": len(set(fragment_sources[frag_key])),
   910→   910→            })
   911→   911→
   912→   912→    return fragments
   913→   913→```

   977→   977→
   978→   978→    Args:
   979→   979→        clusters_path: Input from Phase 3.
   980→   980→        corpus_dir: FORGE corpus directory.
   981→   981→        cycle_number: Current FORGE cycle (0 = extraction run).
   982→   982→
   983→   983→    Returns:
   984→   984→        Summary dict with counts.
   985→   985→    """
   986→   986→    with open(clusters_path, encoding="utf-8") as f:
   987→   987→        data = json.load(f)
   988→   988→

   990→   990→    patterns = data.get("canonical_patterns", [])
   991→   991→    fragments = data.get("common_fragments", [])
   992→   992→
   993→   993→    stats = {
   994→   994→        "observations_created": 0,
   995→   995→        "auto_promoted_patterns": 0,
   996→   996→        "auto_promoted_rules": 0,
   997→   997→        "fragments_ingested": 0,
   998→   998→        "skipped_duplicates": 0,
   999→   999→    }
  1000→  1000→
  1001→  1001→    # ── Ingest canonical patterns ──

  1003→  1003→        entry_id = f"EX-{pat['pattern_id']}"
  1004→  1004→        chain_str = " → ".join(pat["chain_types"])
  1005→  1005→
  1006→  1006→        # Check if this pattern already exists
  1007→  1007→        existing = corpus.search(chain_str, top_k=1)

  1011→  1011→            corpus._save_entry(existing[0])
  1012→  1012→            stats["skipped_duplicates"] += 1
  1013→  1013→            continue
  1014→  1014→
  1015→  1015→        domain = _ROLE_TO_DOMAIN.get(pat["functional_role"], ScenarioDomain.GENERAL)
  1016→  1016→
  1017→  1017→        entry = CorpusEntry(
  1018→  1018→            id=entry_id,
  1019→  1019→            created_cycle=cycle_number,
  1020→  1020→            created_by=AgentRole.RESEARCHER,  # Extraction acts as RESEARCHER
  1021→  1021→            stage=CorpusStage.OBSERVATION,
  1022→  1022→            category=f"solaris_wiring_{pat['functional_role']}",
  1023→  1023→            pattern=(
  1024→  1024→                f"Solaris {pat['functional_role']} chain: {chain_str}. "
  1025→  1025→                f"Linear={pat['is_linear']}. "
  1026→  1026→                f"Display flag on {pat['display_flag_type']}. "
  1027→  1027→                f"Seen in {pat['frequency']} SideFX examples."
  1028→  1028→            ),
  1029→  1029→            context=(

  1032→  1032→                f"Common params: {json.dumps(pat['common_params'])}."
  1033→  1033→            ),
  1034→  1034→            domain=domain,
  1035→  1035→            confidence=min(0.9, pat["confidence"] * 3),  # Boost: SideFX examples are authoritative
  1036→  1036→            recurrence_count=pat["frequency"],
  1037→  1037→            derived_from=[f"hip:{hip}" for hip in pat["source_hips"][:5]],
  1038→  1038→        )
  1039→  1039→
  1040→  1040→        # Auto-promote based on frequency (SideFX examples are high-confidence)
  1041→  1041→        if pat["frequency"] >= 5:
  1042→  1042→            entry.stage = CorpusStage.RULE
  1043→  1043→            entry.confidence = min(0.95, entry.confidence)
  1044→  1044→            entry.promoted_at = time.strftime("%Y-%m-%dT%H:%M:%SZ")
  1045→  1045→            stats["auto_promoted_rules"] += 1

  1047→  1047→            entry.stage = CorpusStage.PATTERN
  1048→  1048→            entry.confidence = min(0.8, entry.confidence)
  1049→  1049→            entry.promoted_at = time.strftime("%Y-%m-%dT%H:%M:%SZ")
  1050→  1050→            stats["auto_promoted_patterns"] += 1
  1051→  1051→        else:
  1052→  1052→            stats["observations_created"] += 1
  1053→  1053→

  1055→  1055→        corpus._update_manifest(entry)
  1056→  1056→
  1057→  1057→    # ── Ingest common fragments as separate observations ──

  1059→  1059→        frag_id = f"EX-FRAG-{i+1:03d}"

  1063→  1063→        if existing and existing[0].content_hash == _hash(frag_str):
  1064→  1064→            stats["skipped_duplicates"] += 1
  1065→  1065→            continue
  1066→  1066→
  1067→  1067→        entry = CorpusEntry(
  1068→  1068→            id=frag_id,
  1069→  1069→            created_cycle=cycle_number,
  1070→  1070→            created_by=AgentRole.RESEARCHER,
  1071→  1071→            stage=CorpusStage.OBSERVATION,
  1072→  1072→            category="solaris_wiring_fragment",
  1073→  1073→            pattern=(
  1074→  1074→                f"Common Solaris wiring fragment: {frag_str}. "
  1075→  1075→                f"Appears in {frag['frequency']} examples "
  1076→  1076→                f"({frag['confidence']:.0%} of all extracted networks)."
  1077→  1077→            ),
  1078→  1078→            context="Subsequence found across multiple SideFX example scenes.",
  1079→  1079→            domain=ScenarioDomain.GENERAL,
  1080→  1080→            confidence=min(0.7, frag["confidence"] * 2),
  1081→  1081→            recurrence_count=frag["frequency"],
  1082→  1082→        )
  1083→  1083→
  1084→  1084→        # Auto-promote frequent fragments
  1085→  1085→        if frag["frequency"] >= 5:
  1086→  1086→            entry.stage = CorpusStage.PATTERN
  1087→  1087→            entry.promoted_at = time.strftime("%Y-%m-%dT%H:%M:%SZ")

  1090→  1090→        corpus._update_manifest(entry)
  1091→  1091→        stats["fragments_ingested"] += 1
  1092→  1092→
  1093→  1093→    print(f"\nCorpus ingestion complete:")
  1094→  1094→    for k, v in sorted(stats.items()):
  1095→  1095→        print(f"  {k}: {v}")
  1096→  1096→
  1097→  1097→    return stats
