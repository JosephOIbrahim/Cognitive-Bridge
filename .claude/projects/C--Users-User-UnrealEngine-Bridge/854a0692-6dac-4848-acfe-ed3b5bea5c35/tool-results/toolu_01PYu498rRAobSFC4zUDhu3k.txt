    62→@contextmanager
    63→def _file_lock(file_path: Path, timeout: float = 5.0):
    64→    """Advisory file lock using msvcrt on Windows, fcntl on Unix.
    65→
    66→    Acquires an exclusive lock on a .lock file adjacent to the target.
    67→    Falls back to no-op if locking is unavailable.
    68→    """
    69→    lock_path = file_path.with_suffix(file_path.suffix + ".lock")
    70→    lock_fd = None
    71→    try:
    72→        lock_fd = open(lock_path, "w", encoding="utf-8")
    73→        try:
    74→            import msvcrt
    75→            import time
    76→            deadline = time.monotonic() + timeout
    77→            while True:
    78→                try:
    79→                    msvcrt.locking(lock_fd.fileno(), msvcrt.LK_NBLCK, 1)
    80→                    break
    81→                except OSError:
    82→                    if time.monotonic() >= deadline:
    83→                        logger.warning("File lock timeout on %s", file_path.name)
    84→                        break
    85→                    time.sleep(0.05)
    86→        except ImportError:
    87→            try:
    88→                import fcntl
    89→                fcntl.flock(lock_fd.fileno(), fcntl.LOCK_EX)
    90→            except ImportError:
    91→                pass  # No locking available — proceed without
    92→        yield
    93→    finally:
    94→        if lock_fd is not None:
    95→            try:
    96→                try:
    97→                    import msvcrt
    98→                    msvcrt.locking(lock_fd.fileno(), msvcrt.LK_UNLCK, 1)
    99→                except (ImportError, OSError):
   100→                    pass
   101→                lock_fd.close()
   102→            except OSError:
   103→                pass

   106→def _atomic_write(file_path: Path, content: str) -> None:
   107→    """Write content to file atomically via tmp + os.replace (NTFS-safe), with file locking."""
   108→    parent = file_path.parent
   109→    with _file_lock(file_path):
   110→        fd, tmp_path = tempfile.mkstemp(dir=str(parent), suffix=".tmp", prefix=".bridge_")
   111→        try:
   112→            with os.fdopen(fd, "w", encoding="utf-8") as f:
   113→                f.write(content)
   114→            os.replace(tmp_path, str(file_path))
   115→        except BaseException:
   116→            # Clean up temp file on any failure
   117→            try:
   118→                os.unlink(tmp_path)
   119→            except OSError:
   120→                pass
   121→            raise

   124→def _safe_read(file_path: Path, retries: int = 3, delay: float = 0.05) -> Optional[str]:
   125→    """Read file with retry on Windows file-lock errors."""
   126→    import time
   127→    for attempt in range(retries):
   128→        try:
   129→            return file_path.read_text(encoding="utf-8")
   130→        except (PermissionError, OSError):
   131→            if attempt < retries - 1:
   132→                time.sleep(delay * (2 ** attempt))  # Exponential backoff
   133→    return None

   140→def _validate_bridge_path(bridge_path: Optional[Path]) -> Path:
   141→    """Validate and resolve a bridge path, preventing path traversal.
   142→
   143→    All bridge file operations MUST go through this function to ensure

   147→    allowed_root = DEFAULT_BRIDGE_PATH.resolve()
   148→    try:
   149→        base_path.relative_to(allowed_root)
   150→    except ValueError:
   151→        raise ValueError(
   152→            f"Bridge path '{base_path}' is outside the allowed directory '{allowed_root}'. "
   153→            f"Path traversal is not permitted."
   154→        )
   155→    return base_path

   186→    total: int,

   190→    """
   191→    Write a question to bridge_state.usda.
   192→
   193→    Args:
   194→        question_id: Unique identifier for the question (e.g., "load", "ground")
   195→        text: Question text to display
   196→        options: List of dicts with 'label' and 'direction' keys
   197→        index: 0-based question index
   198→        total: Total number of questions
   199→        scene: Scene identifier for visual transitions
   200→        bridge_path: Optional custom bridge directory
   201→
   202→    Returns:
   203→        Path to the written USDA file
   204→
   205→    Example:
   206→        write_question_usda(
   207→            question_id="load",
   208→            text="When working on a complex problem, do you prefer to...",
   209→            options=[
   210→                {"label": "Break it into small pieces", "direction": "low"},
   211→                {"label": "See the full picture first", "direction": "high"},
   212→                {"label": "Jump between both", "direction": "mid"}

   215→            total=8
   216→        )
   217→    """
   218→    ensure_bridge_directory(bridge_path)
   219→    file_path = get_bridge_file_path(bridge_path)
   220→    timestamp = get_timestamp()
   221→
   222→    if HAS_PXR:
   223→        return _write_question_pxr(
   224→            file_path, question_id, text, options, index, total, scene, timestamp
   225→        )
   226→    else:
   227→        return _write_question_text(
   228→            file_path, question_id, text, options, index, total, scene, timestamp
   229→        )

   232→def _write_question_pxr(
   233→    file_path: Path,
   234→    question_id: str,
   235→    text: str,
   236→    options: List[Dict[str, str]],
   237→    index: int,
   238→    total: int,
   239→    scene: str,
   240→    timestamp: str
   241→) -> Path:
   242→    """Write question using pxr USD library."""
   243→    # Create or open stage
   244→    if file_path.exists():
   245→        stage = Usd.Stage.Open(str(file_path))
   246→    else:
   247→        stage = Usd.Stage.CreateNew(str(file_path))
   248→        stage.SetDefaultPrim(stage.DefinePrim("/BridgeState", "Xform"))
   249→
   250→    root = stage.GetPrimAtPath("/BridgeState")
   251→
   252→    # Set variants for state machine
   253→    vsets = root.GetVariantSets()
   254→    if vsets.HasVariantSet("sync_status"):
   255→        vsets.GetVariantSet("sync_status").SetVariantSelection("question_pending")
   256→    if vsets.HasVariantSet("message_type"):
   257→        vsets.GetVariantSet("message_type").SetVariantSelection("question")
   258→
   259→    # Write message data
   260→    msg_prim = stage.GetPrimAtPath("/BridgeState/Message")
   261→    if msg_prim:
   262→        msg_prim.GetAttribute("type").Set("question")
   263→        msg_prim.GetAttribute("index").Set(index)
   264→        msg_prim.GetAttribute("total").Set(total)
   265→        msg_prim.GetAttribute("timestamp").Set(timestamp)
   266→        msg_prim.GetAttribute("question_id").Set(question_id)
   267→        msg_prim.GetAttribute("text").Set(text)
   268→        msg_prim.GetAttribute("scene").Set(scene)
   269→        msg_prim.GetAttribute("progress_display").Set(f"{index + 1}/{total}")
   270→
   271→    # Write options
   272→    for i, opt in enumerate(options[:3]):
   273→        opt_prim = stage.GetPrimAtPath(f"/BridgeState/Options/Option_{i}")
   274→        if opt_prim:
   275→            opt_prim.GetAttribute("index").Set(i)
   276→            opt_prim.GetAttribute("label").Set(opt.get("label", ""))
   277→            opt_prim.GetAttribute("direction").Set(opt.get("direction", ""))
   278→            opt_prim.GetAttribute("semantic_tag").Set(opt.get("semantic_tag", ""))
   279→
   280→    stage.Save()
   281→    return file_path

   284→def _update_question_incremental(
   285→    file_path: Path,
   286→    question_id: str,
   287→    text: str,
   288→    options: List[Dict[str, str]],
   289→    index: int,
   290→    total: int,
   291→    scene: str,
   292→    timestamp: str
   293→) -> bool:
   294→    """Incrementally update an existing bridge_state.usda (patch, not rewrite).
   295→
   296→    Returns True if successful, False if full rewrite needed (file missing/corrupt).
   297→    """
   298→    content = _safe_read(file_path)
   299→    if content is None or 'def Xform "Message"' not in content:
   300→        return False
   301→
   302→    def esc(s: str) -> str:
   303→        return s.replace('\\', '\\\\').replace('"', '\\"').replace('\n', '\\n')
   304→
   305→    # 1. Update variants
   306→    content = re.sub(r'(string sync_status = ")[^"]*(")', r'\g<1>question_pending\g<2>', content)
   307→    content = re.sub(r'(string message_type = ")[^"]*(")', r'\g<1>question\g<2>', content)
   308→
   309→    # 2. Replace Message prim
   310→    new_message = f'''def Xform "Message" {{
   311→        string type = "question"
   312→        int index = {index}
   313→        int total = {total}
   314→        string timestamp = "{timestamp}"
   315→        string question_id = "{esc(question_id)}"
   316→        string text = "{esc(text)}"
   317→        string scene = "{esc(scene)}"
   318→        string progress_display = "{index + 1}/{total}"
   319→    }}'''
   320→    content = re.sub(r'def Xform "Message"[^}]*\}', new_message, content, flags=re.DOTALL)
   321→
   322→    # 3. Replace Options prim (rebuild with new options)
   323→    options_inner = ""
   324→    for i, opt in enumerate(options[:3]):
   325→        label = esc(opt.get("label", ""))
   326→        direction = esc(opt.get("direction", ""))
   327→        semantic_tag = esc(opt.get("semantic_tag", ""))
   328→        options_inner += f'''
   329→        def Xform "Option_{i}" {{
   330→            int index = {i}
   331→            string label = "{label}"
   332→            string direction = "{direction}"
   333→            string semantic_tag = "{semantic_tag}"
   334→        }}
   335→'''
   336→    new_options = f'def Xform "Options" {{{options_inner}    }}'
   337→    content = re.sub(r'def Xform "Options"[^}]*(?:\{[^}]*\}[^}]*)*\}', new_options, content, flags=re.DOTALL)
   338→
   339→    # 4. Reset answer prim for new question
   340→    new_answer = '''def Xform "Answer" {
   341→        string question_id = ""
   342→        int option_index = -1
   343→        double response_time_ms = 0.0
   344→        string selected_label = ""
   345→        string selected_direction = ""
   346→        string timestamp = ""
   347→    }'''
   348→    content = re.sub(r'def Xform "Answer"[^}]*\}', new_answer, content, flags=re.DOTALL)
   349→
   350→    _atomic_write(file_path, content)
   351→    return True

   514→def read_answer_usda(bridge_path: Optional[Path] = None) -> Optional[Dict[str, Any]]:
   515→    """
   516→    Read answer from bridge_state.usda.
   517→
   518→    Returns:
   519→        Dict with answer data or None if no answer available.
   520→        {
   521→            "question_id": str,
   522→            "option_index": int,
   523→            "response_time_ms": float,
   524→            "selected_label": str,
   525→            "selected_direction": str,
   526→            "timestamp": str
   527→        }
   528→    """
   529→    file_path = get_bridge_file_path(bridge_path)
   530→
   531→    if not file_path.exists():
   532→        return None
   533→
   534→    if HAS_PXR:
   535→        return _read_answer_pxr(file_path)
   536→    else:
   537→        return _read_answer_text(file_path)

   539→
   540→def _read_answer_pxr(file_path: Path) -> Optional[Dict[str, Any]]:
   541→    """Read answer using pxr USD library."""
   542→    try:
   543→        stage = Usd.Stage.Open(str(file_path))
   544→
   545→        # Check if answer is pending
   546→        root = stage.GetPrimAtPath("/BridgeState")
   547→        vsets = root.GetVariantSets()
   548→        if vsets.HasVariantSet("sync_status"):
   549→            status = vsets.GetVariantSet("sync_status").GetVariantSelection()
   550→            if status != "answer_received":
   551→                return None
   552→
   553→        # Read answer data
   554→        answer_prim = stage.GetPrimAtPath("/BridgeState/Answer")
   555→        if not answer_prim:
   556→            return None
   557→
   558→        question_id = answer_prim.GetAttribute("question_id").Get()
   559→        option_index = answer_prim.GetAttribute("option_index").Get()
   560→
   561→        if option_index < 0:
   562→            return None
   563→
   564→        return {
   565→            "question_id": question_id,
   566→            "option_index": option_index,
   567→            "response_time_ms": answer_prim.GetAttribute("response_time_ms").Get(),
   568→            "selected_label": answer_prim.GetAttribute("selected_label").Get(),
   569→            "selected_direction": answer_prim.GetAttribute("selected_direction").Get(),
   570→            "timestamp": answer_prim.GetAttribute("timestamp").Get(),
   571→        }
   572→
   573→    except Exception as e:
   574→        print(f"[USD Bridge] Error reading answer: {e}")
   575→        return None

   578→def _read_answer_text(file_path: Path) -> Optional[Dict[str, Any]]:
   579→    """Read answer using text parsing (fallback when pxr unavailable)."""
   580→    try:
   581→        content = _safe_read(file_path)
   582→        if content is None:
   583→            return None
   584→
   585→        # Check sync_status variant
   586→        sync_match = re.search(r'string sync_status = "([^"]*)"', content)
   587→        if not sync_match or sync_match.group(1) != "answer_received":
   588→            return None
   589→
   590→        # Find Answer prim section
   591→        answer_section_match = re.search(
   592→            r'def Xform "Answer"[^{]*\{([^}]*)\}',
   593→            content,
   594→            re.DOTALL
   595→        )
   596→        if not answer_section_match:
   597→            return None
   598→
   599→        answer_section = answer_section_match.group(1)
   600→
   601→        # Parse attributes
   602→        def get_attr(pattern: str, default: Any = "") -> Any:
   603→            match = re.search(pattern, answer_section)
   604→            return match.group(1) if match else default
   605→

  1096→def read_ack_usda(bridge_path: Optional[Path] = None) -> Optional[Dict[str, Any]]:
  1097→    """

  1104→

  1107→
  1108→    try:
  1109→        content = _safe_read(file_path)
  1110→        if content is None:
  1111→            return None
  1112→
  1113→        # Check message_type variant
  1114→        type_match = re.search(r'string message_type = "([^"]*)"', content)
  1115→        if not type_match or type_match.group(1) != "ack":
  1116→            return None

  1119→        ack_section_match = re.search(
  1120→            r'def Xform "Ack"[^{]*\{([^}]*)\}',
  1121→            content,

  1125→            return None

  1128→

  1130→        ready_match = re.search(r'bool ready = (true|false)', ack_section)
  1131→        ue_version_match = re.search(r'string ue_version = "([^"]*)"', ack_section)
  1132→        project_match = re.search(r'string project = "([^"]*)"', ack_section)
  1133→

  1135→

  1137→            return None

  1140→            "ready": ready,
  1141→            "ue_version": ue_version_match.group(1) if ue_version_match else "",
  1142→            "project": project_match.group(1) if project_match else "",

  1213→def compute_checksum(dimensions: Dict[str, Any]) -> str:
  1214→    """
  1215→    Compute deterministic checksum for profile (ThinkingMachines [He2025] compliant).
  1216→
  1217→    FIXED algorithm: Sort alphabetically, serialize, DJB2 hash to 8-char hex.
  1218→    Same inputs ALWAYS produce same output regardless of call order or batch size.
  1219→
  1220→    Args:
  1221→        dimensions: Dict of profile dimensions
  1222→
  1223→    Returns:
  1224→        8-character hex checksum
  1225→    """
  1226→    # FIXED: Sort alphabetically for determinism
  1227→    sorted_dims = sorted(dimensions.items())
  1228→
  1229→    # FIXED: Serialize format TRL_v1|key:value|key:value|...
  1230→    serialized = "TRL_v1|" + "|".join(f"{k}:{v}" for k, v in sorted_dims)
  1231→
  1232→    # FIXED: DJB2 hash algorithm (batch-invariant)
  1233→    hash_val = 5381
  1234→    for char in serialized:
  1235→        hash_val = ((hash_val << 5) + hash_val) + ord(char)
  1236→        hash_val &= 0xFFFFFFFF  # Keep 32-bit
  1237→
  1238→    return format(hash_val, '08x')
