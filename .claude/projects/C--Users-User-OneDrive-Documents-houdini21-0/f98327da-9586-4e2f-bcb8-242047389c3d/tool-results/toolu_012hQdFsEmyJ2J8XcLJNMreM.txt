    13→    13→### Prerequisites
    14→    14→
    15→    15→Before ANY Agent δ builder script runs, Synapse must verify:
    16→    16→
    17→    17→```python
    18→    18→# File: python/cinema_camera/synapse_preflight.py
    19→    19→"""
    20→    20→Synapse preflight check. Runs once per session.
    21→    21→Claude Code calls this through the bridge before Agent δ tasks begin.
    22→    22→"""
    23→    23→
    24→    24→def synapse_preflight() -> dict:
    25→    25→    """
    26→    26→    Verify live Houdini session is ready for HDA construction.
    27→    27→    Returns dict of verified conditions. Raises RuntimeError on failure.
    28→    28→    """
    29→    29→    import hou
    30→    30→    import os
    31→    31→
    32→    32→    result = {}
    33→    33→
    34→    34→    # 1. Houdini version
    35→    35→    major, minor, patch = hou.applicationVersion()
    36→    36→    assert major >= 21, f"Requires Houdini 21+, got {major}.{minor}.{patch}"
    37→    37→    result["houdini_version"] = f"{major}.{minor}.{patch}"
    38→    38→
    39→    39→    # 2. Cinema camera path
    40→    40→    cinema_path = os.environ.get("CINEMA_CAMERA_PATH")
    41→    41→    assert cinema_path, "CINEMA_CAMERA_PATH not set in session environment"
    42→    42→    assert os.path.isdir(cinema_path), f"CINEMA_CAMERA_PATH does not exist: {cinema_path}"
    43→    43→    result["cinema_camera_path"] = cinema_path
    44→    44→
    45→    45→    # 3. VEX include path — libcinema_optics.h must be findable
    46→    46→    vex_dir = os.path.join(cinema_path, "vex")
    47→    47→    houdini_path = os.environ.get("HOUDINI_PATH", "")
    48→    48→    if cinema_path not in houdini_path:
    49→    49→        # Append so VEX #include <libcinema_optics.h> resolves
    50→    50→        os.environ["HOUDINI_PATH"] = f"{cinema_path};{houdini_path}"
    51→    51→        hou.hscript(f'setenv HOUDINI_PATH = "{cinema_path};{houdini_path}"')
    52→    52→    result["vex_include_path"] = vex_dir
    53→    53→
    54→    54→    # 4. HDA output directories exist
    55→    55→    for subdir in ["hda/chops", "hda/post"]:
    56→    56→        full = os.path.join(cinema_path, subdir)
    57→    57→        os.makedirs(full, exist_ok=True)
    58→    58→    result["hda_dirs_ready"] = True
    59→    59→
    60→    60→    # 5. Copernicus available
    61→    61→    try:
    62→    62→        hou.nodeType(hou.copNodeTypeCategory(), "vopcop2gen")
    63→    63→        result["copernicus_available"] = True
    64→    64→    except:
    65→    65→        result["copernicus_available"] = False
    66→    66→
    67→    67→    return result
    68→    68→```
    69→    69→
    70→    70→### Session Protocol
    71→    71→
    72→    72→```
    73→    73→SYNAPSE EXECUTION ORDER:
    74→    74→  1. Claude Code calls synapse_preflight() through bridge
    75→    75→  2. If preflight passes → proceed with builder scripts

    85→    85→## PILLAR C: Operator Biomechanics — Synapse Builder
    86→    86→
    87→    87→### C.2 Builder Script (Replaces abstract HDA description)
    88→    88→
    89→    89→```python
    90→    90→# File: python/cinema_camera/builders/build_chops_biomechanics.py
    91→    91→"""
    92→    92→Synapse builder: cinema::chops_biomechanics::1.0
    93→    93→
    94→    94→Execution: Claude Code sends this to live Houdini via Synapse bridge.
    95→    95→Creates the CHOPs HDA, configures internal solvers, saves to disk.
    96→    96→
    97→    97→Dependencies:
    98→    98→  - synapse_preflight() must have passed
    99→    99→  - biomechanics.py must exist at $CINEMA_CAMERA_PATH/python/cinema_camera/
   100→   100→"""
   101→   101→
   102→   102→import hou
   103→   103→import os
   104→   104→
   105→   105→
   106→   106→def build_chops_biomechanics_hda(
   107→   107→    save_dir: str = None,
   108→   108→    hda_name: str = "cinema_chops_biomechanics_1.0.hda",
   109→   109→) -> str:
   110→   110→    """
   111→   111→    Build cinema::chops_biomechanics::1.0 HDA in live Houdini session.
   112→   112→
   113→   113→    Creates a CHOP-level HDA containing:
   114→   114→      - Inertia Calculator (Python SOP reading USD weight attribute)
   115→   115→      - Spring Solver (hou.chopNodeTypeCategory spring node)
   116→   116→      - Lag Solver (hou.chopNodeTypeCategory lag node)
   117→   117→      - Handheld Noise Generator (optional, toggle-controlled)
   118→   118→      - Export target: FluidHead xformOp:rotateXYZ
   119→   119→
   120→   120→    Returns: Absolute path to saved .hda file.
   121→   121→    """
   122→   122→    if save_dir is None:
   123→   123→        save_dir = os.path.join(os.environ["CINEMA_CAMERA_PATH"], "hda", "chops")
   124→   124→
   125→   125→    hda_path = os.path.join(save_dir, hda_name)
   126→   126→
   127→   127→    # ── 1. Create temporary container ────────────────────
   128→   128→    obj = hou.node("/obj")
   129→   129→    temp_geo = obj.createNode("geo", "__cinema_biomech_builder")
   130→   130→    temp_geo.moveToGoodPosition()
   131→   131→
   132→   132→    # Create the CHOP network that will become the HDA
   133→   133→    chop_net = temp_geo.createNode("chopnet", "cinema_biomechanics")
   134→   134→
   135→   135→    # ── 2. Input: Channel source ─────────────────────────
   136→   136→    # Raw pan/tilt/roll channels from animator or mocap
   137→   137→    ch_input = chop_net.createNode("fetch", "raw_camera_input")
   138→   138→    ch_input.parm("nodepath").set("")  # User wires this at HDA level
   139→   139→    ch_input.setComment("INPUT: Pan/Tilt/Roll channels\nConnect to animation source")
   140→   140→    ch_input.setGenericFlag(hou.nodeFlag.DisplayComment, True)
   141→   141→
   142→   142→    # ── 3. Constant node: solver parameters ──────────────
   143→   143→    # These parms get promoted to HDA interface
   144→   144→    ch_params = chop_net.createNode("constant", "solver_params")
   145→   145→    ch_params.parm("name0").set("spring_k")
   146→   146→    ch_params.parm("value0").set(15.0)
   147→   147→    ch_params.parm("name1").set("damping")
   148→   148→    ch_params.parm("value1").set(0.5)
   149→   149→    ch_params.parm("name2").set("lag_frames")
   150→   150→    ch_params.parm("value2").set(2.25)
   151→   151→    ch_params.parm("name3").set("shake_amp")
   152→   152→    ch_params.parm("value3").set(0.2)
   153→   153→    ch_params.parm("name4").set("shake_freq")
   154→   154→    ch_params.parm("value4").set(5.5)
   155→   155→
   156→   156→    # ── 4. Spring Solver ─────────────────────────────────
   157→   157→    ch_spring = chop_net.createNode("spring", "inertia_solver")
   158→   158→    ch_spring.setInput(0, ch_input)
   159→   159→    ch_spring.parm("springk").setExpression('ch("../solver_params/value0")')
   160→   160→    ch_spring.parm("springdamp").setExpression('ch("../solver_params/value1")')
   161→   161→    ch_spring.setComment("Spring solver\nDamping driven by rig weight")
   162→   162→    ch_spring.setGenericFlag(hou.nodeFlag.DisplayComment, True)
   163→   163→
   164→   164→    # ── 5. Lag Solver ────────────────────────────────────
   165→   165→    ch_lag = chop_net.createNode("lag", "operator_delay")
   166→   166→    ch_lag.setInput(0, ch_spring)
   167→   167→    ch_lag.parm("lagmethod").set(1)  # Lag by frames
   168→   168→    ch_lag.parm("lag").setExpression('ch("../solver_params/value2")')
   169→   169→    ch_lag.setComment("Operator reaction time\nHeavier rig = more lag")
   170→   170→    ch_lag.setGenericFlag(hou.nodeFlag.DisplayComment, True)
   171→   171→
   172→   172→    # ── 6. Handheld Noise (Optional) ─────────────────────
   173→   173→    ch_noise = chop_net.createNode("noise", "handheld_shake")
   174→   174→    ch_noise.parm("amp").setExpression('ch("../solver_params/value3")')
   175→   175→    ch_noise.parm("freq").setExpression('ch("../solver_params/value4")')
   176→   176→    ch_noise.parm("type").set(4)  # Sparse noise — more natural than Perlin
   177→   177→    ch_noise.parm("seed").setExpression("$F * 0.1")
   178→   178→    ch_noise.setComment("Handheld camera shake\nAmplitude inversely proportional to weight")
   179→   179→    ch_noise.setGenericFlag(hou.nodeFlag.DisplayComment, True)
   180→   180→
   181→   181→    # ── 7. Enable toggle for noise ───────────────────────
   182→   182→    ch_switch = chop_net.createNode("switch", "handheld_enable")
   183→   183→    ch_switch.setInput(0, ch_lag)      # Off: just spring + lag
   184→   184→    ch_switch.setInput(1, ch_noise)    # On: adds noise
   185→   185→    ch_switch.parm("index").set(0)     # Default: off
   186→   186→
   187→   187→    # ── 8. Math: combine lag output + noise ──────────────
   188→   188→    ch_math = chop_net.createNode("math", "combine_motion")
   189→   189→    ch_math.setInput(0, ch_lag)
   190→   190→    ch_math.setInput(1, ch_switch)
   191→   191→    ch_math.parm("chopop").set(1)  # Add
   192→   192→    ch_math.setComment("Combine spring/lag with optional shake")
   193→   193→    ch_math.setGenericFlag(hou.nodeFlag.DisplayComment, True)
   194→   194→
   195→   195→    # ── 9. Output ────────────────────────────────────────
   196→   196→    ch_output = chop_net.createNode("null", "OUT_biomechanics")
   197→   197→    ch_output.setInput(0, ch_math)
   198→   198→    ch_output.setDisplayFlag(True)
   199→   199→    ch_output.setRenderFlag(True)
   200→   200→    ch_output.setComment("OUTPUT: Physical camera motion\nExport to FluidHead rotateXYZ")
   201→   201→    ch_output.setGenericFlag(hou.nodeFlag.DisplayComment, True)
   202→   202→
   203→   203→    # ── 10. Layout ───────────────────────────────────────
   204→   204→    chop_net.layoutChildren()
   205→   205→
   206→   206→    # ── 11. Create HDA from CHOP network ─────────────────
   207→   207→    hda_def = chop_net.createDigitalAsset(
   208→   208→        name="cinema::chops_biomechanics",
   209→   209→        hda_file_name=hda_path,
   210→   210→        description="Cinema Camera Biomechanics v1.0",
   211→   211→        min_num_inputs=1,
   212→   212→        max_num_inputs=1,
   213→   213→        version="1.0",
   214→   214→    )
   215→   215→
   216→   216→    hda_node = chop_net
   217→   217→    hda_type = hda_node.type()
   218→   218→    hda_def = hda_type.definition()
   219→   219→
   220→   220→    # ── 12. Promote parameters to HDA interface ──────────
   221→   221→    ptg = hda_node.parmTemplateGroup()
   222→   222→
   223→   223→    # Weight-driven folder
   224→   224→    weight_folder = hou.FolderParmTemplate(
   225→   225→        "weight_folder", "Rig Weight",
   226→   226→        folder_type=hou.folderType.Tabs,
   227→   227→    )
   228→   228→    weight_folder.addParmTemplate(hou.FloatParmTemplate(
   229→   229→        "combined_weight_kg", "Combined Weight (kg)", 1,
   230→   230→        default_value=(7.5,),
   231→   231→        min=1.0, max=30.0,
   232→   232→        help="Total rig weight. Reads from USD cinema:rig:combinedWeightKg if available.",
   233→   233→    ))
   234→   234→    weight_folder.addParmTemplate(hou.FloatParmTemplate(
   235→   235→        "moment_arm_cm", "Moment Arm (cm)", 1,
   236→   236→        default_value=(18.0,),
   237→   237→        min=5.0, max=50.0,
   238→   238→        help="Distance from tripod pivot to center of mass. Computed from sensor offset + half lens length.",
   239→   239→    ))
   240→   240→    ptg.append(weight_folder)
   241→   241→
   242→   242→    # Solver folder
   243→   243→    solver_folder = hou.FolderParmTemplate(
   244→   244→        "solver_folder", "Solver",
   245→   245→        folder_type=hou.folderType.Tabs,
   246→   246→    )
   247→   247→    solver_folder.addParmTemplate(hou.FloatParmTemplate(
   248→   248→        "spring_constant", "Spring Constant", 1,
   249→   249→        default_value=(15.0,),
   250→   250→        min=1.0, max=30.0,
   251→   251→        help="Higher = snappier response. Auto-derived from weight if 'Auto Derive' is on.",
   252→   252→    ))
   253→   253→    solver_folder.addParmTemplate(hou.FloatParmTemplate(
   254→   254→        "damping_ratio", "Damping Ratio", 1,
   255→   255→        default_value=(0.5,),
   256→   256→        min=0.0, max=1.0,
   257→   257→        help="0 = undamped oscillation, 1 = critically damped. Heavier rigs need more damping.",
   258→   258→    ))
   259→   259→    solver_folder.addParmTemplate(hou.FloatParmTemplate(
   260→   260→        "lag_frames", "Lag (frames)", 1,
   261→   261→        default_value=(2.25,),
   262→   262→        min=0.0, max=10.0,
   263→   263→        help="Operator reaction delay. Heavy rig = slower response.",
   264→   264→    ))
   265→   265→    solver_folder.addParmTemplate(hou.ToggleParmTemplate(
   266→   266→        "auto_derive", "Auto Derive from Weight",
   267→   267→        default_value=True,
   268→   268→        help="When on, spring/damping/lag auto-compute from combined_weight_kg using biomechanics.py derivation.",
   269→   269→    ))
   270→   270→    ptg.append(solver_folder)
   271→   271→
   272→   272→    # Handheld folder
   273→   273→    shake_folder = hou.FolderParmTemplate(
   274→   274→        "shake_folder", "Handheld Shake",
   275→   275→        folder_type=hou.folderType.Tabs,
   276→   276→    )
   277→   277→    shake_folder.addParmTemplate(hou.ToggleParmTemplate(
   278→   278→        "enable_handheld", "Enable Handheld Shake",
   279→   279→        default_value=False,
   280→   280→    ))
   281→   281→    shake_folder.addParmTemplate(hou.FloatParmTemplate(
   282→   282→        "shake_amplitude_deg", "Amplitude (deg)", 1,
   283→   283→        default_value=(0.2,),
   284→   284→        min=0.0, max=2.0,
   285→   285→        help="Peak random rotation. Inversely proportional to weight.",
   286→   286→    ))
   287→   287→    shake_folder.addParmTemplate(hou.FloatParmTemplate(
   288→   288→        "shake_frequency_hz", "Frequency (Hz)", 1,
   289→   289→        default_value=(5.5,),
   290→   290→        min=1.0, max=15.0,
   291→   291→        help="Dominant shake frequency. Lighter rigs shake faster.",
   292→   292→    ))
   293→   293→    ptg.append(shake_folder)
   294→   294→
   295→   295→    hda_def.setParmTemplateGroup(ptg)
   296→   296→
   297→   297→    # ── 13. Python auto-derive callback ──────────────────
   298→   298→    auto_derive_script = '''
   299→   299→import hou
   300→   300→import sys
   301→   301→import os
   302→   302→
   303→   303→# Ensure cinema_camera package is importable
   304→   304→cinema_path = os.environ.get("CINEMA_CAMERA_PATH", "")
   305→   305→python_path = os.path.join(cinema_path, "python")
   306→   306→if python_path not in sys.path:
   307→   307→    sys.path.insert(0, python_path)
   308→   308→
   309→   309→node = kwargs["node"]
   310→   310→
   311→   311→if node.parm("auto_derive").eval():
   312→   312→    weight = node.parm("combined_weight_kg").eval()
   313→   313→    arm = node.parm("moment_arm_cm").eval()
   314→   314→
   315→   315→    # Inline derivation (mirrors biomechanics.py logic)
   316→   316→    inertia = weight * (arm ** 2)
   317→   317→    spring_k = max(5.0, 25.0 - inertia * 0.012)
   318→   318→    damping = min(0.95, 0.6 + weight * 0.025)
   319→   319→    lag = weight * 0.3
   320→   320→    shake_amp = max(0.05, 1.5 / weight)
   321→   321→    shake_freq = max(2.0, 8.0 - weight * 0.3)
   322→   322→
   323→   323→    node.parm("spring_constant").set(spring_k)
   324→   324→    node.parm("damping_ratio").set(damping)
   325→   325→    node.parm("lag_frames").set(lag)
   326→   326→    node.parm("shake_amplitude_deg").set(shake_amp)
   327→   327→    node.parm("shake_frequency_hz").set(shake_freq)
   328→   328→'''
   329→   329→    # Attach callback to weight and moment_arm parms
   330→   330→    hda_def.addParmCallback(auto_derive_script, ("combined_weight_kg", "moment_arm_cm", "auto_derive"))
   331→   331→
   332→   332→    # ── 14. Wire internal channel references ─────────────
   333→   333→    # Connect promoted parms to internal solver_params constants
   334→   334→    solver_params = hda_node.node("solver_params")
   335→   335→    solver_params.parm("value0").setExpression('ch("../spring_constant")')
   336→   336→    solver_params.parm("value1").setExpression('ch("../damping_ratio")')
   337→   337→    solver_params.parm("value2").setExpression('ch("../lag_frames")')
   338→   338→    solver_params.parm("value3").setExpression('ch("../shake_amplitude_deg")')
   339→   339→    solver_params.parm("value4").setExpression('ch("../shake_frequency_hz")')
   340→   340→
   341→   341→    # Wire handheld enable toggle
   342→   342→    switch_node = hda_node.node("handheld_enable")
   343→   343→    switch_node.parm("index").setExpression('ch("../enable_handheld")')
   344→   344→
   345→   345→    # ── 15. Set HDA metadata ─────────────────────────────
   346→   346→    hda_def.setIcon("CHOP_spring")
   347→   347→    hda_def.setComment("Operator biomechanics: physically-based camera inertia")
   348→   348→    hda_def.setExtraInfo(
   349→   349→        "Cinema Camera Rig v4.0 — Biomechanics CHOPs\n"
   350→   350→        "Converts raw pan/tilt/roll into weight-realistic camera motion.\n"
   351→   351→        "Heavy rigs: high damping, slow ease-in/ease-out, less shake.\n"
   352→   352→        "Light rigs: low damping, snappy response, more shake."
   353→   353→    )
   354→   354→
   355→   355→    # ── 16. Save and clean up ────────────────────────────
   356→   356→    hda_def.save(hda_path)
   357→   357→    hda_node.matchCurrentDefinition()
   358→   358→
   359→   359→    # Remove temporary container (the HDA is saved to disk)
   360→   360→    temp_geo.destroy()
   361→   361→
   362→   362→    return hda_path
   363→   363→```

   403→   403→"""

   414→   414→def build_cop_anamorphic_flare_hda(
   415→   415→    save_dir: str = None,
   416→   416→    hda_name: str = "cinema_cop_anamorphic_flare_2.0.hda",
   417→   417→) -> str:
   418→   418→    """
   419→   419→    Build cinema::cop_anamorphic_flare::2.0 HDA in live Houdini session.
   420→   420→
   421→   421→    Internal pipeline:
   422→   422→      1. Threshold → extract bright pixels
   423→   423→      2. Generate iris kernel (11-sided polygon × squeeze)
   424→   424→      3. FFT convolution of bright pixels × iris kernel

   426→   426→      5. Chromatic fringing on ghosts
   427→   427→      6. Composite over original
   428→   428→

   430→   430→    """
   431→   431→    if save_dir is None:
   432→   432→        save_dir = os.path.join(os.environ["CINEMA_CAMERA_PATH"], "hda", "post")
   433→   433→
   434→   434→    hda_path = os.path.join(save_dir, hda_name)
   435→   435→

   439→   439→    cop_net = temp_geo.createNode("copnet", "cinema_flare_build")
   440→   440→
   441→   441→    # ── 2. Input ─────────────────────────────────────────
   442→   442→    cop_input = cop_net.createNode("null", "IN_image")
   443→   443→    cop_input.setComment("INPUT: RGBA image from render or upstream COP")
   444→   444→    cop_input.setGenericFlag(hou.nodeFlag.DisplayComment, True)
   445→   445→

   447→   447→    cop_thresh = cop_net.createNode("vopcop2gen", "bright_extract")
   448→   448→    # VEX snippet for thresholding
   449→   449→    thresh_vex = '''
   450→   450→// Threshold bright pixels for flare source
   451→   451→float lum = luminance(set(R, G, B));

   453→   453→float mask = smooth(threshold - 0.1, threshold + 0.1, lum);
   454→   454→R *= mask;
   455→   455→G *= mask;
   456→   456→B *= mask;

   459→   459→

   461→   461→    cop_kernel = cop_net.createNode("vopcop2gen", "iris_kernel")

   463→   463→    kernel_vex = '''

   465→   465→// Uses iris_blades to create polygonal pattern
   466→   466→// Squeeze ratio stretches horizontally for anamorphic
   467→   467→int blades = chi("../iris_blades");

   469→   469→float intensity = ch("../intensity");
   470→   470→
   471→   471→// Polar coordinates from pixel center
   472→   472→float cx = (float(X) / float(XRES)) * 2.0 - 1.0;
   473→   473→float cy = (float(Y) / float(YRES)) * 2.0 - 1.0;
   474→   474→cx /= max(squeeze, 0.01);  // Stretch for anamorphic
   475→   475→

   477→   477→float theta = atan2(cy, cx);
   478→   478→
   479→   479→// Polygonal iris shape
   480→   480→float blade_angle = M_TWO_PI / (float)blades;
   481→   481→float sector = theta - blade_angle * floor(theta / blade_angle + 0.5);
   482→   482→float edge = cos(M_PI / (float)blades) / cos(sector);
   483→   483→
   484→   484→float kernel_val = smooth(edge - 0.02, edge + 0.02, r);
   485→   485→kernel_val = 1.0 - kernel_val;
   486→   486→kernel_val *= intensity;
   487→   487→
   488→   488→R = kernel_val;
   489→   489→G = kernel_val;
   490→   490→B = kernel_val;
   491→   491→'''

   497→   497→    cop_convolve.setInput(1, cop_kernel)   # Iris kernel

   499→   499→    cop_convolve.setGenericFlag(hou.nodeFlag.DisplayComment, True)

   505→   505→    cop_streak.parm("sizey").set(2)   # Minimal vertical

   508→   508→
   509→   509→    # ── 7. Chromatic fringing ────────────────────────────

   511→   511→    cop_chroma.setInput(0, cop_streak)
   512→   512→    # R/G/B channel offsets for ghost chromatic aberration
   513→   513→    # Detailed VEX handles per-channel radial offset
   514→   514→
   515→   515→    # ── 8. Composite over original ───────────────────────
   516→   516→    cop_comp = cop_net.createNode("composite", "flare_over")
   517→   517→    cop_comp.setInput(0, cop_input)     # Original image
   518→   518→    cop_comp.setInput(1, cop_chroma)    # Flare layer
   519→   519→    cop_comp.parm("operation").set(0)   # Add (screen blend)
   520→   520→
   521→   521→    # ── 9. Enable switch ─────────────────────────────────
   522→   522→    cop_switch = cop_net.createNode("switch", "enable_switch")
   523→   523→    cop_switch.setInput(0, cop_input)    # Bypass: passthrough
   524→   524→    cop_switch.setInput(1, cop_comp)     # Enabled: flare applied
   525→   525→    cop_switch.parm("index").set(1)
   526→   526→
   527→   527→    # ── 10. Output ───────────────────────────────────────
   528→   528→    cop_output = cop_net.createNode("null", "OUT_flare")
   529→   529→    cop_output.setInput(0, cop_switch)
   530→   530→    cop_output.setDisplayFlag(True)
   531→   531→    cop_output.setRenderFlag(True)
   532→   532→
   533→   533→    # ── 11. Layout ───────────────────────────────────────
   534→   534→    cop_net.layoutChildren()
   535→   535→
   536→   536→    # ── 12. Create HDA ───────────────────────────────────
   537→   537→    hda_def = cop_net.createDigitalAsset(
   538→   538→        name="cinema::cop_anamorphic_flare",
   539→   539→        hda_file_name=hda_path,
   540→   540→        description="Cinema Anamorphic Flare v2.0 (FFT)",
   541→   541→        min_num_inputs=1,
   542→   542→        max_num_inputs=1,
   543→   543→        version="2.0",
   544→   544→    )
   545→   545→

   547→   547→    hda_type = hda_node.type()
   548→   548→    hda_def = hda_type.definition()

   641→   641→    Build cinema::cop_sensor_noise::1.0 HDA in live Houdini session.
   642→   642→
   643→   643→    Internal pipeline:
   644→   644→      1. Luminance extraction (per-pixel signal level)
   645→   645→      2. Photon noise: sqrt(luminance) × random (shot noise)
   646→   646→      3. Read noise: constant × (EI / native_iso) (electronic noise)
   647→   647→      4. Per-channel Bayer-pattern awareness
   648→   648→      5. Temporal coherence control
   649→   649→      6. Composite noise onto image
   650→   650→
   651→   651→    Returns: Absolute path to saved .hda file.
   652→   652→    """
   653→   653→    if save_dir is None:
   654→   654→        save_dir = os.path.join(os.environ["CINEMA_CAMERA_PATH"], "hda", "post")
   655→   655→
   656→   656→    hda_path = os.path.join(save_dir, hda_name)
   657→   657→
   658→   658→    obj = hou.node("/obj")
   659→   659→    temp_geo = obj.createNode("geo", "__cinema_noise_builder")
   660→   660→    cop_net = temp_geo.createNode("copnet", "cinema_noise_build")
   661→   661→
   662→   662→    # ── Input ────────────────────────────────────────────
   663→   663→    cop_input = cop_net.createNode("null", "IN_image")
   664→   664→
   665→   665→    # ── Noise generator (vopcop2gen with dual-gain VEX) ──
   666→   666→    cop_noise_gen = cop_net.createNode("vopcop2gen", "dual_gain_noise")
   667→   667→    cop_noise_gen.setInput(0, cop_input)
   668→   668→
   669→   669→    noise_vex = '''
   670→   670→// Dual-gain sensor noise model
   671→   671→// Physically: shot noise + read noise amplified by gain
   672→   672→float ei = ch("../exposure_index");
   673→   673→float native = ch("../native_iso");
   674→   674→float photon_amt = ch("../photon_noise_amount");
   675→   675→float read_amt = ch("../read_noise_amount");
   676→   676→float temporal = ch("../temporal_coherence");
   677→   677→
   678→   678→// Gain ratio: how far above native ISO
   679→   679→float gain = ei / max(native, 1.0);
   680→   680→
   681→   681→// Signal level (luminance of current pixel)
   682→   682→float signal = luminance(set(R, G, B));
   683→   683→
   684→   684→// Shot noise: scales with sqrt of signal (Poisson statistics)
   685→   685→float shot_noise = sqrt(max(signal, 0.0)) * photon_amt;
   686→   686→
   687→   687→// Read noise: constant floor amplified by gain
   688→   688→float read_noise = read_amt * gain * 0.01;
   689→   689→
   690→   690→// Combined noise sigma
   691→   691→float sigma = sqrt(shot_noise * shot_noise + read_noise * read_noise);
   692→   692→
   693→   693→// Per-channel noise (Bayer-pattern: green gets sqrt(2) less noise)
   694→   694→float seed_r = random(set(X, Y, $F * 1.0));
   695→   695→float seed_g = random(set(X, Y, $F * 2.0));
   696→   696→float seed_b = random(set(X, Y, $F * 3.0));
   697→   697→
   698→   698→// Temporal coherence: blend between per-frame and static
   699→   699→float static_r = random(set(X, Y, 0.0));
   700→   700→float static_g = random(set(X, Y, 1.0));
   701→   701→float static_b = random(set(X, Y, 2.0));
   702→   702→
   703→   703→float nr = fit01(lerp(seed_r, static_r, temporal), -1, 1) * sigma;
   704→   704→float ng = fit01(lerp(seed_g, static_g, temporal), -1, 1) * sigma * 0.707;
   705→   705→float nb = fit01(lerp(seed_b, static_b, temporal), -1, 1) * sigma;
   706→   706→
   707→   707→// Grain size: 1px at native, 2px at 2x, etc.
   708→   708→// (handled by upstream blur if needed — this generates per-pixel noise)
   709→   709→
   710→   710→R += nr;
   711→   711→G += ng;
   712→   712→B += nb;
   713→   713→'''
   714→   714→
   715→   715→    # ── Enable switch ────────────────────────────────────

   720→   720→
   721→   721→    # ── Output ───────────────────────────────────────────
   722→   722→    cop_output = cop_net.createNode("null", "OUT_noise")
   723→   723→    cop_output.setInput(0, cop_switch)
   724→   724→    cop_output.setDisplayFlag(True)
   725→   725→    cop_output.setRenderFlag(True)
   726→   726→
   727→   727→    cop_net.layoutChildren()
   728→   728→
   729→   729→    # ── Create HDA ───────────────────────────────────────
   730→   730→    hda_def = cop_net.createDigitalAsset(
   731→   731→        name="cinema::cop_sensor_noise",
   732→   732→        hda_file_name=hda_path,
   733→   733→        description="Cinema Sensor Noise v1.0 (Dual Gain)",
   734→   734→        min_num_inputs=1,
   735→   735→        max_num_inputs=1,

   737→   737→    )

   751→   751→    sensor_folder.addParmTemplate(hou.MenuParmTemplate(
   752→   752→        "sensor_model", "Sensor Model",

  1125→  1125→    """

  1127→  1127→    Executed through Synapse bridge.
  1128→  1128→    """
