    21→def _usd_to_json(value):
    22→    """Convert USD attribute values to JSON-serializable Python types."""
    23→    if value is None:
    24→        return None
    25→    # Scalars
    26→    if isinstance(value, (bool, int, float, str)):
    27→        return value
    28→    # Matrix types (GfMatrix4d, GfMatrix3d) -- check BEFORE generic sequence
    29→    if hasattr(value, 'GetRow'):
    30→        try:
    31→            size = 4 if hasattr(value, 'IsIdentity') else 3
    32→            return [[float(value[r][c]) for c in range(size)] for r in range(size)]
    33→        except Exception:
    34→            pass
    35→    # Tuples/vectors (GfVec2f, GfVec3f, GfVec4f, GfQuatf, etc.)
    36→    if hasattr(value, '__len__') and hasattr(value, '__getitem__'):
    37→        try:
    38→            return [float(v) for v in value]
    39→        except (TypeError, ValueError):
    40→            return [_usd_to_json(v) for v in value]
    41→    # Asset paths
    42→    if hasattr(value, 'path'):
    43→        return str(value.path)
    44→    return str(value)

    50→    def _resolve_lop_node(self, node_path: str = None):
    51→        """Resolve a LOP node from path or selection."""
    52→        if not HOU_AVAILABLE:
    53→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
    54→
    55→        if node_path:
    56→            node = hou.node(node_path)
    57→            if node is None:
    58→                raise ValueError(
    59→                    f"Couldn't find a node at {node_path} -- "
    60→                    "double-check the path exists"
    61→                )
    62→            if not hasattr(node, 'stage'):
    63→                raise ValueError(
    64→                    f"The node at {node_path} isn't a LOP node -- "
    65→                    "I need a Solaris/LOP node to access the USD stage"
    66→                )
    67→            return node
    68→
    69→        # Search selection for a LOP node
    70→        for n in hou.selectedNodes():
    71→            if hasattr(n, 'stage'):
    72→                return n
    73→
    74→        raise ValueError(
    75→            "Couldn't find a LOP node in your selection -- "
    76→            "select one in the Solaris network or specify the node path"

    79→    def _handle_get_stage_info(self, payload: Dict) -> Dict:
    80→        """Handle get_stage_info command."""
    81→        if not HOU_AVAILABLE:
    82→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
    83→
    84→        node_path = resolve_param(payload, "node", required=False)
    85→        limit = int(resolve_param_with_default(payload, "limit", 500))
    86→
    87→        from .main_thread import run_on_main
    88→
    89→        def _on_main():
    90→            if node_path:
    91→                node = hou.node(node_path)
    92→            else:
    93→                # Try to find the current LOP network
    94→                node = None
    95→                for n in hou.selectedNodes():
    96→                    if hasattr(n, 'stage'):
    97→                        node = n
    98→                        break
    99→
   100→            if node is None or not hasattr(node, 'stage'):
   101→                raise ValueError(
   102→                    "No USD stage found -- select a LOP node or pass "
   103→                    "a node path so I know which stage to look at"
   104→                )
   105→
   106→            stage = node.stage()
   107→            if stage is None:
   108→                raise ValueError(
   109→                    "That node doesn't have an active USD stage yet -- "
   110→                    "it may need to cook first, or check the LOP network is set up"
   111→                )
   112→
   113→            # Import USD types inside _on_main (only available in Houdini context)
   114→            try:
   115→                from pxr import UsdGeom, UsdLux, UsdShade
   116→                _has_pxr = True
   117→            except ImportError:
   118→                _has_pxr = False
   119→
   120→            cameras = []
   121→            lights = []
   122→            renderable_prims = 0
   123→            unassigned_material_prims = []
   124→            unresolved_references = []
   125→            prims = []
   126→
   127→            for prim in stage.Traverse():
   128→                if len(prims) >= limit:
   129→                    break
   130→                path_str = str(prim.GetPath())
   131→                type_name = str(prim.GetTypeName())
   132→                prims.append({"path": path_str, "type": type_name})
   133→
   134→                if _has_pxr:
   135→                    # Cameras
   136→                    if prim.IsA(UsdGeom.Camera):
   137→                        cameras.append(path_str)
   138→                    # Lights -- check schema and fallback to type name
   139→                    if prim.IsA(UsdLux.LightAPI) or "Light" in type_name:
   140→                        lights.append(path_str)
   141→                    # Renderable geometry
   142→                    if prim.IsA(UsdGeom.Gprim):
   143→                        renderable_prims += 1
   144→                        # Check material binding
   145→                        try:
   146→                            binding_api = UsdShade.MaterialBindingAPI(prim)
   147→                            mat, _ = binding_api.ComputeBoundMaterial()
   148→                            if not mat:
   149→                                unassigned_material_prims.append(path_str)
   150→                        except Exception:
   151→                            pass
   152→                    # Unresolved references
   153→                    try:
   154→                        if prim.HasAuthoredReferences():
   155→                            composer = prim.GetPrimStack()
   156→                            for spec in composer:
   157→                                for ref in spec.referenceList.prependedItems:
   158→                                    if ref.assetPath:
   159→                                        resolved = stage.ResolveIdentifierToEditTarget(
   160→                                            ref.assetPath
   161→                                        )
   162→                                        if not resolved:
   163→                                            unresolved_references.append({
   164→                                                "prim": path_str,
   165→                                                "asset": str(ref.assetPath),
   166→                                            })
   167→                    except Exception:
   168→                        pass
   169→
   170→            return {
   171→                "node": node.path(),
   172→                "prim_count": len(prims),
   173→                "prims": prims,
   174→                "cameras": sorted(cameras),
   175→                "lights": sorted(lights),
   176→                "renderable_prims": renderable_prims,
   177→                "unassigned_material_prims": sorted(unassigned_material_prims)[:50],
   178→                "unresolved_references": unresolved_references[:20],
   179→                "truncated": len(prims) >= limit,
   180→            }
   181→
   182→        return run_on_main(_on_main)

   231→        """Handle set_usd_attribute command -- set a USD attribute via Python LOP."""
   232→        node_path_arg = resolve_param(payload, "node", required=False)

   413→    def _handle_reference_usd(self, payload: Dict) -> Dict:
   414→        """Import a USD file into the stage via reference, payload, or sublayer.
   415→
   416→        Modes:
   417→        - 'reference': Standard USD reference composition (default)
   418→        - 'payload': Deferred-load reference — asset loads only when activated.
   419→          Preferred for heavy assets in large scenes.
   420→        - 'sublayer': Layer stacking — all prims merged into the stage.
   421→          Most reliable for Karma rendering visibility.
   422→
   423→        Note: For Karma render scenes, 'sublayer' is the most reliable import
   424→        method. 'reference' and 'payload' may require explicit purpose/kind
   425→        metadata on the imported prims for Karma visibility.
   426→        """
   427→        if not HOU_AVAILABLE:
   428→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
   429→
   430→        file_path = resolve_param(payload, "file")
   431→        prim_path = resolve_param_with_default(payload, "prim_path", "/")
   432→        mode = resolve_param_with_default(payload, "mode", "reference")
   433→        parent = resolve_param_with_default(payload, "parent", "/stage")
   434→
   435→        # Validate mode before touching hou.*
   436→        if mode not in ("sublayer", "reference", "payload"):
   437→            raise ValueError(
   438→                f"'{mode}' isn't a recognized import mode -- "
   439→                "use 'reference', 'payload', or 'sublayer'"
   440→            )
   441→
   442→        from .main_thread import run_on_main
   443→
   444→        def _on_main():
   445→            parent_node = hou.node(parent)
   446→            if parent_node is None:
   447→                raise ValueError(
   448→                    f"Couldn't find the parent node at {parent} -- "
   449→                    "verify this path exists (default is /stage)"
   450→                )
   451→
   452→            # Validate file path exists (catch missing assets early)
   453→            import os as _os
   454→            if not file_path.startswith(("$", "op:")) and not _os.path.isfile(file_path):
   455→                raise ValueError(
   456→                    f"Couldn't find the file at '{file_path}' -- "
   457→                    "double-check the path exists. If you're using a Houdini "
   458→                    "variable like $HIP or $HFS, make sure it's set correctly."
   459→                )
   460→
   461→            with hou.undos.group("SYNAPSE: reference_usd"):
   462→                if mode == "sublayer":
   463→                    node = parent_node.createNode("sublayer", "sublayer_import")
   464→                    node.parm("filepath1").set(file_path)
   465→                else:
   466→                    # Both 'reference' and 'payload' use the reference LOP node
   467→                    node = parent_node.createNode("reference", "ref_import")
   468→                    node.parm("filepath1").set(file_path)
   469→                    if prim_path != "/":
   470→                        node.parm("primpath").set(prim_path)
   471→
   472→                    # Set reference type: payload uses deferred loading
   473→                    if mode == "payload":
   474→                        reftype_parm = node.parm("reftype")
   475→                        if reftype_parm is not None:
   476→                            reftype_parm.set("payload")
   477→
   478→                result = {
   479→                    "node": node.path(),
   480→                    "file": file_path,
   481→                    "mode": mode,
   482→                    "prim_path": prim_path,
   483→                }
   484→
   485→                # Add Karma visibility advisory for non-sublayer modes
   486→                if mode in ("reference", "payload"):
   487→                    result["advisory"] = (
   488→                        "For Karma rendering, 'sublayer' is the most reliable import "
   489→                        "mode. If this asset isn't visible in renders, try switching "
   490→                        "to mode='sublayer' or ensure the imported prims have "
   491→                        "purpose='default' and kind='component' metadata."
   492→                    )
   493→
   494→                return result
   495→
   496→        return run_on_main(_on_main)

   498→    def _handle_query_prims(self, payload: Dict) -> Dict:
   499→        """Query USD stage prims with filtering by type, purpose, name pattern.
   500→
   501→        Walks the stage hierarchy from a root path and returns matching prims
   502→        with their types, paths, and metadata. Useful for discovering what's on
   503→        the stage before making changes.
   504→        """
   505→        if not HOU_AVAILABLE:
   506→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
   507→
   508→        node_path_arg = resolve_param(payload, "node", required=False)
   509→        root_path = resolve_param_with_default(payload, "root_path", "/")
   510→        prim_type = resolve_param(payload, "prim_type", required=False)
   511→        purpose = resolve_param(payload, "purpose", required=False)
   512→        name_pattern = resolve_param(payload, "name_pattern", required=False)
   513→        max_depth = int(resolve_param_with_default(payload, "max_depth", 10))
   514→        limit = int(resolve_param_with_default(payload, "limit", 100))
   515→
   516→        from .main_thread import run_on_main
   517→
   518→        def _on_main():
   519→            node = self._resolve_lop_node(node_path_arg)  # type: ignore[attr-defined]
   520→
   521→            stage = node.stage()
   522→            if stage is None:
   523→                raise ValueError(
   524→                    "That node doesn't have an active USD stage yet -- "
   525→                    "it may need to cook first, or check the LOP network is set up"
   526→                )
   527→
   528→            root_prim = stage.GetPrimAtPath(root_path)
   529→            if not root_prim.IsValid():
   530→                raise ValueError(
   531→                    f"Couldn't find a prim at {root_path} -- "
   532→                    "double-check the path on the USD stage"
   533→                )
   534→
   535→            from pxr import Usd, UsdGeom
   536→
   537→            import re
   538→            name_re = None
   539→            if name_pattern:
   540→                try:
   541→                    name_re = re.compile(name_pattern, re.IGNORECASE)
   542→                except re.error:
   543→                    name_re = None  # fall back to substring match
   544→
   545→            results = []
   546→
   547→            def _walk(prim, depth):
   548→                if depth > max_depth or len(results) >= limit:
   549→                    return
   550→
   551→                prim_type_name = prim.GetTypeName()
   552→
   553→                # Apply type filter
   554→                if prim_type:
   555→                    if prim_type_name.lower() != prim_type.lower():
   556→                        for child in prim.GetChildren():
   557→                            _walk(child, depth + 1)
   558→                        return
   559→
   560→                # Apply purpose filter
   561→                if purpose:
   562→                    imageable = UsdGeom.Imageable(prim)
   563→                    if imageable:
   564→                        prim_purpose = imageable.ComputePurpose()
   565→                        if prim_purpose != purpose:
   566→                            for child in prim.GetChildren():
   567→                                _walk(child, depth + 1)
   568→                            return
   569→
   570→                # Apply name filter
   571→                if name_pattern:
   572→                    prim_name = prim.GetName()
   573→                    if name_re:
   574→                        if not name_re.search(prim_name):
   575→                            for child in prim.GetChildren():
   576→                                _walk(child, depth + 1)
   577→                            return
   578→                    elif name_pattern.lower() not in prim_name.lower():
   579→                        for child in prim.GetChildren():
   580→                            _walk(child, depth + 1)
   581→                        return
   582→
   583→                # Prim passes all filters
   584→                entry = {
   585→                    "path": str(prim.GetPath()),
   586→                    "type": prim_type_name,
   587→                    "name": prim.GetName(),
   588→                }
   589→
   590→                kind = Usd.ModelAPI(prim).GetKind()
   591→                if kind:
   592→                    entry["kind"] = str(kind)
   593→
   594→                imageable = UsdGeom.Imageable(prim)
   595→                if imageable:
   596→                    p = imageable.ComputePurpose()
   597→                    if p:
   598→                        entry["purpose"] = str(p)
   599→                    vis = imageable.ComputeVisibility()
   600→                    if vis:
   601→                        entry["visibility"] = str(vis)
   602→
   603→                entry["active"] = prim.IsActive()
   604→                results.append(entry)
   605→
   606→                for child in prim.GetChildren():
   607→                    _walk(child, depth + 1)
   608→
   609→            _walk(root_prim, 0)
   610→
   611→            return {
   612→                "root_path": root_path,
   613→                "prim_count": len(results),
   614→                "prims": results,
   615→                "truncated": len(results) >= limit,
   616→            }

   618→        return run_on_main(_on_main)
