   182→# =============================================================================
   183→# COMMAND HANDLER REGISTRY
   184→# =============================================================================
   185→
   186→class CommandHandlerRegistry:
   187→    """
   188→    Registry for command handlers.
   189→
   190→    Allows registering handler functions for specific command types.
   191→    Supports both built-in and custom handlers.
   192→    """
   193→
   194→    def __init__(self):
   195→        self._handlers: Dict[str, Callable] = {}
   196→
   197→    def register(self, command_type: str, handler: Callable):
   198→        """Register a handler for a command type."""
   199→        self._handlers[command_type] = handler
   200→
   201→    def get(self, command_type: str) -> Optional[Callable]:
   202→        """Get handler for a command type (expects pre-normalized type)."""
   203→        return self._handlers.get(command_type)
   204→
   205→    def has(self, command_type: str) -> bool:
   206→        """Check if a handler exists for a command type."""
   207→        return self.get(command_type) is not None
   208→
   209→    @property
   210→    def registered_types(self):
   211→        """Get list of registered command types."""
   212→        return list(self._handlers.keys())

   218→class SynapseHandler(NodeHandlerMixin, UsdHandlerMixin, RenderHandlerMixin, TopsHandlerMixin, MaterialHandlerMixin, MemoryHandlerMixin, HdaHandlerMixin, CopsHandlerMixin):
   219→    """
   220→    Main command handler for the Synapse server.
   221→
   222→    Routes commands to appropriate handler functions and manages
   223→    integration with the session tracker / memory bridge.
   224→    """
   225→
   226→    def __init__(self):
   227→        self._registry = CommandHandlerRegistry()
   228→        self._session_id: Optional[str] = None
   229→        self._user_id: str = ""
   230→        self._bridge = None
   231→        self._register_handlers()
   232→
   233→    def set_session_id(self, session_id: str):
   234→        """Set the current session ID for action logging."""
   235→        self._session_id = session_id
   236→
   237→    def set_user_id(self, user_id: str):
   238→        """Set the current user ID for per-user audit trails."""
   239→        self._user_id = user_id
   240→
   241→    def _get_bridge(self):
   242→        """Lazy-load the bridge to avoid circular imports."""
   243→        if self._bridge is None:
   244→            from ..session.tracker import get_bridge
   245→            self._bridge = get_bridge()
   246→        return self._bridge
   247→
   248→    def handle(self, command: SynapseCommand) -> SynapseResponse:
   249→        """
   250→        Handle an incoming command.
   251→
   252→        Args:
   253→            command: The command to handle
   254→
   255→        Returns:
   256→            SynapseResponse with the result
   257→        """
   258→        try:
   259→            # Fast path: try raw type first (aliases are pre-populated in registry)
   260→            cmd_type = command.type
   261→            handler = self._registry.get(cmd_type)
   262→            if handler is None:
   263→                # Fallback: normalize for any aliases not in registry
   264→                cmd_type = normalize_command_type(command.type)
   265→                handler = self._registry.get(cmd_type)
   266→
   267→            if handler is None:
   268→                return SynapseResponse(
   269→                    id=command.id,
   270→                    success=False,
   271→                    error=f"I don't recognize the command '{command.type}' \u2014 try get_help to see what's available",
   272→                    sequence=command.sequence,
   273→                )
   274→
   275→            result = handler(command.payload)
   276→
   277→            # Log action asynchronously — don't block the response path
   278→            if cmd_type not in _READ_ONLY_COMMANDS:
   279→                self._submit_logs(cmd_type, command.payload, result)
   280→
   281→            return SynapseResponse(
   282→                id=command.id,
   283→                success=True,
   284→                data=result,
   285→                sequence=command.sequence,
   286→            )
   287→
   288→        except SynapseUserError as e:
   289→            # User errors: bad input, missing node, bad parm. Don't trip CB.
   290→            return SynapseResponse(
   291→                id=command.id,
   292→                success=False,
   293→                error=str(e),
   294→                sequence=command.sequence,
   295→            )
   296→        except ValueError as e:
   297→            # Legacy ValueError path -- still user error, don't trip CB.
   298→            return SynapseResponse(
   299→                id=command.id,
   300→                success=False,
   301→                error=str(e),
   302→                sequence=command.sequence,
   303→            )
   304→        except SynapseServiceError as e:
   305→            # Service errors: Houdini down, execution crash. DO trip CB.
   306→            return SynapseResponse(
   307→                id=command.id,
   308→                success=False,
   309→                error=str(e),
   310→                sequence=command.sequence,
   311→            )
   312→        except Exception as e:
   313→            return SynapseResponse(
   314→                id=command.id,
   315→                success=False,
   316→                error=f"Hit a snag processing that request: {e}",
   317→                sequence=command.sequence,
   318→            )
   319→
   320→    def _submit_logs(self, cmd_type: str, payload: Dict, result: Any):
   321→        """Submit bridge + audit log in a single executor call."""
   322→        bridge = self._get_bridge()
   323→        sid = self._session_id
   324→        uid = self._user_id
   325→        category = _CMD_CATEGORY.get(cmd_type, AuditCategory.SYNAPSE)
   326→        output = result if isinstance(result, dict) else {}
   327→
   328→        def _do_log():
   329→            if bridge and sid:
   330→                bridge.log_action(f"Executed: {cmd_type}", session_id=sid)
   331→            audit_log().log(
   332→                operation=cmd_type,
   333→                message=f"Executed {cmd_type}",
   334→                level=AuditLevel.AGENT_ACTION,
   335→                category=category,
   336→                user_id=uid,
   337→                input_data=payload,
   338→                output_data=output,
   339→            )
   340→
   341→        _log_executor.submit(_do_log)

   593→    def _handle_batch_commands(self, payload: Dict) -> Dict:
   594→        """Execute a batch of commands in declared order.
   595→
   596→        Payload:
   597→            commands: list of {type: str, payload: dict}
   598→            atomic: bool (default True) -- wrap in single undo group
   599→            stop_on_error: bool (default False) -- halt on first error
   600→        """
   601→        commands = payload.get("commands")
   602→        if not commands or not isinstance(commands, list):
   603→            raise ValueError("'commands' must be a non-empty list")
   604→
   605→        atomic = payload.get("atomic", True)
   606→        stop_on_error = payload.get("stop_on_error", False)
   607→
   608→        from .main_thread import run_on_main, _SLOW_TIMEOUT
   609→
   610→        def _on_main():
   611→            results: list = []
   612→            statuses: list = []
   613→            errors: list = []
   614→
   615→            def _run_batch():
   616→                for i, cmd_spec in enumerate(commands):
   617→                    cmd_type = cmd_spec.get("type", "")
   618→                    cmd_payload = cmd_spec.get("payload", {})
   619→                    handler = self._registry.get(cmd_type)
   620→
   621→                    if handler is None:
   622→                        err = f"Step {i}: unknown command '{cmd_type}'"
   623→                        errors.append(err)
   624→                        statuses.append("error")
   625→                        results.append(None)
   626→                        if stop_on_error:
   627→                            break
   628→                        continue
   629→
   630→                    try:
   631→                        result = handler(cmd_payload)
   632→                        results.append(result)
   633→                        statuses.append("ok")
   634→                        errors.append(None)
   635→                    except Exception as e:
   636→                        results.append(None)
   637→                        statuses.append("error")
   638→                        errors.append(f"Step {i}: {e}")
   639→                        if stop_on_error:
   640→                            break
   641→
   642→            if atomic and HOU_AVAILABLE:
   643→                with hou.undos.group("synapse_batch"):
   644→                    _run_batch()
   645→            else:
   646→                _run_batch()
   647→
   648→            return {
   649→                "results": results,
   650→                "statuses": statuses,
   651→                "errors": errors,
   652→            }
   653→
   654→        return run_on_main(_on_main, timeout=_SLOW_TIMEOUT)

   707→    def _handle_set_parm(self, payload: Dict) -> Dict:
   708→        """Handle set_parm command."""
   709→        if not HOU_AVAILABLE:
   710→            raise HoudiniUnavailableError()
   711→
   712→        node_path = resolve_param(payload, "node")
   713→        parm_name = resolve_param(payload, "parm")
   714→        value = resolve_param(payload, "value")
   715→
   716→        # Guard: reject NaN/Inf before they reach Houdini (they persist and corrupt)
   717→        if isinstance(value, float) and (math.isnan(value) or math.isinf(value)):
   718→            bad = "NaN" if math.isnan(value) else "Infinity"
   719→            raise ValueError(
   720→                f"Can't set {parm_name} to {bad} — that would corrupt the "
   721→                "parameter. Use a finite number instead."
   722→            )
   723→        if isinstance(value, (list, tuple)):
   724→            for i, v in enumerate(value):
   725→                if isinstance(v, float) and (math.isnan(v) or math.isinf(v)):
   726→                    bad = "NaN" if math.isnan(v) else "Infinity"
   727→                    raise ValueError(
   728→                        f"Can't set {parm_name} — element {i} is {bad}, which "
   729→                        "would corrupt the parameter. Use finite numbers instead."
   730→                    )
   731→
   732→        from .main_thread import run_on_main
   733→
   734→        def _on_main():
   735→            nonlocal parm_name
   736→            node = hou.node(node_path)
   737→            if node is None:
   738→                raise NodeNotFoundError(node_path)
   739→
   740→            parm = node.parm(parm_name)
   741→            # USD alias fallback -- resolve human-readable name to encoded parm
   742→            if parm is None:
   743→                usd_encoded = USD_PARM_ALIASES.get(parm_name.lower())
   744→                if usd_encoded:
   745→                    parm = node.parm(usd_encoded)
   746→                    if parm is not None:
   747→                        parm_name = usd_encoded
   748→            if parm is not None:
   749→                # If value is a list/tuple but we found a scalar parm, try the
   750→                # tuple parm instead — this lets callers set color (R,G,B) in one
   751→                # call using the base name (e.g., "base_color" with [1,0,0]).
   752→                if isinstance(value, (list, tuple)):
   753→                    parm_tuple = node.parmTuple(parm_name)
   754→                    if parm_tuple is not None and len(parm_tuple) > 1:
   755→                        parm_tuple.set(value)
   756→                        return {"node": node_path, "parm": parm_name, "value": value}
   757→                parm.set(value)
   758→                result = {"node": node_path, "parm": parm_name, "value": value}
   759→                # Lighting Law: soft warning when intensity > 1.0 on light nodes
   760→                if ("intensity" in parm_name.lower()
   761→                        and isinstance(value, (int, float)) and value > 1.0):
   762→                    try:
   763→                        if "light" in node.type().name().lower():
   764→                            result["warning"] = (
   765→                                "Intensity is usually 1.0 — consider using "
   766→                                "exposure for brightness control (Lighting Law)"
   767→                            )
   768→                    except Exception:
   769→                        pass  # Node type check is best-effort
   770→                return result
   771→
   772→            # Try as parm tuple
   773→            parm_tuple = node.parmTuple(parm_name)
   774→            if parm_tuple is not None:
   775→                if isinstance(value, (list, tuple)):
   776→                    parm_tuple.set(value)
   777→                else:
   778→                    parm_tuple.set([value] * len(parm_tuple))
   779→                result = {"node": node_path, "parm": parm_name, "value": value}
   780→                # Lighting Law: soft warning when intensity > 1.0 on light nodes
   781→                if ("intensity" in parm_name.lower()
   782→                        and isinstance(value, (int, float)) and value > 1.0):
   783→                    try:
   784→                        if "light" in node.type().name().lower():
   785→                            result["warning"] = (
   786→                                "Intensity is usually 1.0 — consider using "
   787→                                "exposure for brightness control (Lighting Law)"
   788→                            )
   789→                    except Exception:
   790→                        pass
   791→                return result
   792→
   793→            hint = _suggest_parms(node, parm_name)
   794→            raise ParameterError(node_path, parm_name, suggestion=hint.strip() if hint else "")
   795→
   796→        return run_on_main(_on_main)

   842→    def _handle_execute_python(self, payload: Dict) -> Dict:
   843→        """
   844→        Handle execute_python command.
   845→
   846→        Executes Python code in Houdini's runtime environment.
   847→        This is a standard DCC scripting pattern for automation.
   848→
   849→        Options:
   850→            dry_run (bool): Compile-only syntax check -- no execution.
   851→            atomic (bool): Wrap in undo group with rollback (default True).
   852→        """
   853→        if not HOU_AVAILABLE:
   854→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
   855→
   856→        code = resolve_param(payload, "content")
   857→        dry_run = resolve_param_with_default(payload, "dry_run", False)
   858→        atomic = resolve_param_with_default(payload, "atomic", True)
   859→
   860→        # Compile first -- catches SyntaxError for both dry_run and real runs
   861→        try:
   862→            compiled = compile(code, "<synapse_exec>", "exec")
   863→        except SyntaxError as e:
   864→            if dry_run:
   865→                return {"valid": False, "error": str(e), "dry_run": True}
   866→            raise
   867→
   868→        if dry_run:
   869→            return {"valid": True, "dry_run": True}
   870→
   871→        # Build execution namespace with Houdini access + idempotent guards
   872→        exec_globals = {"hou": hou, "__builtins__": __builtins__}
   873→        try:
   874→            from .guards import GUARD_FUNCTIONS
   875→            exec_globals.update(GUARD_FUNCTIONS)
   876→        except ImportError:
   877→            pass
   878→        exec_locals: dict = {}
   879→
   880→        from .main_thread import run_on_main, _SLOW_TIMEOUT
   881→
   882→        def _on_main():
   883→            # Execute inside undo group with smart rollback:
   884→            # - Coding bugs (NameError, SyntaxError, TypeError, AttributeError)
   885→            #   -> auto-rollback, since the script is broken and partial state is bad
   886→            # - Operational errors (render timeout, file not found, hou.OperationFailed)
   887→            #   -> keep mutations, since node creation/wiring may have succeeded
   888→            if atomic:
   889→                _needs_rollback = False
   890→                _rollback_exc = None
   891→                with hou.undos.group("synapse_execute"):
   892→                    try:
   893→                        _run_compiled(compiled, exec_globals, exec_locals)
   894→                    except _ROLLBACK_ERRORS as e:
   895→                        # Don't call performUndo() inside the undo group —
   896→                        # that raises "Cannot undo within an undo group".
   897→                        # Instead, flag for rollback after the group closes.
   898→                        _needs_rollback = True
   899→                        _rollback_exc = e
   900→                # Undo group is now closed — safe to roll back
   901→                if _needs_rollback:
   902→                    try:
   903→                        hou.undos.performUndo()
   904→                    except Exception:
   905→                        pass  # Best effort — group may already be empty
   906→                    raise _rollback_exc
   907→            else:
   908→                _run_compiled(compiled, exec_globals, exec_locals)
   909→
   910→            # Try to extract a result variable
   911→            result = exec_locals.get("result", "executed")
   912→
   913→            return {
   914→                "executed": True,
   915→                "result": str(result) if result else "executed",
   916→            }
   917→
   918→        return run_on_main(_on_main, timeout=_SLOW_TIMEOUT)

   920→    def _handle_execute_vex(self, payload: Dict) -> Dict:
   921→        """
   922→        Handle execute_vex command.
   923→
   924→        Creates a temporary attribwrangle node, sets the VEX snippet,
   925→        optionally wires an input geometry, and returns the node path.
   926→        Includes VEX error diagnosis when compilation or cooking fails.
   927→        """
   928→        if not HOU_AVAILABLE:
   929→            raise RuntimeError(_HOUDINI_UNAVAILABLE)
   930→
   931→        snippet = resolve_param(payload, "snippet")
   932→        run_over = resolve_param_with_default(payload, "run_over", "Points")
   933→        input_node = resolve_param_with_default(payload, "input_node", None)
   934→
   935→        # Pre-execution lint check (no hou.* needed)
   936→        from ..routing.vex_diagnostics import (
   937→            lint_vex_snippet, diagnose_vex_error, format_diagnosis,
   938→        )
   939→        lint_issues = lint_vex_snippet(snippet)
   940→        warnings = []
   941→        if lint_issues:
   942→            warnings = [
   943→                {"line": d.line_number, "message": d.symptom, "fix": d.fix}
   944→                for d in lint_issues
   945→            ]
   946→
   947→        from .main_thread import run_on_main, _SLOW_TIMEOUT
   948→
   949→        def _on_main():
   950→            # Find or create a working SOP context
   951→            parent = None
   952→            if input_node:
   953→                src = hou.node(input_node)
   954→                if src is not None:
   955→                    parent = src.parent()
   956→
   957→            if parent is None:
   958→                # Default to /obj -- create a temp geo container
   959→                obj = hou.node("/obj")
   960→                parent = obj.createNode("geo", "synapse_vex_temp")
   961→
   962→            wrangle = parent.createNode("attribwrangle", "synapse_vex")
   963→            wrangle.parm("snippet").set(snippet)
   964→
   965→            # Map run_over string to class menu value
   966→            run_over_map = {
   967→                "detail": 0, "points": 1, "vertices": 2, "primitives": 3,
   968→            }
   969→            class_val = run_over_map.get(run_over.lower(), 1)
   970→            wrangle.parm("class").set(class_val)
   971→
   972→            # Wire input if provided
   973→            if input_node:
   974→                src = hou.node(input_node)
   975→                if src is not None:
   976→                    wrangle.setInput(0, src)
   977→
   978→            wrangle.setDisplayFlag(True)
   979→            wrangle.setRenderFlag(True)
   980→
   981→            # Try to cook and capture any VEX errors
   982→            cook_errors = []
   983→            diagnosis_text = ""
   984→            try:
   985→                wrangle.cook(force=True)
   986→            except Exception as cook_exc:
   987→                error_msg = str(cook_exc)
   988→                cook_errors.append(error_msg)
   989→                # Run pattern-based diagnosis
   990→                diagnoses = diagnose_vex_error(error_msg, snippet, wrangle.path())
   991→                if diagnoses:
   992→                    diagnosis_text = format_diagnosis(diagnoses, snippet)
   993→
   994→            # Also check node error state (Houdini may set errors without raising)
   995→            node_errors = []
   996→            try:
   997→                if wrangle.errors():
   998→                    node_errors = list(wrangle.errors())
   999→                    if not diagnosis_text and node_errors:
  1000→                        all_errors = " | ".join(node_errors)
  1001→                        diagnoses = diagnose_vex_error(all_errors, snippet, wrangle.path())
  1002→                        if diagnoses:
  1003→                            diagnosis_text = format_diagnosis(diagnoses, snippet)
  1004→            except Exception:
  1005→                pass
  1006→
  1007→            # Clean up temp nodes on failure when no input was provided
  1008→            if (cook_errors or node_errors) and not input_node:
  1009→                try:
  1010→                    temp_parent = wrangle.parent()
  1011→                    if temp_parent and temp_parent.name().startswith("synapse_vex_temp"):
  1012→                        temp_parent.destroy()
  1013→                        wrangle = None  # Node no longer exists
  1014→                except Exception:
  1015→                    pass  # Best-effort cleanup
  1016→
  1017→            result = {
  1018→                "node": wrangle.path() if wrangle else "(cleaned up)",
  1019→                "snippet": snippet,
  1020→                "run_over": run_over,
  1021→                "class": class_val,
  1022→            }
  1023→
  1024→            if warnings:
  1025→                result["warnings"] = warnings
  1026→            if cook_errors or node_errors:
  1027→                result["errors"] = cook_errors + node_errors
  1028→            if diagnosis_text:
  1029→                result["diagnosis"] = diagnosis_text
  1030→
  1031→            return result
  1032→
  1033→        return run_on_main(_on_main, timeout=_SLOW_TIMEOUT)

  1237→    def _handle_autonomous_render(self, payload: Dict) -> Dict:
  1238→        """Run the autonomous render pipeline: Plan -> Validate -> Execute -> Evaluate -> Report.
  1239→
  1240→        Payload:
  1241→            intent (str, required): Natural-language render intent (e.g. 'render frames 1-48').
  1242→            max_iterations (int, optional): Max re-render attempts (default: 3).
  1243→            quality_threshold (float, optional): Minimum quality score 0.0-1.0 (default: 0.85).
  1244→
  1245→        Returns:
  1246→            Serialized RenderReport with plan, evaluation, decisions, and success flag.
  1247→        """
  1248→        import asyncio
  1249→        from dataclasses import asdict
  1250→
  1251→        intent = payload.get("intent")
  1252→        if not intent:
  1253→            raise ValueError(
  1254→                "We need an intent to plan the render \u2014 "
  1255→                "e.g. 'render frames 1-48' or 'render turntable at 50mm'"
  1256→            )
  1257→
  1258→        max_iterations = int(payload.get("max_iterations", 3))
  1259→        quality_threshold = float(payload.get("quality_threshold", 0.85))
  1260→
  1261→        # Late import to keep autonomy optional (no import at module level)
  1262→        from ..autonomy import (
  1263→            AutonomousDriver,
  1264→            RenderPlanner,
  1265→            PreFlightValidator,
  1266→            RenderEvaluator,
  1267→        )
  1268→
  1269→        # Adapter: bridge the sync handler registry to the async interface
  1270→        # expected by the autonomy modules.
  1271→        class _HandlerAdapter:
  1272→            """Adapts the sync handler registry to the async interface expected by autonomy modules."""
  1273→
  1274→            def __init__(self, registry):
  1275→                self._registry = registry
  1276→
  1277→            async def call(self, tool_name, params):
  1278→                handler = self._registry.get(tool_name)
  1279→                if handler is None:
  1280→                    raise ValueError(f"Couldn't find handler: {tool_name}")
  1281→                result = handler(params)
  1282→                if asyncio.iscoroutine(result):
  1283→                    return await result
  1284→                return result
  1285→
  1286→        adapter = _HandlerAdapter(self._registry)
  1287→
  1288→        # Optionally wire up memory for decision persistence
  1289→        memory = None
  1290→        try:
  1291→            from ..memory.store import get_synapse_memory
  1292→            memory = get_synapse_memory()
  1293→        except Exception:
  1294→            pass
  1295→
  1296→        planner = RenderPlanner()
  1297→        validator = PreFlightValidator(adapter)
  1298→        evaluator = RenderEvaluator()
  1299→
  1300→        driver = AutonomousDriver(
  1301→            planner=planner,
  1302→            validator=validator,
  1303→            evaluator=evaluator,
  1304→            handler_interface=adapter,
  1305→            memory_system=memory,
  1306→            max_iterations=max_iterations,
  1307→        )
  1308→
  1309→        # The autonomy pipeline is async; run it from this sync handler context.
  1310→        try:
  1311→            loop = asyncio.get_running_loop()
  1312→        except RuntimeError:
  1313→            loop = None
  1314→
  1315→        if loop and loop.is_running():
  1316→            # Already inside an event loop (e.g. hwebserver) — schedule as a task.
  1317→            import concurrent.futures
  1318→            future: concurrent.futures.Future = concurrent.futures.Future()
  1319→
  1320→            async def _run():
  1321→                try:
  1322→                    result = await driver.execute(intent)
  1323→                    future.set_result(result)
  1324→                except Exception as exc:
  1325→                    future.set_exception(exc)
  1326→
  1327→            loop.create_task(_run())
  1328→            report = future.result(timeout=600)
  1329→        else:
  1330→            report = asyncio.run(driver.execute(intent))
  1331→
  1332→        # Serialize the dataclass report to a plain dict
  1333→        def _serialize(obj):
  1334→            """Convert dataclass / enum / datetime to JSON-safe dict."""
  1335→            if hasattr(obj, "__dataclass_fields__"):
  1336→                return {k: _serialize(v) for k, v in asdict(obj).items()}
  1337→            if isinstance(obj, dict):
  1338→                return {k: _serialize(v) for k, v in obj.items()}
  1339→            if isinstance(obj, list):
  1340→                return [_serialize(v) for v in obj]
  1341→            if hasattr(obj, "value"):  # Enum
  1342→                return obj.value
  1343→            return obj
  1344→
  1345→        return _serialize(report)
